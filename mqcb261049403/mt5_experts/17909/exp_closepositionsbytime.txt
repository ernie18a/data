//+----------------------------------------------+
//| Expert Advisor input parameters              |
//+----------------------------------------------+
input
datetime
StopTime=
D'2030.01.01 23:59'
;
//Position closing time
input
uint
Deviation_=
20
;
//max price deviation in points
