A set of Fibo levels (pivot points) built on candles of the timeframe specified in the inputs.
//+----------------------------------------------+
//|  INDICATOR INPUT PARAMETERS                  |
//+----------------------------------------------+
input
ENUM_TIMEFRAMES
TimeFrame=
PERIOD_H12
;
// Chart period
input
double
FiboRatio1=
0.99
;
input
double
FiboRatio2=
0.382
;
input
double
FiboRatio3=
0.618
;
input
int
Shift=
0
;
// Horizontal indicator shift in bars
