# GEEN Smart Signal

Source: https://www.tradingview.com/script/oYvKsbhW-GEEN-Smart-Signal/

## Description

What it does

GEEN Smart Signal is not a single-indicator tool. It combines several classic analysis engines into one weighted Decision Engine that scores every trade candidate from 0 to 100, then only prints signals that pass a minimum confidence threshold. Every signal comes with a full breakdown showing exactly why it was accepted.

How it works

A signal candidate is generated by an ATR trailing-stop flip (with optional Heikin Ashi smoothing of the calculation source). The candidate is then evaluated by 8 engines, each contributing a weighted score:

Market Structure (20 pts) — pivot-based HH/HL/LH/LL classification, BOS and CHoCH detection
Trend (20 pts) — EMA 50/100/200 stack, classified into 5 states from strong bullish to strong bearish
Momentum (15 pts) — RSI position + ADX strength, used as confirmation only
Volume (15 pts) — current volume vs. 20-bar average, rewarding volume spikes
Liquidity (10 pts) — liquidity sweeps of prior swings, price inside a Demand/Supply zone or FVG, and Premium/Discount location vs. equilibrium
Volatility (10 pts) — ATR vs. its average, filtering out dead markets
Multi-Timeframe (10 pts) — 1H/4H/D trend alignment (closed-bar data only)
Risk (10 pts) — estimated reward-to-risk toward the nearest opposing swing

The total is normalized to 100. Below the minimum threshold (default 60) the signal is rejected (WAIT). 60–75 prints as weak, 75–85 as good, above 85 as strong. Clicking any signal arrow shows the per-engine score breakdown, entry, ATR stop, and 1R/2R/3R targets.

Chart elements

Structure labels (HH/HL/LH/LL, BOS, CHoCH), auto Order Blocks with mitigation removal, Fair Value Gaps, Equal Highs/Lows (EQH/EQL), session Kill Zones (Asia/London/New York, with an optional session filter), a main panel (decision, confidence, trend, momentum, risk, entry/SL/TP, RR, 5-timeframe view, active session, SMT check vs. a correlated symbol), and a monthly statistics panel that tracks how many signals reached TP1/TP2/TP3 or hit the stop — so you can measure performance yourself on any symbol and timeframe.

Anti-repaint design

Signals are confirmed on bar close only, higher-timeframe data uses closed bars with lookahead off, and structure breaks are evaluated on confirmed closes.

How to use

Works on any symbol and timeframe. Start with defaults, or raise the minimum confidence and enable the London/New York session filter for intraday trading. Alerts are included for buy/sell and for strong (85+) signals. This tool is for educational purposes and is not financial advice; no indicator guarantees results — always use proper risk management.

var float tTP2 = na
var float tTP3 = na
var int   tDir = 0
var int   tMax = 0
var int   tOpenBar = 0

var int cTotal = 0
var int cTP1 = 0
var int cTP2 = 0
var int cTP3 = 0
var int cSL  = 0

if month != month[1] or year != year[1]
    cTotal := 0
    cTP1 := 0
    cTP2 := 0
    cTP3 := 0
    cSL  := 0

if tDir != 0 and bar_index > tOpenBar
    hitSL = tDir == 1 ? low <= tSL : high >= tSL
    hit1  = tDir == 1 ? high >= tTP1 : low <= tTP1
    hit2  = tDir == 1 ? high >= tTP2 : low <= tTP2
    hit3  = tDir == 1 ? high >= tTP3 : low <= tTP3
    if hitSL and tMax == 0
        cSL  += 1
        tDir := 0
    else
        if hit1 and tMax < 1
            tMax := 1
            cTP1 += 1
        if hit2 and tMax < 2
            tMax := 2
            cTP2 += 1
        if hit3 and tMax < 3
            tMax := 3
            cTP3 += 1
        if tMax == 3 or (hitSL and tMax > 0)
            tDir := 0

// بيانات آخر إشارة مقبولة (تظهر في لوحة GEEN)
var float pEntry = na
var float pSL  = na
var float pTP  = na
var int   pDir = 0
var int   pScore = 0

// ═══════════════ الرسم: الأسهم + الوقف والأهداف + التفاصيل ═══════════════
upColor = color.new(#26a69a, 0)
dnColor = color.new(#ef5350, 0)

if buyOK or sellOK
    d      = buyOK ? 1 : -1
    entry  = close
    slDist = atrVal * slMult
    slPx   = entry - d * slDist
    tp1    = entry + d * slDist * 1
    tp2    = entry + d * slDist * 2
    tp3    = entry + d * slDist * 3
    // تسجيل الصفقة
    tDir := d
    tSL  := slPx
    tTP1 := tp1
    tTP2 := tp2
    tTP3 := tp3
    tMax := 0
    tOpenBar := bar_index
    cTotal += 1
    pEntry := entry
    pSL := slPx
    pTP := tp3
    pDir := d
    pScore := score
    // بطاقة التفاصيل (تظهر عند الضغط على السهم)
    tip = (d == 1 ? "🟢 BUY " : "🔴 SELL ") + str.tostring(score) + "% — إشارة " + tierTxt + " " + starTxt
      + "\n────── تفاصيل الصفقة ──────"
      + "\nدخول: "  + str.tostring(entry, format.mintick)
      + "\nوقف: "   + str.tostring(slPx,  format.mintick)
      + "\nهدف 1: " + str.tostring(tp1,   format.mintick)
      + "\nهدف 2: " + str.tostring(tp2,   format.mintick)
      + "\nهدف 3: " + str.tostring(tp3,   format.mintick)
      + "\nRR حتى الهدف الأخير: 1 : 3"
      + "\n────── تحليل المحركات ──────"
      + "\nالاتجاه: "       + str.tostring(s1) + "/20 (" + trendTxt + ")"
      + "\nالهيكل السعري: " + str.tostring(s2) + "/20"
      + "\nالزخم: "         + str.tostring(s3) + "/15 (RSI " + str.tostring(rsiVal, "#.#") + " / ADX " + str.tostring(adxV, "#.#") + ")"
      + "\nالحجم: "         + str.tostring(s4) + "/15"
      + "\nالسيولة: "       + str.tostring(s5) + "/10"
      + "\nالتذبذب: "       + str.tostring(s6) + "/10"
      + "\nالفريمات: "      + str.tostring(s7) + "/10"
      + "\nالمخاطرة: "      + str.tostring(s8) + "/10"
    label.new(bar_index, d == 1 ? low : high, "",
              style = d == 1 ? label.style_triangleup : label.style_triangledown,
              yloc  = d == 1 ? yloc.belowbar : yloc.abovebar,
              color = d == 1 ? upColor : dnColor, size = size.small, tooltip = tip)
    label.new(bar_index, d == 1 ? low - atrVal * 1.2 : high + atrVal * 1.2, str.tostring(score) + "% " + starTxt,
              style = label.style_none, textcolor = color.new(#f5c542, 0), size = size.small, tooltip = tip)
    if showTargets
        ext = 12
        line.new(bar_index, slPx, bar_index + ext, slPx, color = color.new(#ef5350, 0), width = 2)
        line.new(bar_index, tp1,  bar_index + ext, tp1,  color = color.new(#26a69a, 30), style = line.style_dotted)
        line.new(bar_index, tp2,  bar_index + ext, tp2,  color = color.new(#26a69a, 30), style = line.style_dotted)
        line.new(bar_index, tp3,  bar_index + ext, tp3,  color = color.new(#26a69a, 30), style = line.style_dotted)
        label.new(bar_index + ext, slPx, "وقف",  style = label.style_none, textcolor = color.new(#ef5350, 0),  size = size.small)
        label.new(bar_index + ext, tp1, "هدف 1", style = label.style_none, textcolor = color.new(#26a69a, 20), size = size.small)
        label.new(bar_index + ext, tp2, "هدف 2", style = label.style_none, textcolor = color.new(#26a69a, 20), size = size.small)
        label.new(bar_index + ext, tp3, "هدف 3", style = label.style_none, textcolor = color.new(#26a69a, 20), size = size.small)

// إشارة مرفوضة من محرك القرار (اختياري)
if rejected and showRejected
    label.new(bar_index, candDir == 1 ? low : high, "×",
              style = label.style_none, yloc = candDir == 1 ? yloc.belowbar : yloc.abovebar,
              textcolor = color.new(color.gray, 30), size = size.small,
              tooltip = "إشارة مرفوضة — الثقة " + str.tostring(score) + "%" + (score < minScore ? " أقل من الحد الأدنى " + str.tostring(minScore) + "%" : "") + (kzOK ? "" : "\nالسبب: خارج جلسات لندن / نيويورك"))

// ═══════════════ لوحة GEEN الرئيسية ═══════════════
riskTxt = atrVal < atrAvg ? "منخفضة" : atrVal < atrAvg * 1.5 ? "متوسطة" : "عالية"
momTxt  = adxV >= 25 ? "قوي" : adxV >= adxTh ? "متوسط" : "ضعيف"

var table dash = na

f_tf(tbl, row, name, dir) =>
    table.cell(tbl, 0, row, name, text_color = color.new(color.white, 25), text_size = size.small)
    table.cell(tbl, 1, row, dir == 1 ? "▲" : "▼", text_color = color.white, text_size = size.small, bgcolor = dir == 1 ? color.new(#26a69a, 30) : color.new(#ef5350, 30))

if showDash and barstate.islast
    dash := table.new(position.top_right, 2, 17, border_width = 1, border_color = color.new(color.gray, 70), bgcolor = color.new(#131722, 5))
    decTxt   = tDir == 1 ? "BUY" : tDir == -1 ? "SELL" : "WAIT"
    decColor = tDir == 1 ? color.new(#26a69a, 10) : tDir == -1 ? color.new(#ef5350, 10) : color.new(color.gray, 50)
    table.cell(dash, 0, 0, "GEEN", text_color = color.new(#f5c542, 0), text_size = size.small)
    table.cell(dash, 1, 0, "SMART SIGNAL", text_color = color.new(#f5c542, 0), text_size = size.small)
    table.cell(dash, 0, 1, "القرار", text_color = color.white, text_size = size.small)
    table.cell(dash, 1, 1, decTxt, text_color = color.white, text_size = size.small, bgcolor = decColor)
    table.cell(dash, 0, 2, "درجة الثقة", text_color = color.white, text_size = size.small)
    table.cell(dash, 1, 2, pScore > 0 ? str.tostring(pScore) + "%" : "—", text_color = color.white, text_size = size.small, bgcolor = pScore > 85 ? color.new(#26a69a, 20) : pScore >= 75 ? color.new(#26a69a, 45) : color.new(color.gray, 60))
    table.cell(dash, 0, 3, "الاتجاه", text_color = color.white, text_size = size.small)
    table.cell(dash, 1, 3, trendTxt, text_color = color.white, text_size = size.small, bgcolor = trendState > 0 ? color.new(#26a69a, 40) : trendState < 0 ? color.new(#ef5350, 40) : color.new(color.gray, 60))
    table.cell(dash, 0, 4, "الزخم", text_color = color.white, text_size = size.small)
    table.cell(dash, 1, 4, momTxt, text_color = color.white, text_size = size.small)
    table.cell(dash, 0, 5, "المخاطرة", text_color = color.white, text_size = size.small)
    table.cell(dash, 1, 5, riskTxt, text_color = color.white, text_size = size.small)
    table.cell(dash, 0, 6, "الدخول", text_color = color.white, text_size = size.small)
    table.cell(dash, 1, 6, na(pEntry) ? "—" : str.tostring(pEntry, format.mintick), text_color = color.white, text_size = size.small)
    table.cell(dash, 0, 7, "الوقف", text_color = color.white, text_size = size.small)
    table.cell(dash, 1, 7, na(pSL) ? "—" : str.tostring(pSL, format.mintick), text_color = color.new(#ef5350, 10), text_size = size.small)
    table.cell(dash, 0, 8, "الهدف", text_color = color.white, text_size = size.small)
    table.cell(dash, 1, 8, na(pTP) ? "—" : str.tostring(pTP, format.mintick), text_color = color.new(#26a69a, 10), text_size = size.small)
    table.cell(dash, 0, 9, "RR", text_color = color.white, text_size = size.small)
    table.cell(dash, 1, 9, na(pEntry) ? "—" : "1 : 3", text_color = color.white, text_size = size.small)
    f_tf(dash, 10, "15m", t15)
    f_tf(dash, 11, "1H",  t60)
    f_tf(dash, 12, "4H",  t240)
    f_tf(dash, 13, "1D",  tD)
    f_tf(dash, 14, "1W",  tW)
    table.cell(dash, 0, 15, "الجلسة", text_color = color.white, text_size = size.small)
    table.cell(dash, 1, 15, sessTxt, text_color = color.white, text_size = size.small, bgcolor = inLonKZ or inNYKZ ? color.new(#2962ff, 60) : color.new(color.gray, 70))
    table.cell(dash, 0, 16, "SMT", text_color = color.white, text_size = size.small)
    table.cell(dash, 1, 16, smtTxt, text_color = color.white, text_size = size.small)

// ═══════════════ لوحة إحصائيات الشهر ═══════════════
var table stats = na
if showStats and barstate.islast
    stats := table.new(position.bottom_right, 2, 7, border_width = 1, border_color = color.new(color.gray, 70), bgcolor = color.new(#131722, 10))
    winRate = (cTP1 + cSL) > 0 ? 100.0 * cTP1 / (cTP1 + cSL) : na
    table.cell(stats, 0, 0, "📅 إحصائيات الشهر", text_color = color.new(color.white, 10), text_size = size.small)
    table.cell(stats, 1, 0, str.tostring(month) + "/" + str.tostring(year), text_color = color.new(color.white, 30), text_size = size.small)
    table.cell(stats, 0, 1, "عدد الصفقات", text_color = color.white, text_size = size.small)
    table.cell(stats, 1, 1, str.tostring(cTotal), text_color = color.white, text_size = size.small)
    table.cell(stats, 0, 2, "حققت هدف 1", text_color = color.white, text_size = size.small)
    table.cell(stats, 1, 2, str.tostring(cTP1), text_color = color.white, text_size = size.small, bgcolor = color.new(#26a69a, 50))
    table.cell(stats, 0, 3, "حققت هدف 2", text_color = color.white, text_size = size.small)
    table.cell(stats, 1, 3, str.tostring(cTP2), text_color = color.white, text_size = size.small, bgcolor = color.new(#26a69a, 35))
    table.cell(stats, 0, 4, "حققت 3 أهداف", text_color = color.white, text_size = size.small)
    table.cell(stats, 1, 4, str.tostring(cTP3), text_color = color.white, text_size = size.small, bgcolor = color.new(#26a69a, 20))
    table.cell(stats, 0, 5, "ضربت الستوب", text_color = color.white, text_size = size.small)
    table.cell(stats, 1, 5, str.tostring(cSL), text_color = color.white, text_size = size.small, bgcolor = color.new(#ef5350, 30))
    table.cell(stats, 0, 6, "نسبة النجاح", text_color = color.white, text_size = size.small)
    table.cell(stats, 1, 6, na(winRate) ? "—" : str.tostring(winRate, "#.#") + "%", text_color = color.white, text_size = size.small, bgcolor = na(winRate) ? color.new(color.gray, 70) : winRate >= 50 ? color.new(#26a69a, 30) : color.new(#ef5350, 30))
