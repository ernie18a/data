# MTPC 终极定制版 (直观开收盘聚合版 - 12H&0-4-8固定框版)

Source: https://www.tradingview.com/script/lVHHzElZ/

## Description

Updated the box borders; added dashed and dotted lines.

---

## Source Code

````pine
//@version=6
indicator("MTPC 终极定制版 (直观开收盘聚合版 - 12H&0-4-8固定框版)", overlay=true, max_boxes_count=500)

// ==========================================
// 1. 全自定义映射面板
// ==========================================
G_MAP = "图层1：自定义周期映射 / Custom Mapping"
use_smart_map = input.bool(true, "开启自定义周期智能路由", group=G_MAP)

en_small      = input.bool(true, "1m-5m 图表显示", group=G_MAP, inline="small")
tf_small_tg   = input.timeframe("60", "➔ 加载:", group=G_MAP, inline="small")

en_slot1      = input.bool(true, "标准映射1", group=G_MAP, inline="s1")
src_slot1     = input.timeframe("15", "当图表:", group=G_MAP, inline="s1")
tgt_slot1     = input.timeframe("240", "➔ 加载高周期:", group=G_MAP, inline="s1")

en_slot2      = input.bool(true, "标准映射2", group=G_MAP, inline="s2")
src_slot2     = input.timeframe("60", "当图表:", group=G_MAP, inline="s2")
tgt_slot2     = input.timeframe("D", "➔ 加载高周期:", group=G_MAP, inline="s2")

en_slot3      = input.bool(true, "标准映射3", group=G_MAP, inline="s3")
src_slot3     = input.timeframe("240", "当图表:", group=G_MAP, inline="s3")
tgt_slot3     = input.timeframe("W", "➔ 加载高周期:", group=G_MAP, inline="s3")

en_slot4      = input.bool(true, "标准映射4", group=G_MAP, inline="s4") 
src_slot4     = input.timeframe("D", "当图表:", group=G_MAP, inline="s4")
tgt_slot4     = input.timeframe("M", "➔ 加载高周期:", group=G_MAP, inline="s4")

en_slot5      = input.bool(true, "标准映射5", group=G_MAP, inline="s5") 
src_slot5     = input.timeframe("12", "当图表:", group=G_MAP, inline="s5")
tgt_slot5     = input.timeframe("240", "➔ 加载高周期:", group=G_MAP, inline="s5")

// 图层2：0-4-8固定框专用图层
G_ICT = "图层2：0-4-8固定框专用图层"
en_ict_layer    = input.bool(true, "显示0-4-8固定框（限15分钟图加载）", group=G_ICT)
ict_border_style= input.string("实线 (Solid)", "0-4-8框线型", options=["实线 (Solid)", "虚线 (Dashed)", "点线 (Dotted)"], group=G_ICT)
ict_up_color    = input.color(color.new(#ffeb3b, 85), "上涨填充颜色", group=G_ICT)
ict_dn_color    = input.color(color.new(#ff9800, 85), "下跌填充颜色", group=G_ICT)
ict_border      = input.color(color.new(color.white, 50), "边界线框颜色", group=G_ICT)

// 图层3：12h/1h固定框专用图层
G_L3 = "图层3：12h/1h固定框专用图层"
en_l3_layer     = input.bool(true, "显示 12h/1h固定框 (限1小时图加载)", group=G_L3)
l3_border_style = input.string("实线 (Solid)", "12H框线型", options=["实线 (Solid)", "虚线 (Dashed)", "点线 (Dotted)"], group=G_L3)
l3_up_color     = input.color(color.new(#2196f3, 85), "12H上涨填充", group=G_L3)
l3_dn_color     = input.color(color.new(#9c27b0, 85), "12H下跌填充", group=G_L3)
l3_border       = input.color(color.new(color.white, 50), "12H边框颜色", group=G_L3)

// 全局样式
G_STYLE = "全局样式设置"
calcTypeInput    = input.string("High/Low Range", "模式", options=["High/Low Range", "Open/Close Range", "OHLC"], group=G_STYLE)
borderStyleInput = input.string("实线 (Solid)", "图层1边框线型", options=["实线 (Solid)", "虚线 (Dashed)", "点线 (Dotted)"], group=G_STYLE)
upBorderColor    = input.color(#009688, "上涨边框", group=G_STYLE)
upBodyColor      = input.color(color.new(#009688, 85), "上涨填充", group=G_STYLE)
dnBorderColor    = input.color(#F44336, "下跌边框", group=G_STYLE)
dnBodyColor      = input.color(color.new(#F44336, 85), "下跌填充", group=G_STYLE)

// ==========================================
// 2. 核心路由中枢 
// ==========================================
string cur_tf = timeframe.period
int cur_tf_num = timeframe.isintraday ? timeframe.multiplier : 0
string norm_tf = timeframe.isdaily ? "D" : timeframe.isweekly ? "W" : timeframe.ismonthly ? "M" : timeframe.period

int active_mode = 0 
if use_smart_map
    if en_small and timeframe.isintraday and cur_tf_num >= 1 and cur_tf_num <= 5
        active_mode := -1
    else if en_slot1 and norm_tf == src_slot1
        active_mode := 1
    else if en_slot2 and norm_tf == src_slot2
        active_mode := 2
    else if en_slot3 and norm_tf == src_slot3
        active_mode := 3
    else if en_slot4 and norm_tf == src_slot4
        active_mode := 4
    else if en_slot5 and norm_tf == src_slot5 
        active_mode := 5

// --- 获取标准映射数据 ---
string target_tf = active_mode == -1 ? tf_small_tg : active_mode == 1 ? tgt_slot1 : active_mode == 2 ? tgt_slot2 : active_mode == 3 ? tgt_slot3 : active_mode == 4 ? tgt_slot4 : active_mode == 5 ? tgt_slot5 : "D"
[so, sh, sl, sc, st] = request.security(syminfo.tickerid, target_tf, [open, high, low, close, time], gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_on)

// --- 变量分发 ---
bool show_layer1 = (active_mode != 0)
bool f_is_new = (ta.change(st) != 0)

// ==========================================
// 3. 高周期重构核心组件 (0-4-8图层 & 图层3固定时区)
// ==========================================
// 0-4-8 固定图层组件 (每4小时切分)
int ict_block = math.floor(hour / 4)
bool is_new_ict = ta.change(ict_block) != 0
var float io=na, var float ih=na, var float il=na, var float ic=na
if is_new_ict
    io := open, ih := high, il := low, ic := close
else
    ih := math.max(ih, high), il := math.min(il, low), ic := close
bool show_ict = en_ict_layer and cur_tf == "15"

// 图层3 组件 (12H固定框组件，固定0-12和12-24区间切分)
int l3_block = math.floor(hour / 12)
bool is_new_l3 = ta.change(l3_block) != 0
var float l3_o=na, var float l3_h=na, var float l3_l=na, var float l3_c=na
if is_new_l3
    l3_o := open, l3_h := high, l3_l := low, l3_c := close
else
    l3_h := math.max(l3_h, high), l3_l := math.min(l3_l, low), l3_c := close
bool show_l3 = en_l3_layer and cur_tf == "60"

// ==========================================
// 4. 绘图引擎 (封装箱体渲染单元)
// ==========================================
getLineStyle(string style_str) =>
    switch style_str
        "虚线 (Dashed)" => line.style_dashed
        "点线 (Dotted)" => line.style_dotted
        => line.style_solid

renderBoxLayer(box_main, box_sec, _is_new, _left_idx, _right_idx, _o, _h, _l, _c, _upB, _upF, _dnB, _dnF, _style) =>
    _bc = _c >= _o ? _upB : _dnB
    _fc = _c >= _o ? _upF : _dnF
    _line_style = getLineStyle(_style)
    box _ret_main = box_main
    box _ret_sec  = box_sec
    
    if calcTypeInput != "OHLC"
        _top = calcTypeInput == "High/Low Range" ? _h : math.max(_o, _c)
        _bottom = calcTypeInput == "High/Low Range" ? _l : math.min(_o, _c)
        if _is_new
            _ret_main := box.new(_left_idx, _top, _right_idx, _bottom, xloc=xloc.bar_index, border_color=_bc, border_style=_line_style, bgcolor=_fc)
        else if not na(box_main)
            box.set_top(box_main, _top), box.set_bottom(box_main, _bottom)
            box.set_right(box_main, _right_idx), box.set_left(box_main, _left_idx)
            box.set_border_color(box_main, _bc), box.set_bgcolor(box_main, _fc)
            box.set_border_style(box_main, _line_style)
        [_ret_main, na]
    else
        _btop = math.max(_o, _c), _bbot = math.min(_o, _c)
        if _is_new
            _ret_sec  := box.new(_left_idx, _h, _right_idx, _l, xloc=xloc.bar_index, border_color=color.new(_bc, 40), border_style=line.style_dotted, bgcolor=color.new(_bc, 96))
            _ret_main := box.new(_left_idx, _btop, _right_idx, _bbot, xloc=xloc.bar_index, border_color=_bc, border_style=_line_style, bgcolor=_fc)
        else
            if not na(box_main) and not na(box_sec)
                box.set_top(box_sec, _h), box.set_bottom(box_sec, _l)
                box.set_right(box_sec, _right_idx), box.set_left(box_sec, _left_idx)
                box.set_top(box_main, _btop), box.set_bottom(box_main, _bbot)
                box.set_right(box_main, _right_idx), box.set_left(box_main, _left_idx)
                box.set_border_style(box_main, _line_style)
        [_ret_main, _ret_sec]

// --- 追踪各图层的左侧起点索引 ---
var int b1_left_idx = na
if f_is_new
    b1_left_idx := bar_index

var int b2_left_idx = na
if is_new_ict
    b2_left_idx := bar_index

var int b3_left_idx = na
if is_new_l3
    b3_left_idx := bar_index

// --- 初始化历史箱体存储变量 ---
var box b1_m = na, var box b1_s = na
var box b2_m = na, var box b2_s = na
var box b3_m = na, var box b3_s = na

// --- 图层驱动执行 ---
if show_layer1
    [m, s] = renderBoxLayer(b1_m, b1_s, f_is_new, b1_left_idx, bar_index, so, sh, sl, sc, upBorderColor, upBodyColor, dnBorderColor, dnBodyColor, borderStyleInput)
    b1_m := m, b1_s := s
else
    box.delete(b1_m), box.delete(b1_s)

if show_ict
    [m, s] = renderBoxLayer(b2_m, b2_s, is_new_ict, b2_left_idx, bar_index, io, ih, il, ic, ict_border, ic >= io ? ict_up_color : ict_dn_color, ict_border, ic >= io ? ict_up_color : ict_dn_color, ict_border_style)
    b2_m := m, b2_s := s
else
    box.delete(b2_m), box.delete(b2_s)

if show_l3
    [m, s] = renderBoxLayer(b3_m, b3_s, is_new_l3, b3_left_idx, bar_index, l3_o, l3_h, l3_l, l3_c, l3_border, l3_c >= l3_o ? l3_up_color : l3_dn_color, l3_border, l3_c >= l3_o ? l3_up_color : l3_dn_color, l3_border_style)
    b3_m := m, b3_s := s
else
    box.delete(b3_m), box.delete(b3_s)
````
