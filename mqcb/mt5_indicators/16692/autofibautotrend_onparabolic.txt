The indicator plots price channel and Fibonacci levels based on the latest
parabolic ZigZag
peaks.
Inputs:
//+------------------------------------------------+
//| Inputs                                                                |
//+------------------------------------------------+
//---- ZigZag inputs
input
ENUM_TIMEFRAMES
Timeframe=
PERIOD_H6
;
// ZigZag timeframe for the indicator calculation
input
double
Step=
0.02
;
// SAR step
input
double
Maximum=
0.2
;
// SAR maximum
input
bool
ExtremumsShift=
true
;
// extremum shift flag
//---- indicator display settings
input
string
Sirname=
"AutoFibAutoTrend_OnParabolic"
;
// Name for the indicator labels
input
bool
ShowFib=
true
;
input
color
FibColor=
clrMagenta
;
input
uint
FibSize=
1
;
//----
input
bool
ShowFibFan=
true
;
input
color
FibFanColor=
clrMediumSpringGreen
;
input
uint
FibFanSize=
1
;
//----
input
bool
ShowTrend=
true
;
input
color
TrendColor=
clrDodgerBlue
;
input
uint
TrendSize=
5
;
//+-----------------------------------+
The indicator requires compiled ZigZagOnParabolic.mq5 indicator file. Place it in <terminal_data_folder>\MQL5\Indicators.
