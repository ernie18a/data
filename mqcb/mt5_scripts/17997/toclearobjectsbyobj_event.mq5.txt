//+-----------------------------------------------+
ENUM_OBJECT OBJ_TYPE = OBJ_EVENT;    //Тип удаляемых объектов
//+-----------------------------------------------+

//+------------------------------------------------------------------+ 
//| start function                                                   |
//+------------------------------------------------------------------+
void OnStart()
  {
//---- Удаление всех объектов выбранного типа со всех графиков
   long currChart=ChartFirst();
   while(currChart>-1) 
     {
      ObjectsDeleteAll(currChart,-1,OBJ_TYPE); //удаление всех объектов выбранного типа с выбранного графика     
      currChart=ChartNext(currChart); //на основании предыдущего получим новый график 
      if(currChart<0) break;          //достигли конца списка графиков 
     }
//----
  }
