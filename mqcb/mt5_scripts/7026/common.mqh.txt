#include "DataType.mqh"
//---------------------------------------------------------------------
// Преобразование времени в порядковый номер недели
//---------------------------------------------------------------------
int WeekNum(datetime time)
{
  return(int((time+259200)/604800));
}
//---------------------------------------------------------------------
// Получение ширины графика в пикселях
//---------------------------------------------------------------------
int ChartXSize(long chart_id=0)
{
  return((int)ChartGetInteger(chart_id,CHART_WIDTH_IN_PIXELS));
}
//---------------------------------------------------------------------
// Получение высоты графика в пикселях
//---------------------------------------------------------------------
int ChartYSize(long chart_id=0)
{
  return((int)ChartGetInteger(chart_id,CHART_HEIGHT_IN_PIXELS)-1);
}
//---------------------------------------------------------------------
// Сравнение двух вещественных чисел на равенство друг другу
//---------------------------------------------------------------------
bool Compare(double num1,double num2)
{
  if(NormalizeDouble(num1-num2,8)==0) {return(true);}
  return(false);
}
//---------------------------------------------------------------------
// Преобразование времени в строку
//---------------------------------------------------------------------
string t2s(datetime time)
{
  return(TimeToString(time,TIME_DATE|TIME_SECONDS));
}
//---------------------------------------------------------------------
// Преобразование uchar составляющих argb в тип uint
//---------------------------------------------------------------------
uint argb(uchar a,uchar r,uchar g,uchar b)
{
  ARGB c(a,r,g,b);
  return(c.GetNumber());
}
//---------------------------------------------------------------------
// Преобразование uchar составляющих rgb в тип uint
//---------------------------------------------------------------------
uint rgb(uchar r,uchar g,uchar b)
{
  ARGB c(255,r,g,b);
  return(c.GetNumber());
}
