int start()
  {
//----
   for(int i = ObjectsTotal() - 1; i >= 0; i--)
     {
       if(StringFind(ObjectName(i), "WM(", 0) == 0)
         {
           ObjectDelete(ObjectName(i));
         }
     }
//----
   return(0);
  }
