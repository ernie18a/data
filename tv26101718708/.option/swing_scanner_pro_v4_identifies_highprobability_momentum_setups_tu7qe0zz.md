## Description

This chart demonstrates signals generated by Swing Scanner Pro V4. The scanner combines trend structure, RSI momentum, volume expansion, ATR volatility, and proximity to 52-week highs. Market regime filters require QQQ to remain above its 200-day moving average and optionally avoid periods of elevated VIX. Qualified setups include ATR-based stop-loss, profit targets, and risk/reward calculations. The goal is to identify momentum stocks with favorable swing-trading characteristics while reducing exposure during unfavorable market conditions.

## Source Code

````pine
// @version=6
indicator("Swing Scanner Pro V4", overlay=false, max_lines_count=50)

// ══════════════════════════════════════════════════════════════
// SETTINGS
// ══════════════════════════════════════════════════════════════
grpMarket = "Market Regime"
grpSignal = "Signal Filters"
grpRisk   = "Risk Management"

scanTF = input.timeframe("D", "Scan Timeframe")
spyTF  = input.timeframe("D", "Regime Timeframe")

useSPYFilter = input.bool(true,  "Require QQQ > 200 SMA (Bull Market Only)", group=grpMarket)
useVIXFilter = input.bool(true,  "Skip signals when VIX > threshold",         group=grpMarket)
vixThreshold = input.float(30.0, "VIX Threshold", minval=10, maxval=60,       group=grpMarket)

rsiLow   = input.int(50,    "RSI Min",                    minval=30, maxval=60, group=grpSignal)
rsiHigh  = input.int(72,    "RSI Max",                    minval=60, maxval=85, group=grpSignal)
volMult  = input.float(1.5, "Volume Surge Multiplier",    minval=1.0,           group=grpSignal)
atrMin   = input.float(1.5, "ATR% Min",                   minval=0.5,           group=grpSignal)
atrMax   = input.float(8.0, "ATR% Max",                   minval=2.0,           group=grpSignal)
distMax  = input.float(15.0,"Max % Below 52w High",       minval=5.0,           group=grpSignal)
minScore = input.int(3,     "Min Score to Qualify (1-5)", minval=1, maxval=5,   group=grpSignal)

atrStopMult = input.float(1.5, "Stop  = ATR x", minval=0.5, maxval=4.0, group=grpRisk)
atrTgtMult  = input.float(3.0, "Target = ATR x", minval=1.0, maxval=8.0, group=grpRisk)

// ══════════════════════════════════════════════════════════════
// USER-DEFINED TYPE (UDT) & WATCHLIST ARRAY
// ══════════════════════════════════════════════════════════════
type TickerData
    string name
    bool   buySignal
    int    score
    float  stopPrice
    float  targetPrice
    float  riskReward

// Simply add or remove tickers here! The entire script adapts automatically.
var string[] watchlist = array.from(
     "NASDAQ:AAPL", "NASDAQ:MSFT", "NASDAQ:NVDA", "NASDAQ:AMD", "NASDAQ:TSLA",
     "NASDAQ:AMZN", "NASDAQ:META", "NASDAQ:GOOGL", "NYSE:PLTR", "NASDAQ:AVGO"
     )

// ══════════════════════════════════════════════════════════════
// MARKET REGIME
// ══════════════════════════════════════════════════════════════
qqqClose  = request.security("NASDAQ:QQQ", spyTF, close,            lookahead=barmerge.lookahead_off)
qqqSMA200 = request.security("NASDAQ:QQQ", spyTF, ta.sma(close,200), lookahead=barmerge.lookahead_off)
vixClose  = request.security("CBOE:VIX",   spyTF, close,            lookahead=barmerge.lookahead_off)

bullMarket = not useSPYFilter or (not na(qqqClose) and not na(qqqSMA200) and qqqClose > qqqSMA200)
vixOK      = not useVIXFilter or (not na(vixClose) and vixClose < vixThreshold)
regimeOK   = bullMarket and vixOK

// ══════════════════════════════════════════════════════════════
// CORE SCAN ENGINE
// ══════════════════════════════════════════════════════════════
swingScore() =>
    rsi14       = ta.rsi(close, 14)
    vol5        = ta.sma(volume, 5)
    vol20       = ta.sma(volume, 20)
    sma50       = ta.sma(close, 50)
    sma150      = ta.sma(close, 150)
    sma200      = ta.sma(close, 200)
    atr14       = ta.atr(14)
    atrPct      = close > 0 ? atr14 / close * 100 : 0.0
    high52      = ta.highest(high, 252)
    dist52      = high52 > 0 ? (close / high52 - 1.0) * 100 : -100.0
    volVelocity = vol20 > 0 ? vol5 / vol20 : 0.0

    trendScore = close > sma50 and sma50 > sma150 and sma150 > sma200 ? 1 : 0
    momScore   = rsi14 >= rsiLow and rsi14 <= rsiHigh ? 1 : 0
    volScore   = volVelocity >= volMult and vol20 > 0 ? 1 : 0
    atrScore   = atrPct >= atrMin and atrPct <= atrMax ? 1 : 0
    highScore  = dist52 >= -distMax and dist52 <= -1.0 ? 1 : 0

    [trendScore + momScore + volScore + atrScore + highScore, atr14, close]

// ══════════════════════════════════════════════════════════════
// DYNAMIC PROCESSING LOOP
// ══════════════════════════════════════════════════════════════
var TickerData[] results = array.new<TickerData>()
if barstate.isfirst
    for i = 0 to array.size(watchlist) - 1
        array.push(results, TickerData.new(array.get(watchlist, i), false, 0, na, na, na))

int activeSignalsCount = 0
int techSectorSignals  = 0
string alertMatches    = ""

for i = 0 to array.size(watchlist) - 1
    string sym = array.get(watchlist, i)
    [sc, atrVal, px] = request.security(sym, scanTF, swingScore(), lookahead=barmerge.lookahead_off, ignore_invalid_symbol=true)
    
    bool qualifies = regimeOK and not na(sc) and sc >= minScore
    float stop     = qualifies and px > 0 and atrVal > 0 ? px - atrVal * atrStopMult : na
    float target   = qualifies and px > 0 and atrVal > 0 ? px + atrVal * atrTgtMult  : na
    float rr       = not na(stop) and not na(target) and (px - stop) > 0 ? (target - px) / (px - stop) : na

    // Update state object
    TickerData item = array.get(results, i)
    item.buySignal   := qualifies
    item.score      := sc
    item.stopPrice  := stop
    item.targetPrice:= target
    item.riskReward := rr

    if qualifies
        activeSignalsCount := activeSignalsCount + 1
        // Sector correlation check (Excludes non-pure tech stock PLTR out of the 10)
        if sym != "NYSE:PLTR"
            techSectorSignals := techSectorSignals + 1
        
        alertMatches := alertMatches + str.replace(sym, "NASDAQ:", "") + " | Score:" + str.tostring(sc) + " | Stop:" + str.tostring(math.round(stop, 2)) + " | Tgt:" + str.tostring(math.round(target, 2)) + " | RR:" + str.tostring(math.round(rr, 1)) + "\n"

bool corrWarn = techSectorSignals >= 3
plot(activeSignalsCount, title="Active Signals", color=color.new(color.teal, 20), linewidth=2, style=plot.style_histogram)
hline(0, color=color.gray, linestyle=hline.style_dashed)

// ══════════════════════════════════════════════════════════════
// ALERTS & CONTROLS
// ══════════════════════════════════════════════════════════════
regimeStr = bullMarket ? "BULL" : "BEAR"
vixStr    = vixOK      ? "OK"   : "ELEVATED"
warnStr   = corrWarn   ? "\nWARN: " + str.tostring(techSectorSignals) + " correlated tech signals - size as ONE position\n" : ""

if str.length(alertMatches) > 0 and barstate.isconfirmed
    alert("SWING SETUP | Regime:" + regimeStr + " | VIX:" + vixStr + warnStr + "\n" + alertMatches, alert.freq_once_per_bar_close)

// ══════════════════════════════════════════════════════════════
// DYNAMIC GRAPHIC MATRIX DISPLAY
// ══════════════════════════════════════════════════════════════
scoreColor(sc) =>
    switch
        sc >= 5 => color.new(color.lime,  20)
        sc >= 4 => color.new(color.green, 40)
        sc >= 3 => color.new(color.teal,  50)
        => color.new(color.gray,  70)

// Grid sizing scales cleanly with the input array
var table t = table.new(position.top_right, 6, array.size(watchlist) + 3, border_width=1, border_color=color.new(color.gray, 60))

if barstate.islast
    hdrBg  = color.new(color.navy, 10)
    hdrTxt = color.white

    // ── Header Row ──────────────────────────────────────────
    table.cell(t, 0, 0, "Ticker", text_color=hdrTxt, bgcolor=hdrBg, text_size=size.small)
    table.cell(t, 1, 0, "Signal", text_color=hdrTxt, bgcolor=hdrBg, text_size=size.small)
    table.cell(t, 2, 0, "Score",  text_color=hdrTxt, bgcolor=hdrBg, text_size=size.small)
    table.cell(t, 3, 0, "Stop",   text_color=hdrTxt, bgcolor=hdrBg, text_size=size.small)
    table.cell(t, 4, 0, "Target", text_color=hdrTxt, bgcolor=hdrBg, text_size=size.small)
    table.cell(t, 5, 0, "R:R",    text_color=hdrTxt, bgcolor=hdrBg, text_size=size.small)

    // ── Regime Banner Row ───────────────────────────────────
    regimeBg  = regimeOK ? color.new(color.green, 50) : color.new(color.red, 50)
    regimeTxt = regimeOK ? "REGIME OK  " + regimeStr + "  VIX " + vixStr : "REGIME BLOCKED  " + regimeStr + "  VIX " + vixStr
    table.cell(t, 0, 1, regimeTxt, text_color=color.white, bgcolor=regimeBg, text_size=size.tiny)
    for col = 1 to 5
        table.cell(t, col, 1, "", bgcolor=regimeBg)

    // ── Structural Data Rows Loop ───────────────────────────
    for idx = 0 to array.size(results) - 1
        TickerData r = array.get(results, idx)
        int rowIdx = idx + 2
        
        table.cell(t, 0, rowIdx, r.name, text_color=color.white, bgcolor=color.new(color.gray, 70), text_size=size.tiny)
        table.cell(t, 1, rowIdx, r.buySignal ? "BUY" : "-", text_color=color.white, bgcolor=scoreColor(r.buySignal ? r.score : 0), text_size=size.tiny)
        table.cell(t, 2, rowIdx, r.buySignal ? str.tostring(r.score) + "/5" : "-", text_color=color.white, bgcolor=scoreColor(r.buySignal ? r.score : 0), text_size=size.tiny)
        table.cell(t, 3, rowIdx, r.buySignal and not na(r.stopPrice) ? str.tostring(math.round(r.stopPrice, 2)) : "-", text_color=color.new(color.red, 10), bgcolor=color.new(color.gray, 75), text_size=size.tiny)
        table.cell(t, 4, rowIdx, r.buySignal and not na(r.targetPrice) ? str.tostring(math.round(r.targetPrice, 2)) : "-", text_color=color.new(color.lime, 10), bgcolor=color.new(color.gray, 75), text_size=size.tiny)
        table.cell(t, 5, rowIdx, r.buySignal and not na(r.riskReward) ? str.tostring(math.round(r.riskReward, 1)) + "x" : "-", text_color=color.white, bgcolor=color.new(color.gray, 75), text_size=size.tiny)

    // ── Dynamic Footer Warning Row ──────────────────────────
    warnBg  = corrWarn ? color.new(color.orange, 30) : color.new(color.gray, 80)
    warnMsg = corrWarn ? "CORR WARN: " + str.tostring(techSectorSignals) + " tech signals - size as ONE trade" : "No correlation warning"
    int lastRow = array.size(watchlist) + 2
    table.cell(t, 0, lastRow, warnMsg, text_color=color.white, bgcolor=warnBg, text_size=size.tiny)
    for col = 1 to 5
        table.cell(t, col, lastRow, "", bgcolor=warnBg)
````
