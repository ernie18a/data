## Description

Opening Range 5-Min Levels (1m)
This indicator tracks the New York cash session opening range on the 1-minute chart. It calculates the high and low of the first 5 minutes after the 9:30 AM NY open (9:30–9:35), then plots horizontal levels at 0%, 25%, 50%, 75%, and 100% of that range.

## Source Code

````pine
//@version=5
indicator("OR 5m (1m) 0/25/50/75/100", overlay=true)

show = timeframe.isminutes and timeframe.multiplier == 1
tz = "America/New_York"
t = time

start = timestamp(tz, year(t), month(t), dayofmonth(t), 9, 30)
endOR = timestamp(tz, year(t), month(t), dayofmonth(t), 9, 35)
endPlot = timestamp(tz, year(t), month(t), dayofmonth(t), 11, 0)

inSession = show and t >= start and t < endOR
plotWindow = show and t >= start and t < endPlot

newDay = show and ta.change(timestamp(tz, year(t), month(t), dayofmonth(t), 0, 0))

var float hi = na
var float lo = na
var bool finished = false

if newDay
    hi := na
    lo := na
    finished := false

if inSession
    hi := na(hi) ? high : math.max(hi, high)
    lo := na(lo) ? low : math.min(lo, low)

if show and not inSession and not na(hi) and not na(lo) and not finished
    finished := true

rngReady = plotWindow and (inSession or finished) and not na(hi) and not na(lo)
rng = rngReady ? (hi - lo) : na
level(p) => rngReady ? (lo + rng * p) : na

plot(level(0.0), "0%", color=color.red, linewidth=2)
plot(level(0.25), "25%", color=color.orange, linewidth=1)
plot(level(0.50), "50%", color=color.gray, linewidth=2)
plot(level(0.75), "75%", color=color.teal, linewidth=1)
plot(level(1.0), "100%", color=color.green, linewidth=2)
````
