## Description

Bu indikatör inan dağ tarafından hazırlanmıştır ve ücretsiz kullanabilirsiniz telefon numaram whatsaaptan yada telegramdan soru sorabilirsiniz mutlaka dönerim

trend yönünü,
destek/direnç bölgelerini,
olası giriş noktalarını,
riskli bölgeleri
göstermeye çalışıyor.

1) Yeşil Çizgi (Hareketli Ortalama / Trend Çizgisi)

Bu çizgi kısa vadeli trendi gösteriyor.

Fiyat yeşil çizginin üstündeyse:
kısa vadede bullish güç artar
long işlemler daha güvenli olur
Altındaysa:
bearish baskı vardır
short daha mantıklı olabilir

Şu an:

fiyat tekrar yeşil çizginin üstüne çıkmaya çalışıyor.
bu pozitif bir sinyal.
2) Turuncu / Kırmızı Kalın Çizgi

Bu büyük trend desteği gibi çalışıyor.

Bunu:

“ana savunma hattı”
gibi düşünebilirsin.
Fiyat bunun üstünde kaldıkça:
market tamamen çökmedi demektir.
Altına sert kırarsa:
satış baskısı artabilir.
3) Kırmızı Kutular

Bunlar:

SELL / RESISTANCE bölgeleri

Yani:

fiyat oralarda zorlanabilir,
satış yiyebilir.

Şunu yaparsın:

kırmızı kutunun içine girince dikkatli olursun,
long kovalamazsın,
breakout beklenir.
4) Yeşil Kutular

Bunlar:

BUY / SUPPORT bölgeleri

Yani:

alıcıların geldiği alanlar.

En mantıklı long girişleri genelde:

yeşil kutu yakınında,
destekten tepki alınca olur.
5) Mavi Noktalar

Bunlar:

dip dönüş sinyali

Yani:

satış zayıflıyor olabilir,
tepki yükselişi gelebilir.

Tek başına yeterli değil ama:

destekle birleşirse güçlü olur.
6) Turuncu Nokta

Bu:

tepe / satış baskısı

uyarısı gibi çalışıyor.

Genelde:

yükseliş yoruluyor olabilir demektir.
ŞU AN GRAFİKTE NE GÖRÜYORUM?

Şu an:

dipten tepki alınmış,
fiyat toparlanıyor,
küçük bullish yapı oluşuyor.

Ama:

hâlâ üstte kırmızı direnç bölgesi var.

Yani:

tam güçlü breakout henüz yok.
NEREDE LONG AÇARDIM?
Daha güvenli giriş:
yeşil kutu üstünde tutunursa
ve kırmızı kutuyu kırarsa

long düşünürdüm.

Yani mantık:

destekte tutun
direnç kır
breakout al
NEREDE LONG AÇMAZDIM?
Şuralarda dikkat:
uzun yeşil mumdan sonra
kırmızı kutunun tam altında
RSI aşırı şişmişken

Çünkü:

fake breakout olabilir.
STOP NEREYE KOYULUR?

En mantıklı yer:

yeşil kutunun biraz altı

Çünkü:

destek kırılırsa
senaryo bozulmuş olur.

Profesyoneller:

“işlem fikri bozulduğu yere”
stop koyar.
TP (Kâr Alma) Neresi?

İlk hedef:

bir sonraki kırmızı kutu

Çünkü:

market orada satış yiyebilir.

## Source Code

````pine
//@version=5
indicator("PRO Futures Sniper v1.0", overlay=true, max_labels_count=500, max_lines_count=500)

//====================================================
// INPUTS
//====================================================
fastEMA        = input.int(20, "Fast EMA")
slowEMA        = input.int(200, "Trend EMA")
rsiLen         = input.int(14, "RSI Length")
atrLen         = input.int(14, "ATR Length")
atrSL          = input.float(1.5, "ATR Stop Multiplier")
rr1            = input.float(2.0, "TP1 RR")
rr2            = input.float(4.0, "TP2 RR")
volumeFilter   = input.bool(true, "Use Volume Filter")
smcFilter      = input.bool(true, "Use BOS Filter")
showOB         = input.bool(true, "Show Order Blocks")
showLiquidity  = input.bool(true, "Show Liquidity Sweeps")
showDashboard  = input.bool(true, "Show Dashboard")

//====================================================
// CORE INDICATORS
//====================================================
emaFast = ta.ema(close, fastEMA)
emaSlow = ta.ema(close, slowEMA)

rsi = ta.rsi(close, rsiLen)
atr = ta.atr(atrLen)

avgVol = ta.sma(volume, 20)
highVol = volume > avgVol * 1.3

bullTrend = close > emaSlow
bearTrend = close < emaSlow

//====================================================
// MARKET STRUCTURE
//====================================================
swingHigh = ta.pivothigh(high, 3, 3)
swingLow  = ta.pivotlow(low, 3, 3)

var float lastHigh = na
var float lastLow = na

if not na(swingHigh)
    lastHigh := swingHigh

if not na(swingLow)
    lastLow := swingLow

bullBOS = close > lastHigh
bearBOS = close < lastLow

//====================================================
// LIQUIDITY SWEEP
//====================================================
liqGrabLong = low < ta.lowest(low, 10)[1] and close > open
liqGrabShort = high > ta.highest(high, 10)[1] and close < open

plotshape(showLiquidity and liqGrabLong,
     style=shape.circle,
     color=color.aqua,
     size=size.tiny,
     location=location.belowbar,
     title="Liquidity Grab Long")

plotshape(showLiquidity and liqGrabShort,
     style=shape.circle,
     color=color.orange,
     size=size.tiny,
     location=location.abovebar,
     title="Liquidity Grab Short")

//====================================================
// ORDER BLOCK DETECTION
//====================================================
bullOB = close[1] < open[1] and close > high[1]
bearOB = close[1] > open[1] and close < low[1]

if showOB and bullOB
    box.new(
         left=bar_index-1,
         right=bar_index+5,
         top=high[1],
         bottom=low[1],
         bgcolor=color.new(color.green, 85),
         border_color=color.green)

if showOB and bearOB
    box.new(
         left=bar_index-1,
         right=bar_index+5,
         top=high[1],
         bottom=low[1],
         bgcolor=color.new(color.red, 85),
         border_color=color.red)

//====================================================
// ENTRY CONDITIONS
//====================================================
longSignal =
     bullTrend and
     ta.crossover(emaFast, emaSlow) and
     rsi > 55 and
     liqGrabLong and
     (not volumeFilter or highVol) and
     (not smcFilter or bullBOS)

shortSignal =
     bearTrend and
     ta.crossunder(emaFast, emaSlow) and
     rsi < 45 and
     liqGrabShort and
     (not volumeFilter or highVol) and
     (not smcFilter or bearBOS)

//====================================================
// TRADE LEVELS
//====================================================
longEntry = close
longSL = close - atr * atrSL
longTP1 = longEntry + ((longEntry - longSL) * rr1)
longTP2 = longEntry + ((longEntry - longSL) * rr2)

shortEntry = close
shortSL = close + atr * atrSL
shortTP1 = shortEntry - ((shortSL - shortEntry) * rr1)
shortTP2 = shortEntry - ((shortSL - shortEntry) * rr2)

//====================================================
// VISUALS
//====================================================
plot(emaFast, color=color.lime, linewidth=2, title="Fast EMA")
plot(emaSlow, color=color.red, linewidth=3, title="Trend EMA")

bgcolor(bullTrend ? color.new(color.green, 95) : na)
bgcolor(bearTrend ? color.new(color.red, 95) : na)

//====================================================
// LONG SIGNAL
//====================================================
if longSignal
    label.new(
         bar_index,
         low,
         "🚀 LONG SNIPER\n" +
         "Entry: " + str.tostring(longEntry, format.mintick) +
         "\nSL: " + str.tostring(longSL, format.mintick) +
         "\nTP1: " + str.tostring(longTP1, format.mintick) +
         "\nTP2: " + str.tostring(longTP2, format.mintick),
         style=label.style_label_up,
         color=color.lime,
         textcolor=color.black,
         size=size.normal)

    line.new(bar_index, longSL, bar_index+10, longSL, color=color.red, width=2)
    line.new(bar_index, longTP1, bar_index+10, longTP1, color=color.green, width=2)
    line.new(bar_index, longTP2, bar_index+10, longTP2, color=color.blue, width=2)

//====================================================
// SHORT SIGNAL
//====================================================
if shortSignal
    label.new(
         bar_index,
         high,
         "💥 SHORT SNIPER\n" +
         "Entry: " + str.tostring(shortEntry, format.mintick) +
         "\nSL: " + str.tostring(shortSL, format.mintick) +
         "\nTP1: " + str.tostring(shortTP1, format.mintick) +
         "\nTP2: " + str.tostring(shortTP2, format.mintick),
         style=label.style_label_down,
         color=color.red,
         textcolor=color.white,
         size=size.normal)

    line.new(bar_index, shortSL, bar_index+10, shortSL, color=color.red, width=2)
    line.new(bar_index, shortTP1, bar_index+10, shortTP1, color=color.green, width=2)
    line.new(bar_index, shortTP2, bar_index+10, shortTP2, color=color.blue, width=2)

//====================================================
// DASHBOARD
//====================================================
var table dash = table.new(position.top_right, 2, 6)

if showDashboard
    table.cell(dash, 0, 0, "Trend", bgcolor=color.black)
    table.cell(dash, 1, 0, bullTrend ? "BULLISH" : "BEARISH",
         bgcolor=bullTrend ? color.green : color.red)

    table.cell(dash, 0, 1, "RSI", bgcolor=color.black)
    table.cell(dash, 1, 1, str.tostring(rsi, "#.##"))

    table.cell(dash, 0, 2, "Volume", bgcolor=color.black)
    table.cell(dash, 1, 2, highVol ? "HIGH" : "LOW",
         bgcolor=highVol ? color.green : color.orange)

    table.cell(dash, 0, 3, "BOS", bgcolor=color.black)
    table.cell(dash, 1, 3,
         bullBOS ? "BULL BOS" :
         bearBOS ? "BEAR BOS" : "NONE")

    table.cell(dash, 0, 4, "Signal", bgcolor=color.black)
    table.cell(dash, 1, 4,
         longSignal ? "LONG" :
         shortSignal ? "SHORT" : "WAIT",
         bgcolor=
         longSignal ? color.green :
         shortSignal ? color.red :
         color.gray)

    table.cell(dash, 0, 5, "ATR", bgcolor=color.black)
    table.cell(dash, 1, 5, str.tostring(atr, "#.##"))

//====================================================
// ALERTS
//====================================================
alertcondition(longSignal,
     title="LONG SNIPER",
     message="PRO Futures Sniper LONG Signal")

alertcondition(shortSignal,
     title="SHORT SNIPER",
     message="PRO Futures Sniper SHORT Signal")
````
