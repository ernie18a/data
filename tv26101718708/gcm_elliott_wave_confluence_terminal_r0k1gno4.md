[b]Overview[/b]
The GCM Elliott Wave Confluence Terminal is an advanced, algorithmic market-mapping engine designed for systematic derivatives and crypto traders. Moving beyond standard pivot tools, the EWCT dynamically tracks market structure across three distinct fractal degrees—Minor, Intermediate, and Major—allowing traders to visualize the exact alignment of impulsive trends and corrective pullbacks in real time.

[b]Core Architecture & Mechanics[/b]
Built on the foundational principles of Elliott Wave theory and Fibonacci sequence mathematics, the EWCT automates the complex task of wave counting:

[b]• Multi-Degree Tracking:[/b] Simultaneously monitors micro-rotations (Minor), mid-term trends (Intermediate), and macro market structure (Major).

[b]• Algorithmic Confirmation:[/b] Engineered with strict bar-confirmation logic to eliminate intra-candle repainting, ensuring pivots and wave labels are locked in only when structural shifts are confirmed.

[b]• Fibonacci Confluence Matrix:[/b] Automatically calculates high-probability targets for structural completion—projecting 161.8% extensions for Wave 3 impulses and tracking 61.8% retracements for Wave 2 pullbacks.

[b]Key Features[/b]
[b]• Dynamic Terminal Dashboard:[/b] A heads-up display engineered for minimal chart clutter. It provides real-time status updates for all three degrees, cross-referencing current price action against anticipated Fibonacci targets. Color-coded status cells give you an instant read on wave validity.

[b]• Anti-Repaint Engine:[/b] Utilizes barstate.isconfirmed logic so that wave lines and labels only plot upon candle close, providing institutional-grade charting hygiene.

[b]• Complete Customization:[/b] Traders can toggle individual wave degrees on or off, adjust line opacities and widths, and scale dashboard typography to fit their specific screen layout without overwhelming the price action.

[b]How to Trade with GCM - EWCT[/b]
[b]1. Trend Alignment:[/b] Use the Major degree to dictate your directional bias. If the Major wave is in an upward sequence (Waves 1, 3, or 5), only look for long setups.

[b]2. Optimal Entries (The Pullback):[/b] Wait for a retracement on the Intermediate or Minor degree. High-probability entries occur when a corrective Wave 2 or Wave 4 approaches the dashboard's calculated Fibonacci support levels (e.g., the 61.8% retracement for Wave 2).

[b]3. Targeting (The Impulse):[/b] Once the pullback completes, ride the ensuing impulse. Use the dashboard’s Fibonacci extension targets (such as the 161.8% level for a Wave 3) as your primary take-profit zones.

[b]4. Risk Management (Invalidation):[/b] Elliott Wave structure is strictly rule-based. For example, a Wave 2 can never retrace 100% of Wave 1. If price breaks the origin point of Wave 1, the setup is invalidated. Place your stop-loss just below these structural invalidation levels.

type WaveTracker
    int last_pivot = 0 
    int dir = 0        
    int p_idx
    float p_val
    int wave_count = 0
    float w0 = 0.0
    float w1 = 0.0
    float w2 = 0.0
    float w3 = 0.0
    float w4 = 0.0
    line curr_line
    label curr_label

// EWCT Method
method processZigZagEW(WaveTracker wt, string prefix, bool hiCond, bool loCond, color cUp, color cDn, int wWidth, string wStyle, float atrOffset) =>
    color lineUpColor = color.new(cUp, 40)
    color lineDnColor = color.new(cDn, 40)
    color textUpColor = color.new(cUp, 0)
    color textDnColor = color.new(cDn, 0)

    bool isHigh = hiCond and barstate.isconfirmed
    bool isLow  = loCond and barstate.isconfirmed
    
    if hiCond and loCond and barstate.isconfirmed
        if wt.last_pivot == 1
            isHigh := true
            isLow := false
        else if wt.last_pivot == -1
            isHigh := false
            isLow := true
        else
            isHigh := true
            isLow := false

    if isHigh
        if wt.last_pivot == 1
            if high > wt.p_val
                wt.p_val := high
                wt.p_idx := bar_index
                if not na(wt.curr_line)
                    line.set_xy2(wt.curr_line, bar_index, high)
                    label.set_xy(wt.curr_label, bar_index, high + atrOffset)
            wt.last_pivot := 1 
        else if wt.last_pivot == -1
            wt.curr_line := line.new(wt.p_idx, wt.p_val, bar_index, high, color=lineUpColor, width=wWidth, style=wStyle)
            if wt.dir == 0
                wt.dir := 1
                wt.w0 := wt.p_val
                wt.w1 := high
                wt.wave_count := 1
            else if wt.dir == 1 
                if wt.wave_count == 2
                    wt.w3 := high
                    wt.wave_count := 3
                else if wt.wave_count == 4
                    wt.wave_count := 5
                else if wt.wave_count == 6
                    wt.wave_count := 7
                else if wt.wave_count == 8
                    wt.w0 := wt.p_val
                    wt.w1 := high
                    wt.wave_count := 1
                else
                    wt.w0 := wt.p_val
                    wt.w1 := high
                    wt.wave_count := 1
            else if wt.dir == -1
                if wt.wave_count == 1
                    if high >= wt.w0 
                        wt.w0 := wt.p_val
                        wt.w1 := high
                        wt.dir := 1
                        wt.wave_count := 1
                    else
                        wt.w2 := high
                        wt.wave_count := 2
                else if wt.wave_count == 3
                    if high >= wt.w1 
                        wt.w0 := wt.p_val
                        wt.w1 := high
                        wt.dir := 1
                        wt.wave_count := 1
                    else
                        wt.w4 := high
                        wt.wave_count := 4
                else if wt.wave_count == 5
                    wt.wave_count := 6
                else if wt.wave_count == 7
                    wt.wave_count := 8
                else
                    wt.w0 := wt.p_val
                    wt.w1 := high
                    wt.dir := 1
                    wt.wave_count := 1
                    
            string lblText = switch wt.wave_count
                1 => prefix + "(1)"
                2 => prefix + "(2)"
                3 => prefix + "(3)"
                4 => prefix + "(4)"
                5 => prefix + "(5)"
                6 => prefix + "(a)"
                7 => prefix + "(b)"
                8 => prefix + "(c)"
                => ""
                
            wt.curr_label := label.new(bar_index, high + atrOffset, text=lblText, textcolor=textUpColor, style=label.style_none, size=g_textSize, text_font_family=g_fontFam)
            wt.p_val := high
            wt.p_idx := bar_index
            wt.last_pivot := 1 
        else if wt.last_pivot == 0
            wt.p_val := high
            wt.p_idx := bar_index
            wt.last_pivot := 1 

    else if isLow
        if wt.last_pivot == -1
            if low < wt.p_val
                wt.p_val := low
                wt.p_idx := bar_index
                if not na(wt.curr_line)
                    line.set_xy2(wt.curr_line, bar_index, low)
                    label.set_xy(wt.curr_label, bar_index, low - atrOffset)
            wt.last_pivot := -1 
        else if wt.last_pivot == 1
            wt.curr_line := line.new(wt.p_idx, wt.p_val, bar_index, low, color=lineDnColor, width=wWidth, style=wStyle)
            if wt.dir == 0
                wt.dir := -1
                wt.w0 := wt.p_val
                wt.w1 := low
                wt.wave_count := 1
            else if wt.dir == -1
                if wt.wave_count == 2
                    wt.w3 := low
                    wt.wave_count := 3
                else if wt.wave_count == 4
                    wt.wave_count := 5
                else if wt.wave_count == 6
                    wt.wave_count := 7
                else if wt.wave_count == 8
                    wt.w0 := wt.p_val
                    wt.w1 := low
                    wt.wave_count := 1
                else
                    wt.w0 := wt.p_val
                    wt.w1 := low
                    wt.wave_count := 1
            else if wt.dir == 1
                if wt.wave_count == 1
                    if low <= wt.w0 
                        wt.w0 := wt.p_val
                        wt.w1 := low
                        wt.dir := -1
                        wt.wave_count := 1
                    else
                        wt.w2 := low
                        wt.wave_count := 2
                else if wt.wave_count == 3
                    if low <= wt.w1 
                        wt.w0 := wt.p_val
                        wt.w1 := low
                        wt.dir := -1
                        wt.wave_count := 1
                    else
                        wt.w4 := low
                        wt.wave_count := 4
                else if wt.wave_count == 5
                    wt.wave_count := 6
                else if wt.wave_count == 7
                    wt.wave_count := 8
                else
                    wt.w0 := wt.p_val
                    wt.w1 := low
                    wt.dir := -1
                    wt.wave_count := 1
                    
            string lblText = switch wt.wave_count
                1 => prefix + "(1)"
                2 => prefix + "(2)"
                3 => prefix + "(3)"
                4 => prefix + "(4)"
                5 => prefix + "(5)"
                6 => prefix + "(a)"
                7 => prefix + "(b)"
                8 => prefix + "(c)"
                => ""
                
            wt.curr_label := label.new(bar_index, low - atrOffset, text=lblText, textcolor=textDnColor, style=label.style_none, size=g_textSize, text_font_family=g_fontFam)
            wt.p_val := low
            wt.p_idx := bar_index
            wt.last_pivot := -1 
        else if wt.last_pivot == 0
            wt.p_val := low
            wt.p_idx := bar_index
            wt.last_pivot := -1

var label resLbl = na, var label supLbl = na
var float resP = na, var float supP = na

c_res = #ff0000, c_sup = #12e49e

if show_sr

        supL := line.new(bar_index-pivotLen, pl_sr, bar_index + extLen, pl_sr, color=color.new(c_sup, 30), width=3)

    if not na(resL) and high > resP and not resBroken

    if not na(supL) and low < supP and not supBroken

scalp_val = ta.hma(close, len_scalp)
bool scalp_turn_up = ta.crossover(scalp_val, scalp_val[1])
bool scalp_turn_dn = ta.crossunder(scalp_val, scalp_val[1])
var int active_state = 0 

if barstate.isconfirmed
    if scalp_turn_up
        active_state := 1
    if scalp_turn_dn
        active_state := -1

if ta.change(active_state) != 0 and show_struc
    line.delete(struct_line)
    if active_state == 1
        struct_line := line.new(bar_index-1, low[1], bar_index + 5, low, width=1, style=line.style_dashed, color=#12e49e)
    if active_state == -1
        struct_line := line.new(bar_index-1, high[1], bar_index + 5, high, width=1, style=line.style_dashed, color=#ff0000)

if show_struc and not na(struct_line)
    bool freeze_condition = (active_state == 1 and scalp_turn_dn) or (active_state == -1 and scalp_turn_up)
    if freeze_condition
        line.set_xy2(struct_line, bar_index + 4, (active_state == 1) ? low[1] : high[1])
    else
        line.set_xy2(struct_line, bar_index + 5, (active_state == 1) ? low : high)

ph = ta.pivothigh(lookback, lookback)
pl = ta.pivotlow(lookback, lookback)

if not na(ph)
    v_high := ph, v_high_idx := bar_index[lookback]
if not na(pl)
    v_low := pl, v_low_idx := bar_index[lookback]

avg_body = ta.sma(math.abs(close - open), 20)
is_disp  = math.abs(close - open) > (avg_body * displacement)

bool bos_bull   = close > v_high and is_bull and is_disp
bool choch_bull = close > v_high and not is_bull and is_disp
bool bos_bear   = close < v_low and not is_bull and is_disp
bool choch_bear = close < v_low and is_bull and is_disp

// --- Macro BOS/CHoCH & OB Logic ---
if (bos_bull or choch_bull)
    if show_ob
        for i = 1 to (bar_index - v_low_idx)
            if close[i] < open[i]
                ob_box = box.new(bar_index[i], high[i], bar_index, low[i], bgcolor=color.new(color.green, 65), border_color=color.new(color.green, 10), border_style=line.style_solid, text="OB", text_color=color.green, text_size=g_textSize, text_halign=text.align_right, text_valign=text.align_center, text_font_family=g_fontFam)
                array.push(active_bull_obs, ob_box)
                clean_array(active_bull_obs, max_zones)
                break
    line.new(v_high_idx, v_high, bar_index - 1, v_high, color=color.green, width=1, style=line.style_dashed)
    label.new(int(math.avg(v_high_idx, bar_index-1)), v_high, (bos_bull ? "BOS" : "CHoCH"), color=color.new(color.black, 100), textcolor=color.green, style=label.style_none, size=g_textSize, textalign=text.align_center, text_font_family=g_fontFam)
    is_bull := true, v_high := na 

if (bos_bear or choch_bear)
    if show_ob
        for i = 1 to (bar_index - v_high_idx)
            if close[i] > open[i]
                ob_box = box.new(bar_index[i], high[i], bar_index, low[i], bgcolor=color.new(color.red, 65), border_color=color.new(color.red, 10), border_style=line.style_solid, text="OB", text_color=color.red, text_size=g_textSize, text_halign=text.align_right, text_valign=text.align_center, text_font_family=g_fontFam)
                array.push(active_bear_obs, ob_box)
                clean_array(active_bear_obs, max_zones)
                break
    line.new(v_low_idx, v_low, bar_index - 1, v_low, color=color.red, width=1, style=line.style_dashed)
    label.new(int(math.avg(v_low_idx, bar_index-1)), v_low, (bos_bear ? "BOS" : "CHoCH"), color=color.new(color.black, 100), textcolor=color.red, style=label.style_none, size=g_textSize, textalign=text.align_center, text_font_family=g_fontFam)
    is_bull := false, v_low := na

if show_fvg
    middle_is_disp = math.abs(close[1] - open[1]) > (avg_body * displacement)
    
    if low > high[2] and close[1] > high[2] and middle_is_disp
        if (low - high[2]) > (atrValue * fvg_min_size)

            clean_array(active_bull_fvgs, max_zones)
            
    if high < low[2] and close[1] < low[2] and middle_is_disp
        if (low[2] - high) > (atrValue * fvg_min_size)

            array.push(active_bear_fvgs, fvg_box)
            clean_array(active_bear_fvgs, max_zones)

bool in_bear_ob = false

    for i = array.size(active_bull_fvgs) - 1 to 0
        b = array.get(active_bull_fvgs, i)
        if box.get_right(b) < bar_index and low <= box.get_top(b)
            box.delete(array.remove(active_bull_fvgs, i))
        else
            box.set_right(b, bar_index)

if array.size(active_bear_fvgs) > 0
    for i = array.size(active_bear_fvgs) - 1 to 0
        b = array.get(active_bear_fvgs, i)
        if box.get_right(b) < bar_index and high >= box.get_bottom(b)
            box.delete(array.remove(active_bear_fvgs, i))
        else
            box.set_right(b, bar_index)

if array.size(active_bull_obs) > 0
    for i = array.size(active_bull_obs) - 1 to 0
        b = array.get(active_bull_obs, i)
        if box.get_right(b) < bar_index and low <= box.get_top(b)
            in_bull_ob := true 
            box.delete(array.remove(active_bull_obs, i))
        else
            box.set_right(b, bar_index)

if array.size(active_bear_obs) > 0
    for i = array.size(active_bear_obs) - 1 to 0
        b = array.get(active_bear_obs, i)
        if box.get_right(b) < bar_index and high >= box.get_bottom(b)
            in_bear_ob := true 
            box.delete(array.remove(active_bear_obs, i))
        else
            box.set_right(b, bar_index)

renderRow(int rowIdx, string degreeName, WaveTracker wt, color cUp, color cDn) =>
    string waveStr = "No wave"
    color finalWaveColor = color.white
    float targetFib = 0.0
    string fibLabel = "100%"
    
    if wt.last_pivot != 0 and wt.wave_count > 0
        finalWaveColor := (wt.last_pivot == 1) ? cUp : cDn
        if wt.wave_count == 1
            waveStr := "I Wave 1 (Impulse)\n(Fib ext)"
            targetFib := close
        else if wt.wave_count == 2
            waveStr := "I Wave 2 (Pullback)\n(Fib ret)"
            float d = math.abs(wt.w1 - wt.w0)
            targetFib := wt.dir == 1 ? wt.w1 - (0.618 * d) : wt.w1 + (0.618 * d)
            fibLabel := "61.8%"
        else if wt.wave_count == 3
            waveStr := "I Wave 3 (Impulse)\n(Fib ext)"
            float d = math.abs(wt.w1 - wt.w0)
            targetFib := wt.dir == 1 ? wt.w2 + (1.618 * d) : wt.w2 - (1.618 * d)
            fibLabel := "161.8%"
        else if wt.wave_count == 4
            waveStr := "I Wave 4 (Pullback)\n(Fib ret)"
            float d = math.abs(wt.w3 - wt.w2)
            targetFib := wt.dir == 1 ? wt.w3 - (0.382 * d) : wt.w3 + (0.382 * d)
            fibLabel := "38.2%"
        else if wt.wave_count == 5
            waveStr := "I Wave 5 (Impulse)\n(Fib ext)"
            float d = math.abs(wt.w1 - wt.w0)
            targetFib := wt.dir == 1 ? wt.w4 + (1.000 * d) : wt.w4 - (1.000 * d)
            fibLabel := "100.0%"
        else if wt.wave_count == 6
            waveStr := "C Wave a (Correction)\n(Fib ret)"
            targetFib := close
        else if wt.wave_count == 7
            waveStr := "C Wave b (Correction)\n(Fib ret)"
            targetFib := close
        else if wt.wave_count == 8
            waveStr := "C Wave c (Correction)\n(Fib ext)"
            targetFib := close
    else
        waveStr := "No wave"
        finalWaveColor := color.white
        targetFib := close

    color cellColor = color.yellow
    float threshold = atrValue * 0.15 
    if math.abs(close - targetFib) <= threshold
        cellColor := color.green
    else if (wt.dir == 1 and close > targetFib) or (wt.dir == -1 and close < targetFib)
        cellColor := color.red
    else
        cellColor := color.yellow

    string statusStr = str.tostring(close, "#.##") + "\n(" + fibLabel + ": " + str.tostring(targetFib, "#.##") + ")"
    bool isBullishCandle = close >= open
    bool isBearishCandle = close < open
    color degreeColor = (wt.last_pivot == 1 and isBullishCandle) ? cUp : (wt.last_pivot == -1 and isBearishCandle) ? cDn : color.white

    if barstate.islast
        table.cell(waveTable, 0, rowIdx, " ⬤  " + degreeName, text_color=degreeColor, text_halign=text.align_left, text_size=g_textSize, text_font_family=g_fontFam)
        table.cell(waveTable, 1, rowIdx, waveStr, text_color=finalWaveColor, text_halign=text.align_center, text_size=g_textSize, text_font_family=g_fontFam)
        table.cell(waveTable, 2, rowIdx, statusStr, text_color=cellColor, text_halign=text.align_right, text_size=g_textSize, text_font_family=g_fontFam)

if barstate.islast
    table.set_bgcolor(waveTable, bgColor)
    if showMinor or showInter or showMajor
