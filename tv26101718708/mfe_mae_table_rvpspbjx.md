# MFE / MAE Table

## Description

The MFE/MAE Table is a drop-in Pine Script v5 indicator for TradingView that tracks and displays Maximum Favorable Excursion (MFE) and Maximum Adverse Excursion (MAE) statistics across all long and short trades generated by any entry signal you plug in. It calculates average MFE/MAE, single-trade bests and worsts, last-trade values, and a live updating row for your currently open position — all presented in a color-coded on-chart table with fully configurable position (six corners/centers), text size (six levels), and value display format (percentage of entry price, raw ticks, or both).

## Source Code

````pine
// This Pine Script® code is subject to the terms of the Mozilla Public License 2.0 at https://mozilla.org/MPL/2.0/
// © averagejoe_1919

// ============================================================
// MFE / MAE TABLE  —  Drop-in indicator for TradingView
// ============================================================
// HOW TO USE
// 1. Open TradingView → Pine Script Editor
// 2. Paste this entire script as a NEW indicator
//    (or append the strategy logic from Step 4 into your own)
// 3. Adjust the "ENTRY SIGNAL CONFIG" section to match your
//    indicator's long/short signals
// 4. Add to chart — the table appears in the selected corner
// ============================================================

//@version=5
indicator("MFE / MAE Table", overlay=true, max_bars_back=500)

// ─────────────────────────────────────────────
// 1. INPUTS
// ─────────────────────────────────────────────
grp_entry = "Entry Signal Config"
long_src   = input.source(close,  "Long  entry price source",  group=grp_entry)
short_src  = input.source(close,  "Short entry price source",  group=grp_entry)

// ► Replace these two lines with your actual long/short signals:
//   e.g.  long_signal  = ta.crossover(ta.ema(close,9), ta.ema(close,21))
//         short_signal = ta.crossunder(ta.ema(close,9), ta.ema(close,21))
long_signal  = ta.crossover (close, ta.sma(close, 20))   // ← swap with your signal
short_signal = ta.crossunder(close, ta.sma(close, 200))   // ← swap with your signal

grp_exit = "Exit Signal Config"
// ► Replace with your actual exit signal, or use one of the presets below
use_fixed_exit = input.bool(true,  "Use fixed bar exit (else: re-entry = exit)", group=grp_exit)
exit_bars      = input.int (20,    "Exit after N bars",  minval=1, group=grp_exit)

grp_display = "Display"
tbl_pos = input.string("Top Right", "Table position",
     options=["Top Right","Top Left","Top Center",
              "Bottom Right","Bottom Left","Bottom Center"],
     group=grp_display)
tbl_size = input.string("Small", "Table text size",
     options=["Auto","Tiny","Small","Normal","Large","Huge"],
     group=grp_display)
show_pct    = input.bool(true,  "Show values as % of entry price", group=grp_display)
show_ticks  = input.bool(false, "Also show raw tick values",       group=grp_display)

// ─────────────────────────────────────────────
// 2. STATE VARIABLES
// ─────────────────────────────────────────────
var float entry_price    = na
var bool  in_long        = false
var bool  in_short       = false
var int   bars_in_trade  = 0

var float mfe_long_sum   = 0.0
var float mae_long_sum   = 0.0
var int   long_count     = 0

var float mfe_short_sum  = 0.0
var float mae_short_sum  = 0.0
var int   short_count    = 0

var float mfe_long_last  = na
var float mae_long_last  = na
var float mfe_short_last = na
var float mae_short_last = na

var float mfe_long_max   = na    // best single trade MFE
var float mae_long_max   = na    // worst single trade MAE
var float mfe_short_max  = na
var float mae_short_max  = na

// Running extremes within the current open trade
var float run_mfe = na
var float run_mae = na

// ─────────────────────────────────────────────
// 3. TRADE LOGIC
// ─────────────────────────────────────────────
// Determine exit condition
exit_condition = use_fixed_exit ? (bars_in_trade >= exit_bars) :
                 (in_long ? short_signal : (in_short ? long_signal : false))

// Track bars in trade
if in_long or in_short
    bars_in_trade += 1

// ── Open Long ──
if long_signal and not in_long and not in_short
    in_long      := true
    entry_price  := long_src
    bars_in_trade := 0
    run_mfe      := 0.0
    run_mae      := 0.0

// ── Open Short ──
if short_signal and not in_short and not in_long
    in_short     := true
    entry_price  := short_src
    bars_in_trade := 0
    run_mfe      := 0.0
    run_mae      := 0.0

// ── Update running MFE / MAE while in trade ──
if in_long and not na(entry_price)
    current_excursion = high - entry_price     // favorable = higher
    adverse_excursion = entry_price - low      // adverse   = lower
    run_mfe := math.max(nz(run_mfe), current_excursion)
    run_mae := math.max(nz(run_mae), adverse_excursion)

if in_short and not na(entry_price)
    current_excursion = entry_price - low      // favorable = lower
    adverse_excursion = high - entry_price     // adverse   = higher
    run_mfe := math.max(nz(run_mfe), current_excursion)
    run_mae := math.max(nz(run_mae), adverse_excursion)

// ── Close trade & record stats ──
if exit_condition and in_long
    mfe_long_last  := run_mfe
    mae_long_last  := run_mae
    mfe_long_sum   += run_mfe
    mae_long_sum   += run_mae
    long_count     += 1
    mfe_long_max   := na(mfe_long_max) ? run_mfe : math.max(mfe_long_max, run_mfe)
    mae_long_max   := na(mae_long_max) ? run_mae : math.max(mae_long_max, run_mae)
    in_long        := false
    entry_price    := na

if exit_condition and in_short
    mfe_short_last  := run_mfe
    mae_short_last  := run_mae
    mfe_short_sum   += run_mfe
    mae_short_sum   += run_mae
    short_count     += 1
    mfe_short_max   := na(mfe_short_max) ? run_mfe : math.max(mfe_short_max, run_mfe)
    mae_short_max   := na(mae_short_max) ? run_mae : math.max(mae_short_max, run_mae)
    in_short        := false
    entry_price     := na

// ─────────────────────────────────────────────
// 4. HELPER: format value
// ─────────────────────────────────────────────
format_val(val, ep) =>
    if na(val) or na(ep)
        "—"
    else
        pct_str = show_pct  ? str.tostring(math.round(val / ep * 100, 2)) + "%" : ""
        tck_str = show_ticks? str.tostring(math.round(val / syminfo.mintick)) + "t"  : ""
        sep     = show_pct and show_ticks ? "  " : ""
        pct_str + sep + tck_str == "" ?
             str.tostring(math.round(val, 4)) :
             pct_str + sep + tck_str

// Compute averages
avg_mfe_long  = long_count  > 0 ? mfe_long_sum  / long_count  : na
avg_mae_long  = long_count  > 0 ? mae_long_sum  / long_count  : na
avg_mfe_short = short_count > 0 ? mfe_short_sum / short_count : na
avg_mae_short = short_count > 0 ? mae_short_sum / short_count : na

// Use last known entry price as reference for open trades, or
// use last closed entry approximation (stored avg) for display
ref_long  = in_long  ? entry_price : (avg_mfe_long  > 0 ? close : na)
ref_short = in_short ? entry_price : (avg_mfe_short > 0 ? close : na)

// ─────────────────────────────────────────────
// 5. TABLE
// ─────────────────────────────────────────────
pos_map(_s) =>
    switch _s
        "Top Right"      => position.top_right
        "Top Left"       => position.top_left
        "Top Center"     => position.top_center
        "Bottom Right"   => position.bottom_right
        "Bottom Left"    => position.bottom_left
        => position.bottom_center

size_map(_s) =>
    switch _s
        "Auto"   => size.auto
        "Tiny"   => size.tiny
        "Normal" => size.normal
        "Large"  => size.large
        "Huge"   => size.huge
        => size.small   // default: "Small"

// Recreate table whenever position or size changes so settings apply live
var string _last_pos  = na
var string _last_size = na
var table  t          = na

if tbl_pos != _last_pos or tbl_size != _last_size
    if not na(t)
        table.delete(t)
    t           := table.new(pos_map(tbl_pos), columns=5, rows=8,
         border_width=1,
         border_color=color.new(color.gray, 60),
         frame_color=color.new(color.gray, 40),
         frame_width=1)
    _last_pos   := tbl_pos
    _last_size  := tbl_size

// Resolved size tokens
sz      = size_map(tbl_size)
sz_foot = tbl_size == "Tiny" or tbl_size == "Auto" ? size.tiny : size.tiny   // footer always tiny

// ── Header row ──
table.cell(t, 0, 0, "MFE / MAE",  bgcolor=#1a1a2e, text_color=color.white,                    text_size=sz)
table.cell(t, 1, 0, "Avg MFE",    bgcolor=#1a1a2e, text_color=color.new(color.teal,  10),      text_size=sz)
table.cell(t, 2, 0, "Avg MAE",    bgcolor=#1a1a2e, text_color=color.new(color.red,   10),      text_size=sz)
table.cell(t, 3, 0, "Best MFE",   bgcolor=#1a1a2e, text_color=color.new(color.teal,  30),      text_size=sz)
table.cell(t, 4, 0, "Worst MAE",  bgcolor=#1a1a2e, text_color=color.new(color.red,   30),      text_size=sz)

// ── Sub-header: Longs ──
bg_long = color.new(color.blue, 80)
table.cell(t, 0, 1, "▲ Longs (" + str.tostring(long_count) + ")", bgcolor=bg_long, text_color=color.new(color.blue, 0), text_size=sz)
table.cell(t, 1, 1, format_val(avg_mfe_long,  close), bgcolor=bg_long, text_color=color.new(color.teal, 0), text_size=sz)
table.cell(t, 2, 1, format_val(avg_mae_long,  close), bgcolor=bg_long, text_color=color.new(color.red,  0), text_size=sz)
table.cell(t, 3, 1, format_val(mfe_long_max,  close), bgcolor=bg_long, text_color=color.new(color.teal, 0), text_size=sz)
table.cell(t, 4, 1, format_val(mae_long_max,  close), bgcolor=bg_long, text_color=color.new(color.red,  0), text_size=sz)

// ── Sub-header: Shorts ──
bg_short = color.new(color.orange, 80)
table.cell(t, 0, 2, "▼ Shorts (" + str.tostring(short_count) + ")", bgcolor=bg_short, text_color=color.new(color.orange, 0), text_size=sz)
table.cell(t, 1, 2, format_val(avg_mfe_short, close), bgcolor=bg_short, text_color=color.new(color.teal, 0), text_size=sz)
table.cell(t, 2, 2, format_val(avg_mae_short, close), bgcolor=bg_short, text_color=color.new(color.red,  0), text_size=sz)
table.cell(t, 3, 2, format_val(mfe_short_max, close), bgcolor=bg_short, text_color=color.new(color.teal, 0), text_size=sz)
table.cell(t, 4, 2, format_val(mae_short_max, close), bgcolor=bg_short, text_color=color.new(color.red,  0), text_size=sz)

// ── Divider ──
table.cell(t, 0, 3, "── Last Trade ──", bgcolor=color.new(color.gray,85), text_color=color.gray, text_size=size.tiny)
table.cell(t, 1, 3, "",                 bgcolor=color.new(color.gray,85), text_size=size.tiny)
table.cell(t, 2, 3, "",                 bgcolor=color.new(color.gray,85), text_size=size.tiny)
table.cell(t, 3, 3, "",                 bgcolor=color.new(color.gray,85), text_size=size.tiny)
table.cell(t, 4, 3, "",                 bgcolor=color.new(color.gray,85), text_size=size.tiny)

// ── Last long trade ──
table.cell(t, 0, 4, "Last Long",                       bgcolor=bg_long, text_color=color.new(color.blue,0), text_size=sz)
table.cell(t, 1, 4, format_val(mfe_long_last, close),  bgcolor=bg_long, text_color=color.new(color.teal,0), text_size=sz)
table.cell(t, 2, 4, format_val(mae_long_last, close),  bgcolor=bg_long, text_color=color.new(color.red,0),  text_size=sz)
table.cell(t, 3, 4, "",                                bgcolor=bg_long, text_size=sz)
table.cell(t, 4, 4, "",                                bgcolor=bg_long, text_size=sz)

// ── Last short trade ──
table.cell(t, 0, 5, "Last Short",                        bgcolor=bg_short, text_color=color.new(color.orange,0), text_size=sz)
table.cell(t, 1, 5, format_val(mfe_short_last, close),   bgcolor=bg_short, text_color=color.new(color.teal,0),   text_size=sz)
table.cell(t, 2, 5, format_val(mae_short_last, close),   bgcolor=bg_short, text_color=color.new(color.red,0),    text_size=sz)
table.cell(t, 3, 5, "",                                  bgcolor=bg_short, text_size=sz)
table.cell(t, 4, 5, "",                                  bgcolor=bg_short, text_size=sz)

// ── Live open trade row ──
live_label = in_long  ? "▲ LIVE LONG"  :
             in_short ? "▼ LIVE SHORT" : "No open trade"
live_bg    = in_long  ? color.new(color.blue,   65) :
             in_short ? color.new(color.orange,  65) : color.new(color.gray, 85)
live_col   = in_long  ? color.new(color.blue,    0) :
             in_short ? color.new(color.orange,   0) : color.gray

table.cell(t, 0, 6, live_label,                          bgcolor=live_bg, text_color=live_col,               text_size=sz)
table.cell(t, 1, 6, format_val(run_mfe, entry_price),    bgcolor=live_bg, text_color=color.new(color.teal,0), text_size=sz)
table.cell(t, 2, 6, format_val(run_mae, entry_price),    bgcolor=live_bg, text_color=color.new(color.red,0),  text_size=sz)
table.cell(t, 3, 6, "",                                  bgcolor=live_bg, text_size=sz)
table.cell(t, 4, 6, "",                                  bgcolor=live_bg, text_size=sz)

// ── Footer: units legend ──
units_str = show_pct and show_ticks ? "% price  |  t = ticks" :
         show_pct  ? "values = % of entry price" :
         show_ticks? "values = ticks" : "values = price"
table.cell(t, 0, 7, units_str, bgcolor=color.new(color.gray,90), text_color=color.gray, text_size=size.tiny)
table.cell(t, 1, 7, "",        bgcolor=color.new(color.gray,90), text_size=size.tiny)
table.cell(t, 2, 7, "",        bgcolor=color.new(color.gray,90), text_size=size.tiny)
table.cell(t, 3, 7, "",        bgcolor=color.new(color.gray,90), text_size=size.tiny)
table.cell(t, 4, 7, "",        bgcolor=color.new(color.gray,90), text_size=size.tiny)

// ─────────────────────────────────────────────
// 6. OPTIONAL: plot entry markers
// ─────────────────────────────────────────────
plotshape(long_signal,  style=shape.triangleup,   location=location.belowbar,
     color=color.new(color.blue,   20), size=size.tiny, title="Long Entry")
plotshape(short_signal, style=shape.triangledown, location=location.abovebar,
     color=color.new(color.orange, 20), size=size.tiny, title="Short Entry")
````
