## Description

Crypto Options Walls — Manual Levels

This indicator displays key options-market levels directly on the chart:
Put Walls 1–3 — potential support levels.
Call Walls 1–3 — potential resistance levels.
Max Pain — the price with the lowest estimated aggregate option payout.
Gamma Flip — an optional level indicating a potential change in the market’s gamma regime.

It supports manual entry of the asset, expiration date, and price levels. Lines extend automatically to the right and include clear labels.

Alerts are available when price approaches:
Put Wall 1
Call Wall 1
Max Pain
The alert proximity can be configured as a percentage.

This indicator is intended for analytical purposes only and does not constitute financial advice. Options levels should be used alongside price action, volume, volatility, and market-structure analysis.

## Source Code

````pine
//@version=6
indicator("BTC Options Walls — Manual Levels", overlay = true, max_lines_count = 10, max_labels_count = 10)

// Manual levels generated by the external Python script.
string expiryLabel = input.string("27JUN26", "Expiry label", group = "Levels")
float maxPain = input.float(110000.0, "Max Pain", minval = 0.0, step = 100.0, group = "Levels")
float putWall1 = input.float(105000.0, "Put Wall 1", minval = 0.0, step = 100.0, group = "Levels")
float putWall2 = input.float(100000.0, "Put Wall 2", minval = 0.0, step = 100.0, group = "Levels")
float putWall3 = input.float(95000.0, "Put Wall 3", minval = 0.0, step = 100.0, group = "Levels")
float callWall1 = input.float(115000.0, "Call Wall 1", minval = 0.0, step = 100.0, group = "Levels")
float callWall2 = input.float(120000.0, "Call Wall 2", minval = 0.0, step = 100.0, group = "Levels")
float callWall3 = input.float(125000.0, "Call Wall 3", minval = 0.0, step = 100.0, group = "Levels")
float gammaFlip = input.float(0.0, "Gamma Flip (optional)", minval = 0.0, step = 100.0, group = "Levels")

bool showPutWalls = input.bool(true, "Show Put Walls", group = "Visibility")
bool showCallWalls = input.bool(true, "Show Call Walls", group = "Visibility")
bool showMaxPain = input.bool(true, "Show Max Pain", group = "Visibility")
bool showGammaFlip = input.bool(false, "Show Gamma Flip", group = "Visibility")

float alertDistancePct = input.float(0.3, "Alert distance (%)", minval = 0.01, step = 0.05, group = "Alerts")
int labelOffset = input.int(3, "Label offset (bars)", minval = 1, maxval = 50, group = "Style")

color putColor1 = color.rgb(0, 200, 120)
color putColor2 = color.rgb(45, 170, 100)
color putColor3 = color.rgb(75, 135, 90)
color callColor1 = color.rgb(245, 70, 70)
color callColor2 = color.rgb(210, 80, 80)
color callColor3 = color.rgb(175, 85, 85)
color maxPainColor = color.rgb(255, 170, 0)
color gammaFlipColor = color.rgb(120, 90, 255)

// Create exactly one line and one label per level. The objects are updated,
// rather than recreated, so the script does not leak drawings on every bar.
var line putLine1 = line.new(bar_index, putWall1, bar_index + 1, putWall1, extend = extend.right, color = putColor1, width = 3)
var line putLine2 = line.new(bar_index, putWall2, bar_index + 1, putWall2, extend = extend.right, color = putColor2, width = 2)
var line putLine3 = line.new(bar_index, putWall3, bar_index + 1, putWall3, extend = extend.right, color = putColor3, width = 1)
var line callLine1 = line.new(bar_index, callWall1, bar_index + 1, callWall1, extend = extend.right, color = callColor1, width = 3)
var line callLine2 = line.new(bar_index, callWall2, bar_index + 1, callWall2, extend = extend.right, color = callColor2, width = 2)
var line callLine3 = line.new(bar_index, callWall3, bar_index + 1, callWall3, extend = extend.right, color = callColor3, width = 1)
var line maxPainLine = line.new(bar_index, maxPain, bar_index + 1, maxPain, extend = extend.right, color = maxPainColor, style = line.style_dashed, width = 3)
var line gammaFlipLine = line.new(bar_index, gammaFlip, bar_index + 1, gammaFlip, extend = extend.right, color = gammaFlipColor, style = line.style_dotted, width = 2)

var label putLabel1 = label.new(bar_index, putWall1, "", style = label.style_label_left, textcolor = color.white, color = putColor1)
var label putLabel2 = label.new(bar_index, putWall2, "", style = label.style_label_left, textcolor = color.white, color = putColor2)
var label putLabel3 = label.new(bar_index, putWall3, "", style = label.style_label_left, textcolor = color.white, color = putColor3)
var label callLabel1 = label.new(bar_index, callWall1, "", style = label.style_label_left, textcolor = color.white, color = callColor1)
var label callLabel2 = label.new(bar_index, callWall2, "", style = label.style_label_left, textcolor = color.white, color = callColor2)
var label callLabel3 = label.new(bar_index, callWall3, "", style = label.style_label_left, textcolor = color.white, color = callColor3)
var label maxPainLabel = label.new(bar_index, maxPain, "", style = label.style_label_left, textcolor = color.black, color = maxPainColor)
var label gammaFlipLabel = label.new(bar_index, gammaFlip, "", style = label.style_label_left, textcolor = color.white, color = gammaFlipColor)

f_update_level(line levelLine, label levelLabel, float level, bool visible, string levelName, color levelColor) =>
    bool active = visible and level > 0
    int lineStart = math.max(0, bar_index - 500)
    line.set_xy1(levelLine, lineStart, level)
    line.set_xy2(levelLine, bar_index + 1, level)
    line.set_color(levelLine, active ? levelColor : na)
    label.set_xy(levelLabel, bar_index + labelOffset, level)
    label.set_text(levelLabel, active ? levelName + " · " + expiryLabel : "")
    label.set_color(levelLabel, active ? levelColor : na)
    label.set_textcolor(levelLabel, active ? (levelName == "MAX PAIN" ? color.black : color.white) : na)

if barstate.islast
    f_update_level(putLine1, putLabel1, putWall1, showPutWalls, "PUT WALL 1", putColor1)
    f_update_level(putLine2, putLabel2, putWall2, showPutWalls, "PUT WALL 2", putColor2)
    f_update_level(putLine3, putLabel3, putWall3, showPutWalls, "PUT WALL 3", putColor3)
    f_update_level(callLine1, callLabel1, callWall1, showCallWalls, "CALL WALL 1", callColor1)
    f_update_level(callLine2, callLabel2, callWall2, showCallWalls, "CALL WALL 2", callColor2)
    f_update_level(callLine3, callLabel3, callWall3, showCallWalls, "CALL WALL 3", callColor3)
    f_update_level(maxPainLine, maxPainLabel, maxPain, showMaxPain, "MAX PAIN", maxPainColor)
    f_update_level(gammaFlipLine, gammaFlipLabel, gammaFlip, showGammaFlip, "GAMMA FLIP", gammaFlipColor)

f_is_near(float level, bool enabled) =>
    enabled and level > 0 and math.abs(close - level) / level * 100.0 <= alertDistancePct

bool nearPutWall1 = f_is_near(putWall1, showPutWalls)
bool nearCallWall1 = f_is_near(callWall1, showCallWalls)
bool nearMaxPain = f_is_near(maxPain, showMaxPain)

// Trigger when price enters the configured proximity zone.
bool enteredPutWall1Zone = nearPutWall1 and not nearPutWall1[1]
bool enteredCallWall1Zone = nearCallWall1 and not nearCallWall1[1]
bool enteredMaxPainZone = nearMaxPain and not nearMaxPain[1]

alertcondition(enteredPutWall1Zone, "Price near Put Wall 1", "BTC price entered the configured Put Wall 1 proximity zone.")
alertcondition(enteredCallWall1Zone, "Price near Call Wall 1", "BTC price entered the configured Call Wall 1 proximity zone.")
alertcondition(enteredMaxPainZone, "Price near Max Pain", "BTC price entered the configured Max Pain proximity zone.")
````
