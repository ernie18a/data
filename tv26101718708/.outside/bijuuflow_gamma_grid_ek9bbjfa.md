## Description

BijuuFlow Gamma Grid

Plots dealer options positioning — call walls, put walls, and the zero-gamma flip — directly on your ES/NQ chart.

Levels are generated by the BijuuFlow dashboard from live SPX/NDX options flow (GEX and open-interest), then pasted into the indicator's BijuuFlow Data field. Each wall is drawn as a strength-ranked level with an optional filled zone, sized to the real strike spacing and aligned to futures pricing.

Features

GEX or OI walls, 0DTE or full-expiry views
Strength scoring (1–100) drives line width — the heaviest walls stand out
Min Strength % and Max Walls filters to keep only high-quality levels
Filled zones, Key-Open–style segments, manual price offset, full color/size control
How to use: copy a ladder from the BijuuFlow Options page → paste into BijuuFlow Data → re-paste to refresh. Levels update on demand, not automatically.

## Source Code

````pine
//@version=6
// ─────────────────────────────────────────────────────────────────────────────
// BijuuFlow Gamma Grid
//
// Plots GEX/OI walls and the zero-gamma line exported from the BijuuFlow
// dashboard (Options page → "TradingView Wall Export"). Each wall shows the
// COMBINED call+put value at that strike (stable as price crosses it), coloured
// by strength (and line width scales with strength too).
//
// HOW TO USE
//   1. In BijuuFlow, open the Options page and pick GEX Walls or OI Walls.
//   2. Click "Copy NQ Ladder" (or ES).
//   3. Add this indicator to your ES/NQ chart, open its settings,
//      and paste into the "BijuuFlow Data" field.
//   4. Re-copy + re-paste whenever you want fresh levels.
//
// The exported levels are already futures-aligned (BijuuFlow computes GEX
// against the futures mid), so they plot at the right prices with no editing.
//
// Data format (newline-separated; first line is a header):
//   BJF1|NQ|GEX|UL=30456.62|ZG=29406.54|ZGSRC=FULL|MP=30000|TS=...
//   30570,C,100,337425648,12150000,12.5 ← price,side,strength(1-100),call$,put$,zone-half-width
//   30100,P,48,4200000,101754693,12.5
//   29406,Z,0,0,0,0,FULL                 ← zero gamma; field 6 = flip source 0DTE|FULL
//   30000,M,0,0,0,0                      ← max pain
//   (ZGSRC/field-6: 0DTE = intraday flip while options open; FULL = ≤30D overnight regime level)
// ─────────────────────────────────────────────────────────────────────────────
indicator("BijuuFlow Gamma Grid", overlay = true, max_lines_count = 500, max_labels_count = 500, max_boxes_count = 500)

// ── Inputs ───────────────────────────────────────────────────────────────────
dataStr      = input.text_area("", "BijuuFlow Data", group = "Data")

maxLevels    = input.int(30, "Max Walls", minval = 1, maxval = 100, tooltip = "Most walls to show, strongest first.", group = "Display")
minStrength  = input.int(0, "Min Strength %", minval = 0, maxval = 100, tooltip = "Hide walls weaker than this % of the strongest wall. Raise it to keep only the highest-quality levels.", group = "Display")
showZeroGamma= input.bool(true, "Show Zero Gamma", group = "Display")
showMaxPain  = input.bool(true, "Show Max Pain", group = "Display")
showLabels   = input.bool(true, "Show Labels", group = "Display")
lineLength   = input.int(50, "Level Extent (bars)", minval = 2, maxval = 500, tooltip = "How many bars each wall segment spans, anchored to the latest bar (Key-Open style). Fixed bar count — the same on every timeframe.", group = "Display")
rightExtend  = input.int(0, "Extend Right (bars)", minval = 0, maxval = 100, tooltip = "Push each wall (and its label) this many bars to the right of the latest bar for forward visibility.", group = "Display")
labelGap     = input.int(2, "Label Gap (bars)", minval = 0, maxval = 50, tooltip = "Bars of space between the line end and its label so the line never covers the text.", group = "Display")
labelSize    = input.string("small", "Label Size", options = ["tiny", "small", "normal", "large"], group = "Display")

showZones    = input.bool(true, "Filled Zones", group = "Zones")
zoneScale    = input.float(1.0, "Zone Width ×", minval = 0.1, maxval = 10.0, step = 0.1, tooltip = "Multiplier on the exported strike-spacing band.", group = "Zones")
zoneFallback = input.float(12.0, "Zone Width Fallback (pts)", minval = 0.0, tooltip = "Full zone height used when the pasted data has no band info.", group = "Zones")
zoneTransp   = input.int(85, "Zone Transparency", minval = 0, maxval = 100, group = "Zones")

// Walls are coloured by STRENGTH: weak walls toward 'Wall — weak', strong walls
// toward 'Wall — strong'. Line width also scales with strength (a second cue).
wallWeak     = input.color(color.new(#5a6478, 0), "Wall — weak", tooltip = "Colour for the weakest walls; strong walls fade toward 'Wall — strong'.", group = "Colors")
wallStrong   = input.color(color.new(#ff4d6d, 0), "Wall — strong", group = "Colors")
zgColor      = input.color(color.new(#ffa726, 0), "Zero Gamma", group = "Colors")
mpColor      = input.color(color.new(#42a5f5, 0), "Max Pain", group = "Colors")

// ── Helpers ──────────────────────────────────────────────────────────────────
f_size(s) =>
    s == "tiny" ? size.tiny : s == "small" ? size.small : s == "large" ? size.large : size.normal

f_fmt(v) =>
    na(v) or v == 0 ? "" :
      v >= 1e9 ? str.tostring(v / 1e9, "#.0") + "B" :
      v >= 1e6 ? str.tostring(v / 1e6, "#.0") + "M" :
      v >= 1e3 ? str.tostring(v / 1e3, "#.0") + "K" :
      str.tostring(v, "#")

// ── State ────────────────────────────────────────────────────────────────────
var array<line>  drawnLines  = array.new<line>()
var array<label> drawnLabels = array.new<label>()
var array<box>   drawnBoxes  = array.new<box>()

// ── Render (only on the last bar; redraw fully each time) ─────────────────────
if barstate.islast
    for ln in drawnLines
        line.delete(ln)
    array.clear(drawnLines)
    for lb in drawnLabels
        label.delete(lb)
    array.clear(drawnLabels)
    for bx in drawnBoxes
        box.delete(bx)
    array.clear(drawnBoxes)

    rows = str.split(dataStr, "\n")
    nWalls = 0
    if array.size(rows) > 0
        for i = 0 to array.size(rows) - 1
            row = array.get(rows, i)
            // Skip header and blank lines
            if str.startswith(row, "BJF1") or str.length(row) < 3
                continue

            fields = str.split(row, ",")
            if array.size(fields) < 3
                continue

            price    = str.tonumber(array.get(fields, 0))
            side     = array.get(fields, 1)
            strength = math.round(nz(str.tonumber(array.get(fields, 2)), 1.0))
            // Reference lines (zero gamma, max pain) are single pivots — exempt from
            // the wall strength filter, the wall count, and the filled zones.
            isRef = side == "Z" or side == "M"
            if na(price)
                continue
            if side == "Z" and not showZeroGamma
                continue
            if side == "M" and not showMaxPain
                continue
            if not isRef and strength < minStrength
                continue
            if not isRef and nWalls >= maxLevels
                continue

            lvl   = price
            col   = side == "Z" ? zgColor : side == "M" ? mpColor : color.from_gradient(strength, 1, 100, wallWeak, wallStrong)
            // Width scales with strength: 1..100 → 1..5 px.
            wdt   = isRef ? 2 : int(math.max(1, math.min(5, math.ceil(strength / 20.0))))
            sty   = side == "Z" ? line.style_dashed : side == "M" ? line.style_dotted : line.style_solid

            xRight = bar_index + rightExtend
            xLeft  = bar_index - lineLength

            // Filled zone (skip reference lines — single pivots).
            // Half-width comes from the export (field 6); fall back to the manual input.
            if showZones and not isRef
                hw = array.size(fields) > 5 ? str.tonumber(array.get(fields, 5)) : na
                halfW = (na(hw) or hw <= 0 ? zoneFallback / 2.0 : hw) * zoneScale
                if halfW > 0
                    bx = box.new(xLeft, lvl + halfW, xRight, lvl - halfW, xloc = xloc.bar_index, border_color = color.new(col, 100), bgcolor = color.new(col, zoneTransp))
                    array.push(drawnBoxes, bx)

            // Short segment anchored to the latest bar (Key-Open style) — no full-width line.
            ln = line.new(xLeft, lvl, xRight, lvl, xloc = xloc.bar_index, color = col, width = wdt, style = sty)
            array.push(drawnLines, ln)

            if showLabels
                callV = array.size(fields) > 3 ? str.tonumber(array.get(fields, 3)) : na
                putV  = array.size(fields) > 4 ? str.tonumber(array.get(fields, 4)) : na
                // Show the COMBINED call+put value, not one side — stable as price
                // crosses the strike (the side tag now only affects nothing visible).
                combinedV = nz(callV) + nz(putV)
                // Zero-gamma rows carry a 7th field (field 6): the flip source —
                // "0DTE" (intraday flip) or "FULL" (≤30D overnight regime level) —
                // so the label says which one this is and the two are never confused.
                zgSrc = side == "Z" and array.size(fields) > 6 ? array.get(fields, 6) : ""
                zgTxt = zgSrc == "0DTE" ? "Zero Gamma (0DTE)" : zgSrc == "FULL" ? "Zero Gamma (≤30D)" : "Zero Gamma"
                txt   = side == "Z" ? zgTxt : side == "M" ? "Max Pain" : "Wall " + f_fmt(combinedV)
                // Label placed past the line's right end so the line can never cover the text.
                // Labels always render above lines, and style_label_left puts the text to the right.
                lb = label.new(xRight + labelGap, lvl, txt, xloc = xloc.bar_index, style = label.style_label_left, color = color.new(col, 70), textcolor = color.white, size = f_size(labelSize))
                array.push(drawnLabels, lb)

            if not isRef
                nWalls += 1
````
