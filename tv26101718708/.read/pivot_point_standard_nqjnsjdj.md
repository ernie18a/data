# Pivot Point - Standard 

- url: https://www.tradingview.com/script/nqJNSJdJ-Pivot-Point-Standard/
- type: indicator

## Description

Overview
This indicator plots classic "Floor Trader" Standard Pivot Points (P, R1-R3, S1-S3) on your intraday charts. However, unlike traditional pivot indicators, it is specifically engineered to solve the "Settlement Close vs. Last Traded Price (LTP)" discrepancy that plagues traders in the Indian markets (NSE/BSE) or anyone trying to sync their TradingView charts with brokerages using the ChartIQ engine (like Zerodha, Upstox, etc.).

The Problem It Solves
Most standard TradingView pivot indicators use the daily Last Traded Price (LTP) as the daily Close. In markets like India, the official daily close is not the LTP, but a 30-minute Volume Weighted Average Price (VWAP) settlement. Because TradingView captures the final LTP spike at 15:30, traditional pivot math gets skewed, causing your TradingView support and resistance lines to completely mismatch the lines generated by your broker's platform.

How It Works
To bypass the LTP spike and replicate the official settlement price, this indicator features an Auto Time-Capture system.

Instead of looking at the daily timeframe for the close, it silently captures the closing price of a specific intraday candle right before the market closes (Default: 15:25 exchange time). This 15:25 close acts as a highly accurate proxy for the official VWAP settlement price. It pairs this captured close with the official daily High and Low to calculate mathematically perfect ChartIQ-style pivot levels for the next day.

Key Features:

Auto Time-Capture: Automatically grabs the closing price at a user-defined time (default 15:25) to avoid post-market LTP spikes.

True Pivot UI: Draws clean, horizontal lines for the current day only. It does not draw ugly diagonal lines connecting yesterday's levels to today's levels.

Full Floor Trader Math: Calculates the main Pivot (P), plus 3 levels of Support and Resistance (R1, R2, R3 / S1, S2, S3).

Fully Customizable: Users can adjust the specific hour and minute they want the script to capture the closing price via the settings menu.

How to Use:

Apply this to an intraday timeframe (e.g., 1-minute, 5-minute, or 15-minute). Note: The timeframe must be small enough that your chosen capture time (like 15:25) actually exists as a candle close.

The indicator will automatically capture the price at the designated time and use it to project the standard pivot levels when the next trading day begins.

## Source Code

````pine
// This Pine Script® code is subject to the terms of the Mozilla Public License 2.0 at https://mozilla.org/MPL/2.0/
// © Bhavik51

//@version=6
indicator("Pivots Points & Dual EMA", overlay = true, max_lines_count = 500, max_labels_count = 500)

// ============================================================================
// 1. PIVOT POINTS SETTINGS
// ============================================================================
GRP_PVT_MAIN = "Pivot Points: Main"
maxHistoricalPivotsInput = input.int(2, "Number of pivots back", minval=1, maxval=40, group=GRP_PVT_MAIN)

GRP_PVT_LBL = "Pivot Points: Labels"
showLabelsInput     = input.bool(true,     "Show labels",     group = GRP_PVT_LBL, inline="lbl1")
showPricesInput     = input.bool(true,     "Show prices",     group = GRP_PVT_LBL, inline="lbl1")
positionLabelsInput = input.string("Left", "Labels position", options = ["Left", "Right"], group = GRP_PVT_LBL)

GRP_PVT_LVL = "Pivot Points: Levels"
linewidthInput = input.int(1, "Line width", 1, 100, group = GRP_PVT_LVL)

colP   = input.color(#000000, "P ", inline = "p", group = GRP_PVT_LVL)
pShow  = input.bool(true, "",       inline = "p", group = GRP_PVT_LVL)
colS1  = input.color(#4caf50, "S1", inline = "L1", group = GRP_PVT_LVL)
s1Show = input.bool(true, "",       inline = "L1", group = GRP_PVT_LVL)
colR1  = input.color(#f23645, "   R1", inline = "L1", group = GRP_PVT_LVL)
r1Show = input.bool(true, "",       inline = "L1", group = GRP_PVT_LVL)
colS2  = input.color(#4caf50, "S2", inline = "L2", group = GRP_PVT_LVL)
s2Show = input.bool(true, "",       inline = "L2", group = GRP_PVT_LVL)
colR2  = input.color(#f23645, "   R2", inline = "L2", group = GRP_PVT_LVL)
r2Show = input.bool(true, "",       inline = "L2", group = GRP_PVT_LVL)
colS3  = input.color(#4caf50, "S3", inline = "L3", group = GRP_PVT_LVL)
s3Show = input.bool(true, "",       inline = "L3", group = GRP_PVT_LVL)
colR3  = input.color(#f23645, "   R3", inline = "L3", group = GRP_PVT_LVL)
r3Show = input.bool(true, "",       inline = "L3", group = GRP_PVT_LVL)
colS4  = input.color(#4caf50, "S4", inline = "L4", group = GRP_PVT_LVL)
s4Show = input.bool(true, "",       inline = "L4", group = GRP_PVT_LVL)
colR4  = input.color(#f23645, "   R4", inline = "L4", group = GRP_PVT_LVL)
r4Show = input.bool(true, "",       inline = "L4", group = GRP_PVT_LVL)
colS5  = input.color(#4caf50, "S5", inline = "L5", group = GRP_PVT_LVL)
s5Show = input.bool(true, "",       inline = "L5", group = GRP_PVT_LVL)
colR5  = input.color(#f23645, "   R5", inline = "L5", group = GRP_PVT_LVL)
r5Show = input.bool(true, "",       inline = "L5", group = GRP_PVT_LVL)

// ============================================================================
// 2. EMA 1 SETTINGS
// ============================================================================
GRP_EMA1   = "EMA 1"
ema1Show   = input.bool(true, title="Show EMA 1", group=GRP_EMA1)
ema1Tf     = input.timeframe("", title="Timeframe", group=GRP_EMA1)
ema1Len    = input.int(60, minval=1, title="Length", group=GRP_EMA1)
ema1Src    = input.source(close, title="Source", group=GRP_EMA1)
ema1Offset = input.int(0, title="Offset", minval=-500, maxval=500, group=GRP_EMA1)
ema1Color  = input.color(color.red, title="Color", group=GRP_EMA1)

// ============================================================================
// 3. EMA 2 SETTINGS
// ============================================================================
GRP_EMA2   = "EMA 2"
ema2Show   = input.bool(true, title="Show EMA 2", group=GRP_EMA2)
ema2Tf     = input.timeframe("", title="Timeframe", group=GRP_EMA2)
ema2Len    = input.int(13, minval=1, title="Length", group=GRP_EMA2)
ema2Src    = input.source(close, title="Source", group=GRP_EMA2)
ema2Offset = input.int(0, title="Offset", minval=-500, maxval=500, group=GRP_EMA2)
ema2Color  = input.color(color.blue, title="Color", group=GRP_EMA2)

// ============================================================================
// 4. HELPER FUNCTIONS & VARIABLES (PIVOTS)
// ============================================================================
getLabelText(string lvlName, float price) =>
    txt = ""
    if showLabelsInput
        txt += lvlName + " "
    if showPricesInput
        txt += "(" + str.tostring(price, format.mintick) + ")"
    txt

lblStyle = positionLabelsInput == "Left" ? label.style_label_right : label.style_label_left
var line[] allLines = array.new<line>()
var label[] allLabels = array.new<label>()
isIndianMarket = (syminfo.prefix == "NSE" or syminfo.prefix == "BSE")
isNewDay = timeframe.change("D")
prevHigh  = request.security(syminfo.tickerid, "D", high[1], lookahead=barmerge.lookahead_on)
prevLow   = request.security(syminfo.tickerid, "D", low[1], lookahead=barmerge.lookahead_on)

var float capturedClose = na
if isIndianMarket and hour(time, syminfo.timezone) == 15 and minute(time, syminfo.timezone) == 25
    capturedClose := close

var float p = na
var float r1 = na, var float s1 = na
var float r2 = na, var float s2 = na
var float r3 = na, var float s3 = na
var float r4 = na, var float s4 = na
var float r5 = na, var float s5 = na

var line lineP = na
var line lineR1 = na, var line lineS1 = na
var line lineR2 = na, var line lineS2 = na
var line lineR3 = na, var line lineS3 = na
var line lineR4 = na, var line lineS4 = na
var line lineR5 = na, var line lineS5 = na

var label lblP = na
var label lblR1 = na, var label lblS1 = na
var label lblR2 = na, var label lblS2 = na
var label lblR3 = na, var label lblS3 = na
var label lblR4 = na, var label lblS4 = na
var label lblR5 = na, var label lblS5 = na

// ============================================================================
// 5. PIVOT EXECUTION
// ============================================================================
if isIndianMarket
    if isNewDay
        p := (prevHigh + prevLow + capturedClose) / 3
        pivotRange = prevHigh - prevLow
        r1 := (p * 2) - prevLow, s1 := (p * 2) - prevHigh
        r2 := p + pivotRange,    s2 := p - pivotRange
        r3 := prevHigh + 2 * (p - prevLow), s3 := prevLow - 2 * (prevHigh - p)
        r4 := p + (pivotRange * 3), s4 := p - (pivotRange * 3)
        r5 := p + (pivotRange * 4), s5 := p - (pivotRange * 4)

        lineP  := pShow  ? line.new(bar_index, p, bar_index, p, color=colP, width=linewidthInput) : na
        lineR1 := r1Show ? line.new(bar_index, r1, bar_index, r1, color=colR1, width=linewidthInput) : na
        lineS1 := s1Show ? line.new(bar_index, s1, bar_index, s1, color=colS1, width=linewidthInput) : na
        lineR2 := r2Show ? line.new(bar_index, r2, bar_index, r2, color=colR2, width=linewidthInput) : na
        lineS2 := s2Show ? line.new(bar_index, s2, bar_index, s2, color=colS2, width=linewidthInput) : na
        lineR3 := r3Show ? line.new(bar_index, r3, bar_index, r3, color=colR3, width=linewidthInput) : na
        lineS3 := s3Show ? line.new(bar_index, s3, bar_index, s3, color=colS3, width=linewidthInput) : na
        lineR4 := r4Show ? line.new(bar_index, r4, bar_index, r4, color=colR4, width=linewidthInput) : na
        lineS4 := s4Show ? line.new(bar_index, s4, bar_index, s4, color=colS4, width=linewidthInput) : na
        lineR5 := r5Show ? line.new(bar_index, r5, bar_index, r5, color=colR5, width=linewidthInput) : na
        lineS5 := s5Show ? line.new(bar_index, s5, bar_index, s5, color=colS5, width=linewidthInput) : na

        lblP  := pShow  ? label.new(bar_index, p, getLabelText("P", p), style=lblStyle, textcolor=colP, color=color.new(color.white, 100), size=size.normal) : na
        lblR1 := r1Show ? label.new(bar_index, r1, getLabelText("R1", r1), style=lblStyle, textcolor=colR1, color=color.new(color.white, 100), size=size.normal) : na
        lblS1 := s1Show ? label.new(bar_index, s1, getLabelText("S1", s1), style=lblStyle, textcolor=colS1, color=color.new(color.white, 100), size=size.normal) : na
        lblR2 := r2Show ? label.new(bar_index, r2, getLabelText("R2", r2), style=lblStyle, textcolor=colR2, color=color.new(color.white, 100), size=size.normal) : na
        lblS2 := s2Show ? label.new(bar_index, s2, getLabelText("S2", s2), style=lblStyle, textcolor=colS2, color=color.new(color.white, 100), size=size.normal) : na
        lblR3 := r3Show ? label.new(bar_index, r3, getLabelText("R3", r3), style=lblStyle, textcolor=colR3, color=color.new(color.white, 100), size=size.normal) : na
        lblS3 := s3Show ? label.new(bar_index, s3, getLabelText("S3", s3), style=lblStyle, textcolor=colS3, color=color.new(color.white, 100), size=size.normal) : na
        lblR4 := r4Show ? label.new(bar_index, r4, getLabelText("R4", r4), style=lblStyle, textcolor=colR4, color=color.new(color.white, 100), size=size.normal) : na
        lblS4 := s4Show ? label.new(bar_index, s4, getLabelText("S4", s4), style=lblStyle, textcolor=colS4, color=color.new(color.white, 100), size=size.normal) : na
        lblR5 := r5Show ? label.new(bar_index, r5, getLabelText("R5", r5), style=lblStyle, textcolor=colR5, color=color.new(color.white, 100), size=size.normal) : na
        lblS5 := s5Show ? label.new(bar_index, s5, getLabelText("S5", s5), style=lblStyle, textcolor=colS5, color=color.new(color.white, 100), size=size.normal) : na

        allLines.push(lineP), allLines.push(lineR1), allLines.push(lineS1), allLines.push(lineR2), allLines.push(lineS2)
        allLines.push(lineR3), allLines.push(lineS3), allLines.push(lineR4), allLines.push(lineS4), allLines.push(lineR5), allLines.push(lineS5)
        allLabels.push(lblP), allLabels.push(lblR1), allLabels.push(lblS1), allLabels.push(lblR2), allLabels.push(lblS2)
        allLabels.push(lblR3), allLabels.push(lblS3), allLabels.push(lblR4), allLabels.push(lblS4), allLabels.push(lblR5), allLabels.push(lblS5)

        if allLines.size() > maxHistoricalPivotsInput * 11
            for i = 1 to 11
                oldLine = allLines.shift()
                if not na(oldLine)
                    line.delete(oldLine)
                oldLabel = allLabels.shift()
                if not na(oldLabel)
                    label.delete(oldLabel)
    else
        if not na(lineP)
            line.set_x2(lineP, bar_index)
            if not na(lineR1)
                line.set_x2(lineR1, bar_index)
            if not na(lineS1)
                line.set_x2(lineS1, bar_index)
            if not na(lineR2)
                line.set_x2(lineR2, bar_index)
            if not na(lineS2)
                line.set_x2(lineS2, bar_index)
            if not na(lineR3)
                line.set_x2(lineR3, bar_index)
            if not na(lineS3)
                line.set_x2(lineS3, bar_index)
            if not na(lineR4)
                line.set_x2(lineR4, bar_index)
            if not na(lineS4)
                line.set_x2(lineS4, bar_index)
            if not na(lineR5)
                line.set_x2(lineR5, bar_index)
            if not na(lineS5)
                line.set_x2(lineS5, bar_index)

        if positionLabelsInput == "Right"
            if not na(lblP)
                label.set_x(lblP, bar_index)
            if not na(lblR1)
                label.set_x(lblR1, bar_index)
            if not na(lblS1)
                label.set_x(lblS1, bar_index)
            if not na(lblR2)
                label.set_x(lblR2, bar_index)
            if not na(lblS2)
                label.set_x(lblS2, bar_index)
            if not na(lblR3)
                label.set_x(lblR3, bar_index)
            if not na(lblS3)
                label.set_x(lblS3, bar_index)
            if not na(lblR4)
                label.set_x(lblR4, bar_index)
            if not na(lblS4)
                label.set_x(lblS4, bar_index)
            if not na(lblR5)
                label.set_x(lblR5, bar_index)
            if not na(lblS5)
                label.set_x(lblS5, bar_index)

// ============================================================================
// 6. DUAL EMA EXECUTION
// ============================================================================
// EMA 1
baseEma1 = ta.ema(ema1Src, ema1Len)
mtfEma1  = request.security(syminfo.tickerid, ema1Tf, baseEma1)
finalEma1 = ema1Tf == "" ? baseEma1 : mtfEma1
plot(ema1Show ? finalEma1 : na, title="EMA 1", color=ema1Color, offset=ema1Offset)

// EMA 2
baseEma2 = ta.ema(ema2Src, ema2Len)
mtfEma2  = request.security(syminfo.tickerid, ema2Tf, baseEma2)
finalEma2 = ema2Tf == "" ? baseEma2 : mtfEma2
plot(ema2Show ? finalEma2 : na, title="EMA 2", color=ema2Color, offset=ema2Offset)
````
