[quote][b]Inside this defined box, what is the realistic rebound potential of this ticker based on its historical behavior?[/b][/quote]

ID 0009 is for traders who want [b]high-quality, controlled rebound trades[/b]. The main idea is not to predict everything that happens after the move. The main idea is:

rowDip(i) =>
    gridDipStart + clampInt(i, 0, DIP_ROWS - 1) * gridDipStep

colTgt(j) =>
    gridTgtStart + clampInt(j, 0, TGT_COLS - 1) * gridTgtStep

gridMaxDip() =>
    rowDip(DIP_ROWS - 1)

rowCntRaw(i) =>
    array.get(dCnt, clampInt(i, 0, DIP_ROWS - 1))

rawWinsAt(dipIdx, tgtIdx) =>
    array.get(wMat, matIdx(dipIdx, tgtIdx))

idxFromDip(dipPc) =>
    float raw = (dipPc - gridDipStart) / math.max(0.0001, gridDipStep)
    clampInt(int(math.round(raw)), 0, DIP_ROWS - 1)

idxFromTgt(tgtPc) =>
    float raw = (tgtPc - gridTgtStart) / math.max(0.0001, gridTgtStep)
    clampInt(int(math.round(raw)), 0, TGT_COLS - 1)

idxFloorFromTgt(tgtPc) =>
    float raw = (tgtPc - gridTgtStart) / math.max(0.0001, gridTgtStep)
    clampInt(int(math.floor(raw)), 0, TGT_COLS - 1)

aggCnt(centerIdx, radius) =>
    int lo = clampInt(centerIdx - radius, 0, DIP_ROWS - 1)
    int hi = clampInt(centerIdx + radius, 0, DIP_ROWS - 1)
    float c = 0.0
    for i = lo to hi
        c += rowCntRaw(i)
    c

aggWins(centerIdx, tgtIdx, radius) =>
    int lo = clampInt(centerIdx - radius, 0, DIP_ROWS - 1)
    int hi = clampInt(centerIdx + radius, 0, DIP_ROWS - 1)
    float w = 0.0
    for i = lo to hi
        w += rawWinsAt(i, tgtIdx)
    w

aggSumMae(centerIdx, radius) =>
    int lo = clampInt(centerIdx - radius, 0, DIP_ROWS - 1)
    int hi = clampInt(centerIdx + radius, 0, DIP_ROWS - 1)
    float s = 0.0
    for i = lo to hi
        s += array.get(dSumMae, i)
    s

aggSumMaxR(centerIdx, radius) =>
    int lo = clampInt(centerIdx - radius, 0, DIP_ROWS - 1)
    int hi = clampInt(centerIdx + radius, 0, DIP_ROWS - 1)
    float s = 0.0
    for i = lo to hi
        s += array.get(dSumMaxR, i)
    s

avgMaeAtAgg(centerIdx, radius) =>
    float c = aggCnt(centerIdx, radius)
    c > 0 ? aggSumMae(centerIdx, radius) / c : na

avgReboundAtAgg(centerIdx, radius) =>
    float c = aggCnt(centerIdx, radius)
    c > 0 ? aggSumMaxR(centerIdx, radius) / c : na

selectRadius(centerIdx) =>
    int hiR = int(math.max(baseBucketRadius, maxBucketRadius))
    int out = hiR
    bool found = false
    for r = baseBucketRadius to hiR
        if not found and aggCnt(centerIdx, r) >= minEventsPerRow
            out := r
            found := true
    out

wrAtAgg(centerIdx, tgtIdx, radius) =>
    float c = aggCnt(centerIdx, radius)
    c > 0 ? aggWins(centerIdx, tgtIdx, radius) * 100.0 / c : na

adjWrAtAgg(centerIdx, tgtIdx, radius) =>
    float c = aggCnt(centerIdx, radius)
    float out = na
    if c > 0
        float wins = aggWins(centerIdx, tgtIdx, radius)
        float p = wins / c
        float z = 1.96
        float denom = 1.0 + z * z / c
        float center = p + z * z / (2.0 * c)
        float margin = z * math.sqrt((p * (1.0 - p) + z * z / (4.0 * c)) / c)
        out := 100.0 * (center - margin) / denom
    out

allCnt() =>
    float c = 0.0
    for i = 0 to DIP_ROWS - 1
        c += rowCntRaw(i)
    c

allWins(tgtIdx) =>
    float w = 0.0
    for i = 0 to DIP_ROWS - 1
        w += rawWinsAt(i, tgtIdx)
    w

allSumMaxR() =>
    float s = 0.0
    for i = 0 to DIP_ROWS - 1
        s += array.get(dSumMaxR, i)
    s

avgReboundAll() =>
    float c = allCnt()
    c > 0 ? allSumMaxR() / c : na

wrAtAll(tgtIdx) =>
    float c = allCnt()
    c > 0 ? allWins(tgtIdx) * 100.0 / c : na

adjWrAtAll(tgtIdx) =>
    float c = allCnt()
    float out = na
    if c > 0
        float wins = allWins(tgtIdx)
        float p = wins / c
        float z = 1.96
        float denom = 1.0 + z * z / c
        float center = p + z * z / (2.0 * c)
        float margin = z * math.sqrt((p * (1.0 - p) + z * z / (4.0 * c)) / c)
        out := 100.0 * (center - margin) / denom
    out

selectWrSimilar(tgtIdx, minWR) =>

selectWrAll(tgtIdx, minWR) =>
    projectionMethod == "Wilson adjusted" and minWR < 99.999 ? adjWrAtAll(tgtIdx) : wrAtAll(tgtIdx)

bestTargetSimilar(minWR) =>

    if selectedCnt >= minEventsPerRow

            float t = colTgt(tgtIdx)
            float w = selectWrSimilar(tgtIdx, minWR)
            if not na(w) and w >= minWR
                bestTgt := t
                bestWR  := w
    [bestTgt, bestWR]

bestTargetAll(minWR) =>

    float bestWR  = na
    float c = allCnt()
    if c >= minEventsPerRow

            float w = selectWrAll(tgtIdx, minWR)
            if not na(w) and w >= minWR
                bestTgt := t
                bestWR  := w
    [bestTgt, bestWR]

    out
