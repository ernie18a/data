    array.from(c, c + r * 1.1 / 12.0, c - r * 1.1 / 12.0, c + r * 1.1 / 6.0, c - r * 1.1 / 6.0, c + r * 1.1 / 4.0, c - r * 1.1 / 4.0, c + r * 1.1 / 2.0, c - r * 1.1 / 2.0)

calcFib(float h, float l, float c) =>
    float p  = (h + l + c) / 3.0
    float r  = h - l
    array.from(p, p + 0.382 * r, p - 0.382 * r, p + 0.618 * r, p - 0.618 * r, p + 1.000 * r, p - 1.000 * r)

drawLevel(int startTime, int endTime, float val, string txt, color col, int thick, string stl, bool isRightSide, array<pivotGraphic> arr) =>
    line l = na
    if not na(val)
        color finalCol = color.new(col, pivotTransp)
        l := line.new(startTime, val, endTime, val, xloc = xloc.bar_time, extend = extend.none, color = finalCol, width = thick, style = getLineStyle(stl))
        label lb = label.new(isRightSide ? endTime : startTime, val, txt + " (" + str.tostring(val, format.mintick) + ")", 
           style = isRightSide ? label.style_label_left : label.style_label_right, textcolor = finalCol, color = color.new(color.black, 100), xloc = xloc.bar_time, size = size.small)
        // Store original un-opacified color for the radar glow
        arr.push(pivotGraphic.new(l, lb, col, thick, val, txt))
    l

drawFullMatrix(int startTime, int endTime, float h, float l, float c, float p2H, float p2L, float p2C, float checkH, float checkL, bool checkVirgin, array<pivotGraphic> targetArr) =>
    float p = (h + l + c) / 3.0
    float bc = (h + l) / 2.0
    float tc = (p - bc) + p
    float actualTC = math.max(bc, tc)
    float actualBC = math.min(bc, tc)
    
    line pLineObj = drawLevel(startTime, endTime, p, "P", pColIn, pThickIn, pStyleIn, false, targetArr)
    
    if showCPR and not na(pLineObj)
        color tcCol = #f23645
        color bcCol = #388e3c
        line tcL = line.new(startTime, actualTC, endTime, actualTC, xloc = xloc.bar_time, extend = extend.none, color = color.new(tcCol, pivotTransp), width = 1, style = line.style_solid)
        label tcLb = label.new(startTime, actualTC, "TC (" + str.tostring(actualTC, format.mintick) + ")", style = label.style_label_right, textcolor = color.new(tcCol, pivotTransp), color = color.new(color.black, 100), xloc = xloc.bar_time, size = size.small)
        targetArr.push(pivotGraphic.new(tcL, tcLb, tcCol, 1, actualTC, "TC"))
        
        line bcL = line.new(startTime, actualBC, endTime, actualBC, xloc = xloc.bar_time, extend = extend.none, color = color.new(bcCol, pivotTransp), width = 1, style = line.style_solid)
        label bcLb = label.new(startTime, actualBC, "BC (" + str.tostring(actualBC, format.mintick) + ")", style = label.style_label_right, textcolor = color.new(bcCol, pivotTransp), color = color.new(color.black, 100), xloc = xloc.bar_time, size = size.small)
        targetArr.push(pivotGraphic.new(bcL, bcLb, bcCol, 1, actualBC, "BC"))
        
        linefill.new(tcL, pLineObj, color.new(#f23645, 80))
        linefill.new(bcL, pLineObj, color.new(#388e3c, 80))

    if showVCPR and checkVirgin and not na(p2H) and not na(checkH)
        float vP  = (p2H + p2L + p2C) / 3.0
        float vBC = (p2H + p2L) / 2.0
        float vTC = (vP - vBC) + vP
        float vActualTC = math.max(vBC, vTC)
        float vActualBC = math.min(vBC, vTC)
        
        bool isVirgin = (checkL > vActualTC) or (checkH < vActualBC)
        if isVirgin
            color vpCol = pColIn
            color vtCol = #f23645
            color vbCol = #388e3c
            line vPLine  = line.new(startTime, vP, endTime, vP, xloc = xloc.bar_time, extend = extend.none, color = color.new(vpCol, pivotTransp), width = 1, style = line.style_dotted)
            line vTCL    = line.new(startTime, vActualTC, endTime, vActualTC, xloc = xloc.bar_time, extend = extend.none, color = color.new(vtCol, pivotTransp), width = 1, style = line.style_dotted)
            line vBCL    = line.new(startTime, vActualBC, endTime, vActualBC, xloc = xloc.bar_time, extend = extend.none, color = color.new(vbCol, pivotTransp), width = 1, style = line.style_dotted)
            
            label vLbl = label.new(startTime, vP, "VCPR (" + str.tostring(vP, format.mintick) + ")", style = label.style_label_right, textcolor = color.new(#ffb74d, pivotTransp), color = color.new(color.black, 100), xloc = xloc.bar_time, size = size.small)
            
            linefill.new(vTCL, vPLine, color.new(#f23645, 92))
            linefill.new(vBCL, vPLine, color.new(#388e3c, 92))
            
            targetArr.push(pivotGraphic.new(vPLine, vLbl, vpCol, 1, vP, "VCPR"))
            targetArr.push(pivotGraphic.new(vTCL, label.new(na, na), vtCol, 1, vActualTC, "V-TC"))
            targetArr.push(pivotGraphic.new(vBCL, label.new(na, na), vbCol, 1, vActualBC, "V-BC"))

    if showTrad
        array<float> tVals = calcTrad(h, l, c)
        array<bool>  tVis  = array.from(false, t_showR1, t_showS1, t_showR2, t_showS2, t_showR3, t_showS3)
        for i = 1 to tVals.size() - 1
            if tVis.get(i)
                drawLevel(startTime, endTime, tVals.get(i), "T-" + tradNames.get(i), i % 2 != 0 ? tColRIn : tColSIn, i % 2 != 0 ? tThickRIn : tThickSIn, tStyleIn, false, targetArr)

    if showCam
        array<float> cVals = calcCam(h, l, c)
        array<bool>  cVis  = array.from(c_showCCP, c_showR1, c_showS1, c_showR2, c_showS2, c_showR3, c_showS3, c_showR4, c_showS4)
        for i = 0 to cVals.size() - 1
            if cVis.get(i)
                color levelCol = i == 0 ? pColIn : (i % 2 != 0 ? cColRIn : cColSIn)
                int levelThick = i == 0 ? cThickSIn : (i % 2 != 0 ? cThickRIn : cThickSIn)
                string levelStl = i == 0 ? "Dotted" : cStyleIn
                
                drawLevel(startTime, endTime, cVals.get(i), "C-" + camNames.get(i), levelCol, levelThick, levelStl, false, targetArr)

    if showFib
        array<float> fVals = calcFib(h, l, c)
        array<bool>  fVis  = array.from(false, f_showR1, f_showS1, f_showR2, f_showS2, f_showR3, f_showS3)
        for i = 1 to fVals.size() - 1
            if fVis.get(i)
                drawLevel(startTime, endTime, fVals.get(i), "F-" + fibNames.get(i), i % 2 != 0 ? fColRIn : fColSIn, i % 2 != 0 ? fThickRIn : fThickSIn, fStyleIn, true, targetArr)

// SMART RADAR UPDATE ENGINE (TOUCH-ONLY GLOW & DYNAMIC LABELS)

updateSmartRadar(array<pivotGraphic> arr) =>
    bool isHit = false
    if enableRadar and arr.size() > 0
        if radarGlowLines.size() > 0
            for l in radarGlowLines
                l.delete()
            radarGlowLines.clear()
        if radarDynLabels.size() > 0
            for lb in radarDynLabels
                lb.delete()
            radarDynLabels.clear()
            
        int bDurMs = timeframe.in_seconds(timeframe.period) * 1000
        color darkBg = color.new(#1e222d, 0) 
        
        for i = 0 to arr.size() - 1
            pivotGraphic g = arr.get(i)
            if not na(g.pivotLine)
                g.pivotLine.set_color(color.new(g.origColor, pivotTransp))
                g.pivotLine.set_width(g.origWidth)
                
                float lvl = g.levelVal
                if not na(lvl)
                    float buffer = lvl * (bufferPct / 100.0)
                    bool isTouch = (high >= lvl - buffer) and (low <= lvl + buffer)
                    
                    if isTouch
                        isHit := true
                        color activeCol = g.origColor
                        color glowCol = color.new(activeCol, 80) 
                        color coreCol = color.new(activeCol, 0)  
                        
                        g.pivotLine.set_color(coreCol)
                        g.pivotLine.set_width(1)
                        
                        line glLine = line.new(g.pivotLine.get_x1(), lvl, g.pivotLine.get_x2(), lvl, xloc=xloc.bar_time, extend=extend.none, color=glowCol, width=12, style=line.style_solid)
                        radarGlowLines.push(glLine)
                        
                        label dynLb = label.new(time + bDurMs, lvl, g.levelName + " | " + str.tostring(lvl, format.mintick), xloc=xloc.bar_time, color=darkBg, textcolor=coreCol, style=label.style_label_left, size=size.normal)
                        radarDynLabels.push(dynLb)
    isHit
