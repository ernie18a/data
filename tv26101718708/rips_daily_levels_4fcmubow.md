## Description

A clean, no-frills indicator built around the daily key levels generated by the Rips algorithm — a proprietary system designed to identify high-probability price reaction zones on the ES and NQ futures contracts. The algorithm processes market structure, volatility, and order flow context to produce a refined set of levels each session, giving traders a clear roadmap of where price is most likely to react, reverse, or break with conviction.
The indicator itself is intentionally simple. There's no clutter, no signals, no noise — just precise horizontal lines drawn exactly where they need to be. Each trading day, paste the new ES and NQ levels directly into the indicator's settings panel (commas, spaces, or new lines all work), and the chart instantly updates with clean horizontal lines and price labels at every level.
Features:

Paste-in level inputs for ES and NQ — update your levels in seconds, no code editing required
Auto-detects whether you're on an ES or NQ chart and draws the correct set of levels
Manual override to show ES, NQ, or both sets on any chart
Fully configurable line color, style (solid/dashed/dotted), and width
Adjustable price labels with size, color, background, and offset controls
Lines extend across the full chart for clean visual reference at any timeframe

Whether you're scalping the open, holding swing positions, or just looking for context around price, RIPS Daily Levels keeps the algorithm's work right where you need it — directly on your chart.

//@version=6
indicator("RIPS Daily Levels", "RIPS Levels", overlay = true, max_lines_count = 500, max_labels_count = 500)

// ─── Inputs ──────────────────────────────────────────────────────────────────
gLvl   = "Levels (paste here)"
gSym   = "Symbol"
gStyle = "Line Style"
gLabel = "Labels"
gTbl   = "Info Table"

esInput = input.text_area(
     "7684.00, 7656.25, 7638.75, 7624.75, 7605.25, 7593.00, 7583.00, 7565.75, 7550.25, 7541.50, 7529.00, 7516.50, 7509.00, 7502.75, 7496.50, 7492.25, 7483.00, 7477.00, 7467.75, 7461.00, 7449.75, 7438.50, 7429.50, 7420.50, 7413.50, 7407.75, 7399.00, 7391.25, 7377.50, 7366.50",
     "ES Levels", group = gLvl,
     tooltip = "Paste the ES levels. Separate by commas, spaces, or new lines.")

nqInput = input.text_area(
     "30830.00, 30663.25, 30560.00, 30475.50, 30376.25, 30314.75, 30265.25, 30215.50, 30178.00, 30122.00, 30030.25, 29973.25, 29925.50, 29881.25, 29826.50, 29789.25, 29732.25, 29685.50, 29640.50, 29586.25, 29525.25, 29473.25, 29426.00, 29390.00, 29354.25, 29301.75, 29262.75, 29210.75, 29116.75, 29059.75, 29011.50, 28964.75",
     "NQ Levels", group = gLvl,
     tooltip = "Paste the NQ levels. Separate by commas, spaces, or new lines.")

mode = input.string("Auto", "Show Levels For",
     options = ["Auto", "ES", "NQ", "Both"], group = gSym,
     tooltip = "Auto detects ES or NQ from the chart's ticker. Use 'Both' to plot every level.")

lineColor = input.color(color.yellow, "Color",  group = gStyle)
styleStr  = input.string("Solid", "Style",      options = ["Solid", "Dashed", "Dotted"], group = gStyle)
lineWidth = input.int(1, "Width",               minval = 1, maxval = 5, group = gStyle)

showLabels  = input.bool(true, "Show Price Labels",   group = gLabel)
sizeStr     = input.string("Normal", "Label Size",    options = ["Tiny", "Small", "Normal", "Large", "Huge"], group = gLabel)
labelText   = input.color(color.black, "Text Color",  group = gLabel)
labelBg     = input.color(color.new(color.yellow, 0), "Background", group = gLabel)
labelOffset = input.int(5, "Label Offset (bars right)", minval = 0, maxval = 200, group = gLabel)

showTable    = input.bool(true, "Show Table", group = gTbl)
tblPosStr    = input.string("Top Right", "Position",
     options = ["Top Left", "Top Center", "Top Right",
                "Middle Left", "Middle Center", "Middle Right",
                "Bottom Left", "Bottom Center", "Bottom Right"], group = gTbl)
tblSizeStr   = input.string("Normal", "Text Size",
     options = ["Tiny", "Small", "Normal", "Large", "Huge"], group = gTbl)
tblHdrBg     = input.color(#39FF14,                     "Header Background", group = gTbl)
tblHdrText   = input.color(color.black,                 "Header Text",       group = gTbl)
tblBg        = input.color(color.new(#1e222d, 0),       "Cell Background",   group = gTbl)
tblText      = input.color(color.white,                 "Cell Text",         group = gTbl)
tblBorder    = input.color(color.new(color.gray, 50),   "Border Color",      group = gTbl)

// ─── Helpers ─────────────────────────────────────────────────────────────────
lineStyle = switch styleStr
    "Dashed" => line.style_dashed
    "Dotted" => line.style_dotted
    =>          line.style_solid

lblSize = switch sizeStr
    "Tiny"  => size.tiny
    "Small" => size.small
    "Large" => size.large
    "Huge"  => size.huge
    =>         size.normal

tblSize = switch tblSizeStr
    "Tiny"  => size.tiny
    "Small" => size.small
    "Large" => size.large
    "Huge"  => size.huge
    =>         size.normal

tblPos = switch tblPosStr
    "Top Left"      => position.top_left
    "Top Center"    => position.top_center
    "Top Right"     => position.top_right
    "Middle Left"   => position.middle_left
    "Middle Center" => position.middle_center
    "Middle Right"  => position.middle_right
    "Bottom Left"   => position.bottom_left
    "Bottom Center" => position.bottom_center
    =>                 position.bottom_right

parseLevels(string src) =>
    out = array.new<float>()
    cleaned = str.replace_all(src, ",", " ")
    cleaned := str.replace_all(cleaned, "\n", " ")
    cleaned := str.replace_all(cleaned, "\t", " ")
    parts = str.split(cleaned, " ")
    n = array.size(parts)
    if n > 0
        for i = 0 to n - 1
            tok = str.trim(array.get(parts, i))
            if str.length(tok) > 0
                v = str.tonumber(tok)
                if not na(v)
                    array.push(out, v)
    out

// ─── Symbol auto-detect ──────────────────────────────────────────────────────
tk = syminfo.ticker
isES = str.contains(tk, "ES") or str.contains(tk, "MES") or str.contains(tk, "SPX") or str.contains(tk, "SPY")
isNQ = str.contains(tk, "NQ") or str.contains(tk, "MNQ") or str.contains(tk, "NDX") or str.contains(tk, "QQQ")

drawES = mode == "ES"   or mode == "Both" or (mode == "Auto" and isES)
drawNQ = mode == "NQ"   or mode == "Both" or (mode == "Auto" and isNQ)

// ─── ATR ─────────────────────────────────────────────────────────────────────
atrVal = ta.atr(14)

// ─── Drawing: levels ─────────────────────────────────────────────────────────
var array<line>  storedLines  = array.new<line>()
var array<label> storedLabels = array.new<label>()

if barstate.islast
    while array.size(storedLines) > 0
        line.delete(array.pop(storedLines))
    while array.size(storedLabels) > 0
        label.delete(array.pop(storedLabels))

    esLevels = parseLevels(esInput)
    nqLevels = parseLevels(nqInput)

    levels = array.new<float>()
    if drawES
        nEs = array.size(esLevels)
        if nEs > 0
            for i = 0 to nEs - 1
                array.push(levels, array.get(esLevels, i))
    if drawNQ
        nNq = array.size(nqLevels)
        if nNq > 0
            for i = 0 to nNq - 1
                array.push(levels, array.get(nqLevels, i))

    nL = array.size(levels)
    if nL > 0
        for i = 0 to nL - 1
            p = array.get(levels, i)
            ln = line.new(bar_index - 1, p, bar_index, p,
                 extend = extend.both, color = lineColor,
                 style = lineStyle, width = lineWidth)
            array.push(storedLines, ln)
            if showLabels
                lb = label.new(bar_index + labelOffset, p,
                     text      = str.tostring(p, format.mintick),
                     style     = label.style_label_left,
                     color     = labelBg,
                     textcolor = labelText,
                     size      = lblSize,
                     xloc      = xloc.bar_index)
                array.push(storedLabels, lb)

// ─── Drawing: info table ─────────────────────────────────────────────────────
var table infoTbl = table.new(position.top_right, 2, 2, border_width = 1)

if barstate.islast and showTable
    table.clear(infoTbl, 0, 0, 1, 1)
    infoTbl := table.new(tblPos, 2, 2, border_width = 1, border_color = tblBorder)
    table.cell(infoTbl, 0, 0, "RIPS Levels",
         text_color = tblHdrText, bgcolor = tblHdrBg, text_size = tblSize)
    table.cell(infoTbl, 1, 0, "",
         text_color = tblHdrText, bgcolor = tblHdrBg, text_size = tblSize)
    table.merge_cells(infoTbl, 0, 0, 1, 0)
    table.cell(infoTbl, 0, 1, "ATR",
         text_color = tblText, bgcolor = tblBg, text_size = tblSize)
    table.cell(infoTbl, 1, 1, str.tostring(atrVal, format.mintick),
         text_color = tblText, bgcolor = tblBg, text_size = tblSize)
