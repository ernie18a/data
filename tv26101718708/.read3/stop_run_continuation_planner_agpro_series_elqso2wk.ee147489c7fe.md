Calculation: ATR value = max(nz(ATR(ATR lookback), TrueRange), minimum price increment)
Calculation: Upper reference = Highest(high[1], reference lookback)
Calculation: Lower reference = Lowest(low[1], reference lookback)
Calculation: Minimum sweep depth = max(sweep depth parameter + sensitivity adjustment, nonnegative floor)
Calculation: Close continuation buffer = continuation trigger buffer + sensitivity adjustment
Calculation: Volume ratio = volume / SMA(volume, volume lookback), or neutral volume ratio when the baseline is unavailable or zero
Calculation: Candle range = max(high - low, minimum price increment)
Calculation: Body efficiency = abs(close - open) / candle range * percentage scale
Calculation: Bull close location = (close - low) / candle range * percentage scale
Calculation: Bear close location = (high - close) / candle range * percentage scale
Calculation: Bull upper rejection = (high - close) / candle range * percentage scale
Calculation: Bear lower rejection = (close - low) / candle range * percentage scale
Calculation: Reference range = max(upper reference - lower reference, ATR value)
Calculation: Bull trigger = upper reference + ATR value * continuation trigger buffer
Calculation: Bear trigger = lower reference - ATR value * continuation trigger buffer
Calculation: Bull invalidation edge = upper reference - ATR value * invalidation buffer
Calculation: Bear invalidation edge = lower reference + ATR value * invalidation buffer
Calculation: Bull risk distance = max(bull trigger - bull invalidation edge, ATR value * minimum risk distance)
Calculation: Bear risk distance = max(bear invalidation edge - bear trigger, ATR value * minimum risk distance)
Calculation: Bull target = bull trigger + max(bull risk distance * target risk multiple, ATR value * fallback target distance, reference range * target range fraction)
Calculation: Bear target = bear trigger - max(bear risk distance * target risk multiple, ATR value * fallback target distance, reference range * target range fraction)
Calculation: Bull target room in ATR = max((bull target - close) / ATR value, nonnegative floor)
Calculation: Bear target room in ATR = max((close - bear target) / ATR value, nonnegative floor)
Calculation: Clamp(value, lower bound, upper bound) = min(max(value, lower bound), upper bound)
Calculation: NormalizeHigh(value, weak threshold, strong threshold) = Clamp((value - weak threshold) / max(strong threshold - weak threshold, normalization epsilon) * percentage scale, score floor, score ceiling)
Calculation: NormalizeLow(value, strong threshold, weak threshold) = Clamp((weak threshold - value) / max(weak threshold - strong threshold, normalization epsilon) * percentage scale, score floor, score ceiling)
Calculation: PeakScore(value, low bound, ideal value, high bound) = NormalizeHigh(value, low bound, ideal value) when value <= ideal value, otherwise NormalizeLow(value, ideal value, high bound)
Calculation: SweepScore(depth in ATR) = PeakScore(depth in ATR, minimum sweep depth, ideal sweep depth, maximum sweep depth)
Calculation: CloseScore(close distance in ATR, close location) = close distance weight * NormalizeHigh(close distance in ATR, weak close distance, strong close distance) + close location weight * NormalizeHigh(close location, sensitivity-adjusted minimum close quality, strong close location) + body weight * NormalizeHigh(body efficiency, weak body efficiency, strong body efficiency)
Calculation: VolumeScore = NormalizeHigh(volume ratio, weak volume ratio, strong volume ratio)
Calculation: TrendScore(supported, neutral) = strong trend score when supported, neutral trend score when neutral, otherwise weak trend score
Calculation: RoomScore(target room in ATR) = NormalizeHigh(target room in ATR, weak target room, clean target room)
Calculation: ReversalRisk(rejection, close score, supported, neutral) = Clamp(rejection weight * rejection + weak close weight * (score ceiling - close score) + (supported ? no trend penalty : neutral ? neutral trend penalty : weak trend penalty) + (volume ratio < weak volume threshold ? low volume penalty : no volume penalty), score floor, score ceiling)
Calculation: ContinuationScore(sweep score, close score, trend score, room score, reversal risk) = Clamp(sweep weight * sweep score + close weight * close score + volume weight * VolumeScore + trend weight * trend score + room weight * room score - reversal risk weight * reversal risk, score floor, score ceiling)
Calculation: Bull continuation score = when high > upper reference + ATR value * minimum sweep depth, ContinuationScore(SweepScore((high - upper reference) / ATR value), CloseScore((close - upper reference) / ATR value, bull close location), TrendScore(close > EMA(close, longer EMA lookback) AND EMA(close, shorter EMA lookback) >= EMA(close, longer EMA lookback), close > EMA(close, longer EMA lookback) OR EMA(close, shorter EMA lookback) >= EMA(close, longer EMA lookback)), RoomScore(bull target room in ATR), ReversalRisk(bull upper rejection, CloseScore((close - upper reference) / ATR value, bull close location), close > EMA(close, longer EMA lookback) AND EMA(close, shorter EMA lookback) >= EMA(close, longer EMA lookback), close > EMA(close, longer EMA lookback) OR EMA(close, shorter EMA lookback) >= EMA(close, longer EMA lookback)))
Calculation: Bear continuation score = when low < lower reference - ATR value * minimum sweep depth, ContinuationScore(SweepScore((lower reference - low) / ATR value), CloseScore((lower reference - close) / ATR value, bear close location), TrendScore(close < EMA(close, longer EMA lookback) AND EMA(close, shorter EMA lookback) <= EMA(close, longer EMA lookback), close < EMA(close, longer EMA lookback) OR EMA(close, shorter EMA lookback) <= EMA(close, longer EMA lookback)), RoomScore(bear target room in ATR), ReversalRisk(bear lower rejection, CloseScore((lower reference - close) / ATR value, bear close location), close < EMA(close, longer EMA lookback) AND EMA(close, shorter EMA lookback) <= EMA(close, longer EMA lookback), close < EMA(close, longer EMA lookback) OR EMA(close, shorter EMA lookback) <= EMA(close, longer EMA lookback)))
