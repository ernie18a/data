## Description

Short Description :
An advanced hybrid oscillator combining Heikin Ashi RSI (HARSI) candles with the CBC Flip trend tracking logic.

Long Description:
Overview
This experimental hybrid oscillator bridges two powerful concepts into a single sub-window (overlay = false) indicator: Heikin Ashi RSI Candles (HARSI) by @JayRogers and the CBC Flip trend engine by @friksa & @AsiaRoo. It is fully optimized and updated to Pine Script v6.

How It Works & Core Features:

HARSI Candles: Instead of standard price data, it converts the Relative Strength Index (RSI) into a zero-median Heikin Ashi candlestick structure. This filters out the internal noise of RSI movements, offering a highly readable view of momentum direction, overbought (OB), and oversold (OS) conditions.

CBC Flip Integration: The CBC Flip logic tracks the price action on the main chart and is layered into this oscillator in two clean visual ways:

Dynamic Background: Shifts color dynamically based on structural market control (Bullish/Bearish transitions).

Signal Shapes: Plots direct Triangle Up (Bulls) and Triangle Down (Bears) markers right on the HARSI candles when a momentum flip is confirmed.

Stochastic RSI Overlay: Optionally includes a scalable Stochastic RSI ribbon to gauge extreme exhaustion points within the channel.

How to Trade with It:

Trend & Momentum Alignment: Look for periods where the HARSI candles match the color of the CBC Flip background. Green candles during a Yellow background signify strong bullish continuation.

Flips & Reversals: Triangle shapes mark key technical turning points based on historical candle extremes.

OB/OS Extremes: The indicator oscillates around a zero-median line. The custom boundary channels highlight institutional overbought and oversold thresholds.

## Source Code

````pine
// This source code is free to use, copy, and alter in any way you choose.
// ...but credit is always nice :)
// HARSI Originally by @JayRogers
// CBC Flip Originally by @friksa & @AsiaRoo

//@version=6
indicator("HARSI+CBC", "HARSI+CBC", overlay = false, format = format.price, precision = 2)

////////////////////////////////////////////////////////////////////////////////
// ====== TOOLTIPS ======
////////////////////////////////////////////////////////////////////////////////
string TT_HARSI = "Period for the RSI calculations used to generate the candles."
string TT_PBIAS = "Smoothing feature for the OPEN of the HARSI candles."
string TT_SMRSI = "Smoothes the RSI in a manner similar to HA open, but uses the realtime rsi rather than the prior close value."
string TT_STOCH = "Uses the RSI generated by the above settings, and will be affected by the smoothing option."
string TT_STFIT = "Adjusts the vertical scaling of the stochastic."

////////////////////////////////////////////////////////////////////////////////
// ====== INPUTS ======
////////////////////////////////////////////////////////////////////////////////
// -- Candle config
string GROUP_CAND = "Config » HARSI Candles"
int i_lenHARSI     = input.int(14, "Length", group = GROUP_CAND, minval = 1, tooltip = TT_HARSI)
int i_smoothing    = input.int(1, "Open Smoothing", group = GROUP_CAND, minval = 1, maxval = 100, tooltip = TT_PBIAS)
string INLINE_COL  = "Colour Pallette"
color i_colUp      = input.color(color.teal, "Colour Pallette  ", group = GROUP_CAND, inline = INLINE_COL)
color i_colDown    = input.color(color.red, " ", group = GROUP_CAND, inline = INLINE_COL)
color i_colWick    = input.color(color.gray, " ", group = GROUP_CAND, inline = INLINE_COL)

// -- RSI plot config
string GROUP_PLOT = "Config » RSI Plot"
series float i_source = input.source(ohlc4, "Source", group = GROUP_PLOT)
int i_lenRSI       = input.int(7, "Length", group = GROUP_PLOT, minval = 1)
bool i_mode        = input.bool(true, "Smoothed Mode RSI?", group = GROUP_PLOT, tooltip = TT_SMRSI)
bool i_showPlot    = input.bool(true, "Show RSI Plot?", group = GROUP_PLOT)
bool i_showHist    = input.bool(true, "Show RSI Histogram?", group = GROUP_PLOT)

// -- Stochastic RSI plots config
string GROUP_STOCH = "Config » Stochastic RSI Plot"
string INLINE_STDS = "Stoch Draw States"
bool i_showStoch   = input.bool(false, "Show Stochastic? ", group = GROUP_STOCH, inline = INLINE_STDS, tooltip = TT_STOCH)
bool i_ribbon      = input.bool(true, "Ribbon?", group = GROUP_STOCH, inline = INLINE_STDS)
int i_smoothK      = input.int(3, "Smoothing K", group = GROUP_STOCH, minval = 1)
int i_smoothD      = input.int(3, "Smoothing D", group = GROUP_STOCH, minval = 1)
int i_stochLen     = input.int(14, "Stochastic Length", group = GROUP_STOCH, minval = 1)
int i_stochFit     = input.int(80, "Stoch Scaling %", group = GROUP_STOCH, minval = 1, maxval = 100, tooltip = TT_STFIT)

// -- Channel OB/OS config
string GROUP_CHAN = "Config » OB/OS Boundaries"
int i_upper        = input.int(20, "OB", group = GROUP_CHAN, inline = "OB", minval = 1, maxval = 50)
int i_upperx       = input.int(30, "OB Extreme", group = GROUP_CHAN, inline = "OB", minval = 1, maxval = 50)
int i_lower        = input.int(-20, "OS", group = GROUP_CHAN, inline = "OS", minval = -50, maxval = -1)
int i_lowerx       = input.int(-30, "OS Extreme", group = GROUP_CHAN, inline = "OS", minval = -50, maxval = -1)

// -- CBC Flip Config
string GROUP_CBC   = "Config » CBC Flip"
bool i_showCbcBg   = input.bool(true, "Show CBC Background?", group = GROUP_CBC)
bool i_showCbcShapes = input.bool(true, "Show CBC Shapes?", group = GROUP_CBC)

////////////////////////////////////////////////////////////////////////////////
// ====== FUNCTIONS & CALCULATIONS ======
////////////////////////////////////////////////////////////////////////////////

var bool cbc = false
if cbc and close < low[1]
    cbc := false
else if not cbc and close > high[1]
    cbc := true

f_zrsi(series float _src, int _len) => 
    ta.rsi(_src, _len) - 50.0

f_zstoch(series float _src, int _len, int _smooth, int _scale) =>
    float _zstoch   = ta.stoch(_src, _src, _src, _len) - 50.0
    float _smoothed = ta.sma(_zstoch, _smooth)
    float _scaled   = (_smoothed / 100.0) * _scale
    _scaled

f_rsi(series float _src, int _len, bool _m) =>
    float _zrsi = f_zrsi(_src, _len)
    var float _smoothed = na
    _smoothed := na(_smoothed[1]) ? _zrsi : (_smoothed[1] + _zrsi) / 2.0
    _m ? _smoothed : _zrsi

f_rsiHeikinAshi(int _len) =>
    float _closeRSI = f_zrsi(close, _len)
    float _openRSI  = nz(_closeRSI[1], _closeRSI)
    float _highRSI_raw  = f_zrsi(high, _len)
    float _lowRSI_raw   = f_zrsi(low, _len)
    float _highRSI  = math.max(_highRSI_raw, _lowRSI_raw)
    float _lowRSI   = math.min(_highRSI_raw, _lowRSI_raw)
    float _close    = (_openRSI + _highRSI + _lowRSI + _closeRSI) / 4.0
    
    var float _open = na
    _open  := na(_open[i_smoothing]) ? (_openRSI + _closeRSI) / 2.0 : ((_open[1] * i_smoothing) + _close[1]) / (i_smoothing + 1)
    
    float _high     = math.max(_highRSI, math.max(_open, _close))
    float _low      = math.min(_lowRSI,  math.min(_open, _close))
    [_open, _high, _low, _close]

////////////////////////////////////////////////////////////////////////////////
// ====== DATA PROCESSING ======
////////////////////////////////////////////////////////////////////////////////

float RSI = f_rsi(i_source, i_lenRSI, i_mode)
float StochK = f_zstoch(RSI, i_stochLen, i_smoothK, i_stochFit)
float StochD = ta.sma(StochK, i_smoothD)

[O, H, L, C] = f_rsiHeikinAshi(i_lenHARSI)

color bodyColour    = C > O ? i_colUp : i_colDown
color wickColour    = i_colWick
color colShadow     = color.rgb(0, 0, 0, 20)
color colNone       = color.rgb(0, 0, 0, 100)
color colRSI        = color.rgb(250, 200, 50, 0)
color colStochK     = color.new(#0094FF, 0)
color colStochD     = color.new(#FF6A00, 0)
color colStochFill  = StochK >= StochD ? color.new(colStochK, 50) : color.new(colStochD, 50)

////////////////////////////////////////////////////////////////////////////////
// ====== DRAWING AND PLOTTING ======
////////////////////////////////////////////////////////////////////////////////

// Channel Levels
upperx  = hline(i_upperx,  "OB Extreme",   color = color.new(color.silver, 60))
upper   = hline(i_upper,   "OB",           color = color.new(color.silver, 80))
median  = hline(0,         "Median",       color = color.orange, linestyle = hline.style_dotted)
lower   = hline(i_lower,   "OS",           color = color.new(color.silver, 80))
lowerx  = hline(i_lowerx,  "OS Extreme",   color = color.new(color.silver, 60))

// Channel Fills
fill(upper, upperx, color.new(color.red, 90),    title = "Background Fill OB")
fill(upper, lower,  color.new(color.blue, 90),   title = "Background Channel")
fill(lower, lowerx, color.new(color.green, 90),  title = "Background Fill OS")

// CBC Flip Background Color
bgcolor(i_showCbcBg ? (cbc and not cbc[1] ? color.rgb(255, 235, 59, 85) : not cbc and cbc[1] ? color.rgb(5, 185, 240, 85) : na) : na)

// RSI Histogram
plot(i_showHist ? RSI : na,    "RSI Histogram",    color = color.new(color.silver, 80),  style = plot.style_histogram)

// HARSI Candles
plotcandle(O, H, L, C, "HARSI", bodyColour, wickColour, bordercolor = bodyColour)

// CBC Flip Signal Shapes
plotshape(i_showCbcShapes and (not cbc and cbc[1]), style = shape.triangledown, location = location.abovebar, color = color.rgb(228, 102, 102), size = size.small, title = 'Bears')
plotshape(i_showCbcShapes and (cbc and not cbc[1]), style = shape.triangleup, location = location.belowbar, color = color.rgb(120, 206, 176), size = size.small, title = 'Bulls')

// RSI Plots
plot(i_showPlot ? RSI : na,                    "RSI Shadow",       color = colShadow,  linewidth = 3)
plot_rsi    = plot(i_showPlot ? RSI : na,      "RSI Overlay",      color = colRSI,     linewidth = 1)

// Stochastic RSI Plots & Ribbon Fill
plot(i_showStoch ? StochK : na,                "Stoch K Shadow",   color = not i_ribbon ? colShadow : colNone,  linewidth = 3)
plot(i_showStoch ? StochD : na,                "Stoch D Shadow",   color = not i_ribbon ? colShadow : colNone,  linewidth = 3)
plot_stochK = plot(i_showStoch ? StochK : na,  "Stoch K",          color = not i_ribbon ? colStochK : colNone,  linewidth = 1)
plot_stochD = plot(i_showStoch ? StochD : na,  "Stoch D",          color = not i_ribbon ? colStochD : colNone,  linewidth = 1)
fill(plot_stochK, plot_stochD, i_ribbon ? colStochFill : na, title = "Stoch Ribbon")

////////////////////////////////////////////////////////////////////////////////
// ====== ALERTS ======
////////////////////////////////////////////////////////////////////////////////
alertcondition(not cbc and cbc[1], 'Bears', 'Bears in Control')
alertcondition(cbc and not cbc[1], 'Bulls', 'Bulls in Control')
````
