## Description

[b]What it does:[/b]
Big traders push price down to hunt stop losses. When small traders get stopped out, big traders buy cheap  then price shoots up.
LiquiditySniper detects this exact moment and tells you when to enter.

[b]3 simple signals:[/b]
SWEEP  → stops being hunted
CONFIRM → buyers stepping in  
ENTER  → your entry point

[b]What you get:[/b]

Automatic liquidity zone detection
Sweep candle alert
Entry signal after confirmation
SL and TP lines drawn automatically
Built-in alerts for all signals

Settings:

Lookback Period — how far to detect zones
Zone Sensitivity — tight or wide zones
Confirmation Type — Candle Close / BOS / Both
Risk/Reward Ratio — auto calculates TP

Built for traders who want clean signals with zero noise[/i]

//@version=5
indicator("LiquiditySniper", overlay=true, max_bars_back=500)

// ─── SETTINGS ───────────────────────────────────────────
lookback     = input.int(20,    "Lookback Period",     minval=5, maxval=50)
sensitivity  = input.float(0.5, "Zone Sensitivity %",  minval=0.1, maxval=2.0, step=0.1)
confirm_type = input.string("Both", "Confirmation Type", options=["Candle Close", "BOS", "Both"])
show_sl      = input.bool(true,  "Show SL Line")
show_tp      = input.bool(true,  "Show TP Line")
rr_ratio     = input.float(2.0,  "Risk/Reward Ratio",  minval=1.0, maxval=5.0, step=0.5)

// ─── COLORS ─────────────────────────────────────────────
col_zone    = input.color(color.new(#f85149, 80), "Zone Color")
col_sweep   = color.new(#f85149, 0)
col_confirm = color.new(#3fb950, 0)
col_entry   = color.new(#ffd700, 0)
col_sl      = color.new(#f85149, 0)
col_tp      = color.new(#3fb950, 0)

// ─── VARIABLES ──────────────────────────────────────────
var float zone_level    = na
var float sweep_low     = na
var float entry_price   = na
var float sl_price      = na
var float tp_price      = na
var bool  swept         = false
var bool  confirmed     = false
var bool  bos_confirmed = false
var int   sweep_bar     = na
var line  sl_line       = na
var line  tp_line       = na

// ─── ZONE DETECTION ─────────────────────────────────────
is_swing_low = true
for i = 1 to lookback / 2
    if low[i] < low[lookback / 2]
        is_swing_low := false
    if low[lookback - i] < low[lookback / 2]
        is_swing_low := false

if is_swing_low
    zone_level := low[lookback / 2]

// ─── SWEEP DETECTION ────────────────────────────────────
tolerance = zone_level * (sensitivity / 100)

is_sweep = not na(zone_level) and low < (zone_level - tolerance) and close > (zone_level - tolerance)

if is_sweep and not swept
    swept         := true
    sweep_low     := low
    sweep_bar     := bar_index
    confirmed     := false
    bos_confirmed := false

// ─── CONFIRMATION: CANDLE CLOSE ABOVE ZONE ──────────────
candle_confirm = swept and not confirmed and close > zone_level and bar_index > sweep_bar

if candle_confirm
    confirmed := true

// ─── CONFIRMATION: BOS ──────────────────────────────────
prev_high = ta.highest(high, lookback)[1]

bos = swept and confirmed and not bos_confirmed and high > prev_high

if bos
    bos_confirmed := true
    entry_price   := close
    sl_price      := sweep_low - (zone_level - sweep_low) * 0.1
    tp_price      := close + (close - sl_price) * rr_ratio

// ─── ENTRY CONDITION ────────────────────────────────────
entry_signal = false

if confirm_type == "Candle Close"
    entry_signal := swept and candle_confirm
else if confirm_type == "BOS"
    entry_signal := swept and bos
else
    entry_signal := swept and bos_confirmed and bos

// ─── RESET after entry ──────────────────────────────────
if entry_signal
    swept         := false
    confirmed     := false
    bos_confirmed := false

// ─── PLOTS ──────────────────────────────────────────────
plot(zone_level, "Liquidity Zone", color=col_zone, style=plot.style_line, linewidth=1)

plotshape(is_sweep, title="Sweep", style=shape.labeldown, location=location.abovebar, color=col_sweep, textcolor=color.white, text="SWEEP", size=size.small)

plotshape(candle_confirm, title="Confirm", style=shape.labelup, location=location.belowbar, color=col_confirm, textcolor=color.white, text="CONFIRM", size=size.small)

plotshape(bos, title="BOS", style=shape.labelup, location=location.belowbar, color=color.new(#ffd700, 20), textcolor=color.black, text="BOS", size=size.small)

plotshape(entry_signal, title="Entry", style=shape.triangleup, location=location.belowbar, color=col_entry, size=size.normal)

plotshape(entry_signal, title="Entry Label", style=shape.labelup, location=location.belowbar, color=col_entry, textcolor=color.black, text="ENTER", size=size.small)

// ─── SL / TP LINES ──────────────────────────────────────
if entry_signal
    if show_sl
        sl_line := line.new(bar_index, sl_price, bar_index + 20, sl_price, color=col_sl, style=line.style_dashed, width=1)
    if show_tp
        tp_line := line.new(bar_index, tp_price, bar_index + 20, tp_price, color=col_tp, style=line.style_dashed, width=1)

// ─── SL / TP LABELS ─────────────────────────────────────
if entry_signal
    if show_sl
        label.new(bar_index + 1, sl_price, "SL: " + str.tostring(math.round(sl_price, 2)), color=color.new(col_sl, 80), textcolor=col_sl, style=label.style_label_right, size=size.small)
    if show_tp
        label.new(bar_index + 1, tp_price, "TP: " + str.tostring(math.round(tp_price, 2)), color=color.new(col_tp, 80), textcolor=col_tp, style=label.style_label_right, size=size.small)

// ─── ALERTS ─────────────────────────────────────────────
alertcondition(is_sweep, title="Liquidity Swept", message="LiquiditySniper: Liquidity zone swept on {{ticker}} {{interval}}")
alertcondition(entry_signal, title="Entry Signal", message="LiquiditySniper: Entry signal on {{ticker}} {{interval}} — Price: {{close}}")
alertcondition(candle_confirm, title="Confirmation", message="LiquiditySniper: Confirmation candle on {{ticker}} {{interval}}")
