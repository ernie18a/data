## Description

Power Earnings Gap (PEG) & HVE is a momentum and earnings-reaction indicator designed to identify stocks with powerful post-earnings gaps and unusually high-volume events.

The PEG module looks for earnings-related gap-ups that meet multiple quality filters: minimum gap percentage, high relative volume, price above the 200-day moving average, and strong close location within the day’s range.

The indicator also tracks whether the most recent earnings event qualified as a PEG and whether that event remains active within a user-defined lookback period. A small table shows the most recent PEG value and volume spike giving an indication of the momentum generated by the PEG. 

The HVE/HVY module identifies major volume events:
- HVE: Highest Volume Ever
- HVY: Highest Volume in a Year

This can help traders quickly spot stocks with institutional-style volume, earnings momentum, and recent high-impact price/volume behavior. 

Alerts are included for:
[list]
[*] New PEG detected
[*] Qualified PEG active
[*] HVE within lookback
[*] HVY within lookback
[/list]

All fields that drive the outcome are configurable.

## Source Code

````pine
// This Pine Script™ code is subject to the terms of the Mozilla Public License 2.0 at https://mozilla.org/MPL/2.0/
// © TXInvestor

//@version=6
indicator("Power Earnings Gap (PEG) & HVE", max_labels_count=500, overlay=true)

// --- User Inputs ---
gapMin       = input.float(10.0, "Min Gap %", minval=0.1)
volMult      = input.float(3.0, "Vol Multiplier (20-day MA)", minval=0.1)
lookbackX    = input.int(60, "Lookback Period (Days/Bars)", minval=1)
minClosePerc = input.float(50.0, "Min Close Strength %", minval=0.0, maxval=100.0)

// --- PEG Core Logic ---
// We check today and yesterday for earnings data availability
earningsVal      = request.earnings(syminfo.tickerid, earnings.actual, gaps=barmerge.gaps_on)
isEarningsWindow = not na(earningsVal) or not na(earningsVal[1])

// Technical Criteria
sma200 = ta.sma(close, 200)
isAbove200 = close > sma200
avgVol20 = ta.sma(volume, 20)
isHighVol = volume >= (avgVol20 * volMult)

// Price Action
gapPercent = ((open - close[1]) / close[1]) * 100
dayRange   = high - low
candleStrength = dayRange > 0 ? ((close - low) / dayRange) * 100 : 0

// The PEG Condition
isPEG = isEarningsWindow and gapPercent >= gapMin and isHighVol and isAbove200 and candleStrength >= minClosePerc

// --- State Machine & Double-Plot Prevention ---
var bool lastEarningsWasPEG = false
var float lastGap = 0.0
var float lastVol = 0.0
var int lastPegIndex = -1

// Logic to ensure the PEG only "triggers" once per event
// If a PEG is found and we haven't already marked one for this specific earnings event
bool triggerNewPeg = isPEG and (lastPegIndex == -1 or (bar_index - lastPegIndex > 1))

if isEarningsWindow
    if triggerNewPeg
        lastEarningsWasPEG := true
        lastGap := gapPercent
        lastVol := volume / avgVol20
        lastPegIndex := bar_index
    else if not isPEG and not na(earningsVal)
        // Reset only if a NEW earnings report fails the criteria
        // We ensure we aren't resetting on the "day after" a successful PEG
        if (bar_index - lastPegIndex > 1)
            lastEarningsWasPEG := false

// Lookback calculation
barsSincePEG = ta.barssince(triggerNewPeg)
isWithinLookback = not na(barsSincePEG) and barsSincePEG <= lookbackX

// FINAL QUALIFIER: 
// 1. The most recent earnings report encountered was a PEG
// 2. That report is within the user-defined lookback window
isQualified = lastEarningsWasPEG and isWithinLookback

// --- Visuals ---
// Only plot the label on the actual trigger bar to prevent double-labels
plotshape(triggerNewPeg, "PEG Found", shape.labelup, location=location.belowbar, color=color.green, text="PEG", textcolor=color.white)
bgcolor(isQualified ? color.new(color.green, 90) : na)

// --- Table Display ---
var table pegTable = table.new(position.top_right, 2, 2, border_width=1)
if barstate.islast
    if isQualified
        table.cell(pegTable, 0, 0, "Last PEG Gap", bgcolor=color.new(color.gray, 80), text_color=color.white)
        table.cell(pegTable, 1, 0, "Last PEG Vol", bgcolor=color.new(color.gray, 80), text_color=color.white)
        table.cell(pegTable, 0, 1, str.tostring(lastGap, "#.##") + "%", text_color=color.green)
        table.cell(pegTable, 1, 1, str.tostring(lastVol, "#.##") + "x", text_color=color.green)
    else
        table.delete(pegTable)

// --- Alerts ---
alertcondition(triggerNewPeg, title="New PEG Detected", message="New PEG found on {{ticker}}! Gap: {{plot_0}}%, Vol: {{plot_1}}x")
alertcondition(isQualified, title="Scanner: Qualified PEG Active", message="{{ticker}} is now a Qualified PEG (Last earnings was a PEG within lookback)")

// Hidden plots for alert placeholders
plot(gapPercent, "Gap Value", display=display.none)
plot(volume / avgVol20, "Vol Multiplier", display=display.none)


// Highest Volume Bar Logic

// --- Inputs ---
lookbackDays = input.int(30, "Alert Lookback (Days)", minval = 1, tooltip = "Triggers an alert if the HVE or HVY occurred within this many days.")
hvyWindow    = input.int(252, "HVY Window (Bars)", minval = 10, tooltip = "Number of bars to check for 'Highest Volume in a Year'.")

// --- Logic: Deep History Check ---
// We use nz() to ensure new stocks (like ARM) don't break the logic if they have < 250 weeks of data.
float weeklyMaxVol = request.security(syminfo.tickerid, "W", nz(ta.highest(volume, 250)[1], 0))

// --- Logic: Highest Volume Ever (HVE) ---
var float maxVolEver = 0.0
var int hveBarIndex  = na
var float hvePrice   = na
var int hveTime      = 0

// Current bar breaks all-time records
if volume > maxVolEver and volume > weeklyMaxVol
    maxVolEver  := volume
    hveBarIndex := bar_index
    hvePrice    := high
    hveTime     := time

// --- Logic: Highest Volume in a Year (HVY) ---
// Find the bar with the highest volume in the last 252 bars
int hvyOffset = -ta.highestbars(volume, hvyWindow)
int currentHvyIndex = bar_index - hvyOffset
float currentHvyVol = volume[hvyOffset]
float currentHvyPrice = high[hvyOffset]
int currentHvyTime = time[hvyOffset]

// --- Visuals: Labels ---
// Initialize labels only once
var label hveLabel = label.new(na, na, "", color = #5b9cf6, textcolor = chart.fg_color, style = label.style_label_down, size = size.small)
var label hvyLabel = label.new(na, na, "", color = #089981, textcolor = chart.fg_color, style = label.style_label_down, size = size.small)

// Update HVE Label
if not na(hveBarIndex)
    label.set_xy(hveLabel, hveBarIndex, hvePrice)
    label.set_text(hveLabel, "HVE")

// Update HVY Label
// We check if HVY is DIFFERENT from HVE. If HVE is na, we treat it as different.
bool isDifferentFromHve = na(hveBarIndex) or currentHvyIndex != hveBarIndex

if isDifferentFromHve
    label.set_xy(hvyLabel, currentHvyIndex, currentHvyPrice)
    label.set_text(hvyLabel, "HVY")
else
    label.set_x(hvyLabel, na) // Hide HVY if it is the same as HVE

// --- Alert / Scanning Logic ---
int lookbackMs = lookbackDays * 86400000

// HVE check: Within lookback window
bool hveInWindow = (time - hveTime) <= lookbackMs and hveTime != 0

// HVY check: Within lookback window
bool hvyInWindow = (time - currentHvyTime) <= lookbackMs and currentHvyTime != 0

// Alert conditions
alertcondition(hveInWindow, "HVE in Lookback", "The Highest Volume Ever occurred within the lookback period.")
alertcondition(hvyInWindow, "HVY in Lookback", "The Highest Volume in a Year occurred within the lookback period.")

// Visual indication
//bgcolor(hveInWindow ? color.new(#5b9cf6, 80) : (hvyInWindow ? color.new(#089981, 80) : na))
````
