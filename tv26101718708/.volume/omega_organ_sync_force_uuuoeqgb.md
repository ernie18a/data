## Source Code

````pine
//@version=6
indicator("🧠 OMEGA ORGAN SYNC FORCE", overlay=false, max_labels_count=500, max_lines_count=500)

//━━━━━━━━━━━━━━━━━━
// LANGUAGE
//━━━━━━━━━━━━━━━━━━
lang = input.string("EN", "Language / Dil", options=["EN", "TR"], group="🌐 Language")
txt(en, tr) => lang == "TR" ? tr : en

//━━━━━━━━━━━━━━━━━━
// ENGINE INPUTS
//━━━━━━━━━━━━━━━━━━
trendLen  = input.int(55, "Trend EMA Length", group="⚙️ Organ Engine")
momLen    = input.int(10, "Momentum ROC Length", group="⚙️ Organ Engine")
atrLenIn  = input.int(14, "ATR Length", group="⚙️ Organ Engine")
volEMALen = input.int(20, "Vol Direction EMA", group="⚙️ Organ Engine")
zLen      = input.int(34, "Z-Score Lookback", group="⚙️ Organ Engine")
smoothLen = input.int(8, "Base Smoothing", group="⚙️ Organ Engine")
flowLen   = input.int(13, "Organ Flow EMA", group="⚙️ Organ Engine")

//━━━━━━━━━━━━━━━━━━
// TIMEFRAME FORCE
//━━━━━━━━━━━━━━━━━━
tf1 = input.timeframe("5", "TF Force 1", group="🚀 Timeframe Force")
tf2 = input.timeframe("15", "TF Force 2", group="🚀 Timeframe Force")
tf3 = input.timeframe("60", "TF Force 3", group="🚀 Timeframe Force")

wTf1 = input.float(0.30, "TF1 Weight", step=0.05, group="🚀 Timeframe Force")
wTf2 = input.float(0.35, "TF2 Weight", step=0.05, group="🚀 Timeframe Force")
wTf3 = input.float(0.35, "TF3 Weight", step=0.05, group="🚀 Timeframe Force")

forceSmooth = input.int(8, "TF Force Smooth", group="🚀 Timeframe Force")

//━━━━━━━━━━━━━━━━━━
// THRESHOLDS
//━━━━━━━━━━━━━━━━━━
syncStrongTh    = input.float(3.0, "Sync Strong", step=0.1, group="📊 Thresholds")
syncFlipTh      = input.float(2.0, "Sync Flip", step=0.1, group="📊 Thresholds")
chaosLowTh      = input.float(2.0, "Chaos Low", step=0.1, group="📊 Thresholds")
chaosHighTh     = input.float(5.0, "Chaos High", step=0.1, group="📊 Thresholds")
chaosCollapseTh = input.float(1.5, "Chaos Collapse", step=0.1, group="📊 Thresholds")

//━━━━━━━━━━━━━━━━━━
// SAFETY
//━━━━━━━━━━━━━━━━━━
confirmClose = input.bool(true, "Confirm on Bar Close", group="🛡️ Safety")
isConf = confirmClose ? barstate.isconfirmed : true

//━━━━━━━━━━━━━━━━━━
// DISPLAY
//━━━━━━━━━━━━━━━━━━
showInfo    = input.bool(true, "Show Premium Panel", group="🎨 Display")
showSignals = input.bool(true, "Show Signals", group="🎨 Display")
showOrgans  = input.bool(false, "Show Individual Organs", group="🎨 Display")
showHLines  = input.bool(true, "Show Threshold Lines", group="🎨 Display")
showBg      = input.bool(true, "Show Regime Background", group="🎨 Display")
showGlow    = input.bool(true, "Show Glow Effect", group="🎨 Display")

//━━━━━━━━━━━━━━━━━━
// COLORS
//━━━━━━━━━━━━━━━━━━
bullColor  = input.color(color.lime, "Bull Color", group="🎨 Colors")
bearColor  = input.color(color.red, "Bear Color", group="🎨 Colors")
chaosColor = input.color(color.orange, "Chaos Color", group="🎨 Colors")
neutColor  = input.color(color.gray, "Neutral Color", group="🎨 Colors")
goldColor  = input.color(color.yellow, "Force Color", group="🎨 Colors")

//━━━━━━━━━━━━━━━━━━
// HELPERS
//━━━━━━━━━━━━━━━━━━
z(src, l) =>
    mean = ta.sma(src, l)
    dev  = ta.stdev(src, l)
    dev != 0 ? (src - mean) / dev : 0.0

clamp(v, lo, hi) =>
    math.max(math.min(v, hi), lo)

barMeter(v, maxv) =>
    n = int(math.round(clamp(v / maxv, 0.0, 1.0) * 8.0))
    n == 8 ? "████████" :
     n == 7 ? "███████░" :
     n == 6 ? "██████░░" :
     n == 5 ? "█████░░░" :
     n == 4 ? "████░░░░" :
     n == 3 ? "███░░░░░" :
     n == 2 ? "██░░░░░░" :
     n == 1 ? "█░░░░░░░" :
     "░░░░░░░░"

//━━━━━━━━━━━━━━━━━━
// ORGAN CALCULATION FUNCTION
//━━━━━━━━━━━━━━━━━━
organSync() =>
    emaTrend = ta.ema(close, trendLen)
    trendRaw = emaTrend - emaTrend[1]

    momentumRaw = ta.roc(close, momLen)

    obvVal = ta.obv
    obvRaw = obvVal - obvVal[1]

    atr = ta.atr(atrLenIn)
    priceDir = close > ta.ema(close, volEMALen) ? 1.0 : -1.0
    atrRaw = (atr - atr[1]) * priceDir

    trendZ = z(trendRaw, zLen)
    momentumZ = z(momentumRaw, zLen)
    obvZ = z(obvRaw, zLen)
    atrZ = z(atrRaw, zLen)

    syncScore = trendZ + momentumZ + obvZ + atrZ
    syncSmoothBase = ta.ema(syncScore, smoothLen)

    chaosRaw = (
         math.abs(trendZ - momentumZ) +
         math.abs(trendZ - obvZ) +
         math.abs(momentumZ - obvZ)
    ) / 3.0

    chaosSmoothBase = ta.ema(chaosRaw, smoothLen)

    [syncSmoothBase, chaosSmoothBase, trendZ, momentumZ, obvZ, atrZ]

//━━━━━━━━━━━━━━━━━━
// MAIN ORGAN ENGINE
//━━━━━━━━━━━━━━━━━━
[syncBase, chaosSmooth, trendZ, momentumZ, obvZ, atrZ] = organSync()

// Extra EMA flow smoothing: daha akıcı organ çizgisi
organFlow = ta.ema(syncBase, flowLen)
organSignal = ta.ema(organFlow, flowLen)

//━━━━━━━━━━━━━━━━━━
// TIMEFRAME FORCE ENGINE
//━━━━━━━━━━━━━━━━━━
tfSync1 = request.security(syminfo.tickerid, tf1, organFlow[1], gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_off)
tfSync2 = request.security(syminfo.tickerid, tf2, organFlow[1], gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_off)
tfSync3 = request.security(syminfo.tickerid, tf3, organFlow[1], gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_off)

tfBull1 = tfSync1 > 0
tfBull2 = tfSync2 > 0
tfBull3 = tfSync3 > 0

tfBear1 = tfSync1 < 0
tfBear2 = tfSync2 < 0
tfBear3 = tfSync3 < 0

bullCount = (tfBull1 ? 1 : 0) + (tfBull2 ? 1 : 0) + (tfBull3 ? 1 : 0)
bearCount = (tfBear1 ? 1 : 0) + (tfBear2 ? 1 : 0) + (tfBear3 ? 1 : 0)

tfForceRaw =
     tfSync1 * wTf1 +
     tfSync2 * wTf2 +
     tfSync3 * wTf3

tfForce = ta.ema(tfForceRaw, forceSmooth)

forceBull = tfForce > 0 and bullCount >= 2
forceBear = tfForce < 0 and bearCount >= 2

cascadeBull = bullCount == 3
cascadeBear = bearCount == 3

//━━━━━━━━━━━━━━━━━━
// PHASE / REGIME
//━━━━━━━━━━━━━━━━━━
phaseSpread = math.abs(organFlow - organSignal)
phaseExpansion = phaseSpread > phaseSpread[1]
phaseContraction = phaseSpread < phaseSpread[1]

regime =
     chaosSmooth > chaosHighTh ? "CHAOS" :
     math.abs(organFlow) > syncStrongTh and organFlow > 0 ? "TREND_UP" :
     math.abs(organFlow) > syncStrongTh and organFlow < 0 ? "TREND_DOWN" :
     "RANGE"

//━━━━━━━━━━━━━━━━━━
// SIGNALS
//━━━━━━━━━━━━━━━━━━
syncFlipUp = ta.crossover(organFlow, syncFlipTh)
syncFlipDown = ta.crossunder(organFlow, -syncFlipTh)

chaosCollapse =
     ta.crossunder(chaosSmooth, chaosCollapseTh) and
     ta.highest(chaosSmooth, 10) > chaosHighTh

strongBuy =
     isConf and
     syncFlipUp and
     chaosSmooth < chaosLowTh and
     forceBull

strongSell =
     isConf and
     syncFlipDown and
     chaosSmooth < chaosLowTh and
     forceBear

forceBuy =
     isConf and
     organFlow > organSignal and
     forceBull and
     phaseExpansion

forceSell =
     isConf and
     organFlow < organSignal and
     forceBear and
     phaseExpansion

chaosClearBuy =
     isConf and
     chaosCollapse and
     organFlow > syncFlipTh and
     forceBull

chaosClearSell =
     isConf and
     chaosCollapse and
     organFlow < -syncFlipTh and
     forceBear

//━━━━━━━━━━━━━━━━━━
// COLORS
//━━━━━━━━━━━━━━━━━━
organColor =
     organFlow > syncStrongTh and phaseExpansion ? bullColor :
     organFlow > syncFlipTh ? color.new(bullColor, 20) :
     organFlow < -syncStrongTh and phaseExpansion ? bearColor :
     organFlow < -syncFlipTh ? color.new(bearColor, 20) :
     neutColor

forceColor =
     cascadeBull ? goldColor :
     cascadeBear ? color.fuchsia :
     forceBull ? color.new(bullColor, 15) :
     forceBear ? color.new(bearColor, 15) :
     color.new(neutColor, 20)

chaosC =
     chaosSmooth > chaosHighTh ? chaosColor :
     chaosSmooth < chaosLowTh ? color.new(bullColor, 10) :
     color.new(neutColor, 10)

//━━━━━━━━━━━━━━━━━━
// PREMIUM BACKGROUND
//━━━━━━━━━━━━━━━━━━
bgCol =
     regime == "CHAOS" ? color.new(chaosColor, 88) :
     cascadeBull and organFlow > 0 ? color.new(bullColor, 92) :
     cascadeBear and organFlow < 0 ? color.new(bearColor, 92) :
     regime == "TREND_UP" ? color.new(bullColor, 95) :
     regime == "TREND_DOWN" ? color.new(bearColor, 95) :
     na

bgcolor(showBg ? bgCol : na)

//━━━━━━━━━━━━━━━━━━
// PLOTS - GLOW LAYERS
//━━━━━━━━━━━━━━━━━━
hline(0, "Zero", color=color.new(color.gray, 50))

plot(showGlow ? organFlow : na, "Organ Glow Outer", color=color.new(organColor, 88), linewidth=9)
plot(showGlow ? organFlow : na, "Organ Glow Mid", color=color.new(organColor, 78), linewidth=6)
plot(organFlow, "Organ Flow", color=organColor, linewidth=3)

plot(organSignal, "Organ EMA Signal", color=color.new(color.white, 15), linewidth=2)

plot(showGlow ? tfForce : na, "TF Force Glow", color=color.new(forceColor, 82), linewidth=7)
plot(tfForce, "Timeframe Force", color=forceColor, linewidth=3)

plot(chaosSmooth, "Chaos Flow", color=chaosC, linewidth=2)

// Threshold lines
plot(showHLines ? syncStrongTh : na, "Strong +", color=color.new(bullColor, 60), style=plot.style_circles)
plot(showHLines ? -syncStrongTh : na, "Strong -", color=color.new(bearColor, 60), style=plot.style_circles)
plot(showHLines ? syncFlipTh : na, "Flip +", color=color.new(bullColor, 75), style=plot.style_cross)
plot(showHLines ? -syncFlipTh : na, "Flip -", color=color.new(bearColor, 75), style=plot.style_cross)
plot(showHLines ? chaosHighTh : na, "Chaos High", color=color.new(chaosColor, 50), style=plot.style_circles)
plot(showHLines ? chaosLowTh : na, "Chaos Low", color=color.new(chaosColor, 75), style=plot.style_cross)
````
