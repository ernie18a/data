## Description

MACD-based dynamic zoning indicator that derives key price levels by normalizing the current market price through a division factor of 500, creating baseline zones that adapt proportionally across instruments and timeframes. These zones act as lightweight reference bands for identifying equilibrium regions, mean reversion opportunities, and subtle momentum pauses. To enhance signal clarity, stronger zones are generated by applying a multiplier of 4 to the derived values, highlighting high-confluence areas where price reactions, breakouts, or reversals are statistically more likely. The indicator is designed to be non-intrusive yet structurally informative, allowing traders to overlay it seamlessly with MACD signals, trend analysis, or price action strategies. Particularly useful in volatile markets, it helps contextualize MACD momentum shifts within a spatial framework, enabling better decision-making for entries, exits, and risk management. Suitable for intraday as well as swing trading scenarios.

````pine
// This Pine Script® code is subject to the terms of the Mozilla Public License 2.0 at https://mozilla.org/MPL/2.0/

//@version=6
indicator('Custom MACD RSL', shorttitle = 'MACD RSL', format = format.price, timeframe = '', timeframe_gaps = true)

// === Inputs ===
fastLength = input.int(12, 'Fast Length')
slowLength = input.int(26, 'Slow Length')
signalLength = input.int(9, 'Signal Smoothing')
src = input.source(close, 'Source')
//mmmm
// === MACD Calculation ===
fastMA = ta.ema(src, fastLength)
slowMA = ta.ema(src, slowLength)
macd = fastMA - slowMA
signal = ta.ema(macd, signalLength)
hist = macd - signal

// === Day close (zones based on daily close even on intraday TF) ===
dayClose = request.security(syminfo.tickerid, 'D', close, lookahead = barmerge.lookahead_on)

// === Adaptive Neutral Zone Logic ===
neutralLevel = if dayClose >= 20000 and dayClose <= 50000
    5.0
else if dayClose > 50000
    10.0
else
    dayClose / 500.0

upperZone = neutralLevel
lowerZone = -neutralLevel
upperExtreme = neutralLevel * 4.0
lowerExtreme = -neutralLevel * 4.0

// === MACD & Signal Lines ===
plot(macd, title = 'MACD Line', linewidth = 2, color = color.new(#00b7ff, 0))
plot(signal, title = 'Signal Line', linewidth = 2, color = color.new(#ffee00, 0))

// === Histogram Color Logic (rising = brighter, falling = darker) ===
rising = hist > hist[1]

// Brighter when rising
bull_rising = color.new(#00ff22, 0)
bear_rising = color.new(#7f0000, 50)

// Darker/faded when falling
bull_falling = color.new(#007f11, 50)
bear_falling = color.new(#ff0000, 0)

histColor = hist >= 0 ? rising ? bull_rising : bull_falling : rising ? bear_rising : bear_falling

// === Visible Histogram Bars ===
plot(hist, title = 'Histogram', style = plot.style_columns, color = histColor)

// === Zero line ===
hline(0, 'Zero Line', color = color.new(color.gray, 70))

// === Zone boundaries (same colors, thicker & brighter) ===
plot(upperZone, title = 'Neutral Upper (DayClose/500)', color = color.new(#00ff22, 0), linewidth = 2)
plot(lowerZone, title = 'Neutral Lower (-DayClose/500)', color = color.new(#ff0000, 0), linewidth = 2)
plot(upperExtreme, title = 'Extreme Upper (x4 Neutral)', color = color.new(#00660e, 0), linewidth = 3)
plot(lowerExtreme, title = 'Extreme Lower (x4 Neutral)', color = color.new(#810000, 0), linewidth = 3)

// === BG Highlight based on MACD crossing Zones (NOT histogram) ===
isBullBreak = macd > upperZone
isBearBreak = macd < lowerZone

bgColor = isBullBreak ? color.new(color.green, 85) : isBearBreak ? color.new(color.red, 85) : na
bgcolor(bgColor, title = 'MACD Zone Break Highlight')

// === PCO / NCO Signals ===
pco = ta.crossover(macd, signal) and macd > lowerZone
nco = ta.crossunder(macd, signal) and macd < upperZone

plotshape(pco, title = 'PCO', text = '▲PCO', style = shape.labelup, color = color.new(color.green, 0), textcolor = color.black, size = size.tiny, location = location.bottom)
plotshape(nco, title = 'NCO', text = '▼NCO', style = shape.labeldown, color = color.new(color.red, 0), textcolor = color.white, size = size.tiny, location = location.top)
````
