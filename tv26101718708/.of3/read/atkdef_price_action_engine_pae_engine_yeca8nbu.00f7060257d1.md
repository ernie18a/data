Calculation: SignedBodyFraction = if high > low then (close - open) / (high - low) else 0
Calculation: UpperWickFraction = if high > low then (high - max(open, close)) / (high - low) else 0
Calculation: LowerWickFraction = if high > low then (min(open, close) - low) / (high - low) else 0
Calculation: RelativeVolume = if SMA(volume, P_volumeAverageLookback) > 0 then volume / SMA(volume, P_volumeAverageLookback) else 1
Calculation: ChaikinMoneyFlow = CMF(high, low, close, volume, P_flowLookback)
Calculation: PriceChangeDirection = sign(close - close[1])
Calculation: MoneyFlowChangeDirection = sign(ChaikinMoneyFlow - ChaikinMoneyFlow[1])
Calculation: EMATrendSide = sign(EMA(close, P_shortTrendLookback) - EMA(close, P_longTrendLookback)), where P_shortTrendLookback is shorter than P_longTrendLookback
Calculation: TurnScore = if PriceChangeDirection != PriceChangeDirection[1] and PriceChangeDirection != 0 and MoneyFlowChangeDirection != MoneyFlowChangeDirection[1] and MoneyFlowChangeDirection != 0 and RelativeVolume > P_volumeSurgeThreshold and abs(SignedBodyFraction) > P_bodyConfirmationThreshold then clamp((PriceChangeDirection + MoneyFlowChangeDirection) / 2 * P_bothTurnMultiplier, -P_turnClamp, P_turnClamp) else if ((PriceChangeDirection != PriceChangeDirection[1] and PriceChangeDirection != 0) or (MoneyFlowChangeDirection != MoneyFlowChangeDirection[1] and MoneyFlowChangeDirection != 0)) and RelativeVolume > P_volumeSurgeThreshold and abs(SignedBodyFraction) > P_bodyConfirmationThreshold then clamp((PriceChangeDirection + MoneyFlowChangeDirection) / 2 * P_oneTurnMultiplier, -P_turnClamp, P_turnClamp) else 0
Calculation: MomentumScore = clamp(SMA(max(SignedBodyFraction, 0) * clamp(RelativeVolume, P_volumeRatioFloor, P_volumeRatioCeiling), P_momentumSmoothingLookback) - SMA(max(-SignedBodyFraction, 0) * clamp(RelativeVolume, P_volumeRatioFloor, P_volumeRatioCeiling), P_momentumSmoothingLookback), -P_momentumClamp, P_momentumClamp)
Calculation: CompositeBehaviorScore = clamp(P_shortMomentumWeight * ((close - close[P_shortPriceLookback]) / close * 100) + P_midMomentumWeight * ((close - close[P_midPriceLookback]) / close * 100) + P_accelerationWeight * (((close - close[P_shortPriceLookback]) / close * 100) - ((close[P_shortPriceLookback] - close[2 * P_shortPriceLookback]) / close[P_shortPriceLookback] * 100)) + P_volumeMomentumWeight * ((SMA(volume, P_volumeAverageLookback) - SMA(volume, P_volumeAverageLookback)[P_volumeMomentumLag]) / SMA(volume, P_volumeAverageLookback) * 100) + P_flowMomentumWeight * ChaikinMoneyFlow * P_flowScale, -P_behaviorClamp, P_behaviorClamp), where P_midPriceLookback is longer than P_shortPriceLookback
Calculation: FalseLiquidityPush = (SignedBodyFraction > P_pushBodyThreshold and RelativeVolume > P_volumeSurgeThreshold and UpperWickFraction > P_fakeWickThreshold and (EMATrendSide <= 0 or close <= Highest(high, P_breakoutLookback)[1])) or (SignedBodyFraction < -P_pushBodyThreshold and RelativeVolume > P_volumeSurgeThreshold and LowerWickFraction > P_fakeWickThreshold and (EMATrendSide >= 0 or close >= Lowest(low, P_breakoutLookback)[1]) )
