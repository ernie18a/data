## Description

# Breakout Retest Detector Pro

Breakout Retest Detector Pro is a price-action indicator designed to identify one of the highest-probability market behaviors: a breakout followed by a retest of the broken level.

The indicator automatically detects significant support and resistance levels using swing highs and swing lows. When price breaks above resistance or below support with sufficient momentum, the indicator records the breakout level and begins monitoring for a retest.

A bullish retest occurs when price breaks above resistance, pulls back to the former resistance level, and successfully holds it as new support before continuing higher. A bearish retest occurs when price breaks below support, rallies back to the former support level, and rejects it as new resistance before continuing lower.

To improve signal quality, the indicator incorporates multiple confirmation filters:

• Pivot-based support and resistance detection
• ATR-based breakout confirmation to filter weak breakouts
• ATR-based retest tolerance to adapt to market volatility
• Optional EMA trend filter to align trades with the dominant trend
• Optional volume confirmation filter
• Bullish and bearish rejection candle confirmation
• Fakeout protection to reduce false retest signals
• Real-time alerts for breakouts and confirmed retests

The indicator is particularly effective on NAS100, US100, XAUUSD, major forex pairs, indices, and other highly liquid markets. It works across multiple timeframes but generally performs best on 15-minute, 1-hour, 4-hour, and Daily charts where support and resistance levels carry greater significance.

Signals generated by the indicator are not intended to predict market direction on their own. Instead, they identify locations where institutional participation often becomes visible after a breakout. Traders can combine these signals with market structure, trend analysis, order flow, liquidity concepts, or risk management rules to build a complete trading strategy.

Key Signals:

• Bullish Breakout — Price closes decisively above resistance.
• Bearish Breakout — Price closes decisively below support.
• Bullish Retest Buy — Price retests and holds former resistance as support.
• Bearish Retest Sell — Price retests and rejects former support as resistance.

The goal of Breakout Retest Detector Pro is to help traders avoid chasing breakouts and instead focus on higher-quality entries that occur when the market confirms a breakout through a successful retest.

## Source Code

````pine
//@version=6
indicator("BOBO 1H Institutional Buy/Sell System", overlay=true, max_labels_count=500, max_lines_count=500)

// ===================== INPUTS =====================
fastEmaLen     = input.int(21, "Fast EMA", minval=1)
slowEmaLen     = input.int(50, "Slow EMA", minval=1)

rsiLen         = input.int(14, "RSI Length")
rsiBuyLevel    = input.int(55, "RSI Buy Level")
rsiSellLevel   = input.int(45, "RSI Sell Level")

atrLen         = input.int(14, "ATR Length")
atrMultSL      = input.float(1.5, "ATR Stop Loss Multiplier", step=0.1)
atrMinFilter   = input.float(0.5, "Minimum ATR Filter", step=0.1)

stAtrLen       = input.int(10, "Supertrend ATR Length")
stFactor       = input.float(3.0, "Supertrend Factor", step=0.1)

htfTf          = input.timeframe("240", "Higher Timeframe")
useHTF         = input.bool(true, "Use Higher Timeframe Confirmation")

adxLen         = input.int(14, "ADX Length")
adxSmooth      = input.int(14, "ADX Smoothing")
adxMin         = input.int(20, "Minimum ADX")
useADX         = input.bool(true, "Use ADX Filter")

pivotLen       = input.int(5, "Swing Pivot Strength", minval=2)
retestBars     = input.int(12, "Max Bars After BOS For Retest", minval=1)
retestTolerance = input.float(0.15, "Retest Tolerance %", step=0.05)

useSweep       = input.bool(true, "Use Liquidity Sweep Filter")
useRetest      = input.bool(true, "Use BOS Retest Filter")
useSession     = input.bool(false, "Use NY/London Session Filter")
sessionInput   = input.session("0700-1200", "Trading Session")

minScore       = input.int(75, "Minimum Signal Score", minval=0, maxval=100)

showTargets    = input.bool(true, "Show Entry / SL / TP")
showDashboard  = input.bool(true, "Show Dashboard")

// ===================== BASIC INDICATORS =====================
fastEma = ta.ema(close, fastEmaLen)
slowEma = ta.ema(close, slowEmaLen)

rsi = ta.rsi(close, rsiLen)
rsiRising = rsi > rsi[1]
rsiFalling = rsi < rsi[1]

atr = ta.atr(atrLen)
atrOk = atr >= atrMinFilter

[supertrend, stDirection] = ta.supertrend(stFactor, stAtrLen)
stBull = stDirection < 0
stBear = stDirection > 0

[diPlus, diMinus, adx] = ta.dmi(adxLen, adxSmooth)
adxOk = not useADX or adx >= adxMin

emaBull = fastEma > slowEma
emaBear = fastEma < slowEma

rsiBull = rsi > rsiBuyLevel and rsiRising
rsiBear = rsi < rsiSellLevel and rsiFalling

// ===================== HTF CONFIRMATION =====================
htfFastEma = request.security(syminfo.tickerid, htfTf, ta.ema(close, fastEmaLen))
htfSlowEma = request.security(syminfo.tickerid, htfTf, ta.ema(close, slowEmaLen))
[htfSt, htfDir] = request.security(syminfo.tickerid, htfTf, ta.supertrend(stFactor, stAtrLen))

htfBull = htfFastEma > htfSlowEma and htfDir < 0
htfBear = htfFastEma < htfSlowEma and htfDir > 0

htfBuyOk = not useHTF or htfBull
htfSellOk = not useHTF or htfBear

// ===================== SESSION FILTER =====================
inSession = not na(time(timeframe.period, sessionInput))
sessionOk = not useSession or inSession

// ===================== MARKET STRUCTURE =====================
pivotHigh = ta.pivothigh(high, pivotLen, pivotLen)
pivotLow  = ta.pivotlow(low, pivotLen, pivotLen)

lastSwingHigh = ta.valuewhen(not na(pivotHigh), pivotHigh, 0)
lastSwingLow  = ta.valuewhen(not na(pivotLow), pivotLow, 0)

bullBOS = not na(lastSwingHigh) and close > lastSwingHigh and close[1] <= lastSwingHigh
bearBOS = not na(lastSwingLow) and close < lastSwingLow and close[1] >= lastSwingLow

var float bullBreakLevel = na
var float bearBreakLevel = na
var int bullBosBar = na
var int bearBosBar = na

if bullBOS
    bullBreakLevel := lastSwingHigh
    bullBosBar := bar_index

if bearBOS
    bearBreakLevel := lastSwingLow
    bearBosBar := bar_index

bullRecentBOS = not na(bullBosBar) and bar_index - bullBosBar <= retestBars
bearRecentBOS = not na(bearBosBar) and bar_index - bearBosBar <= retestBars

bullRetest = bullRecentBOS and not na(bullBreakLevel) and low <= bullBreakLevel * (1 + retestTolerance / 100) and close > bullBreakLevel
bearRetest = bearRecentBOS and not na(bearBreakLevel) and high >= bearBreakLevel * (1 - retestTolerance / 100) and close < bearBreakLevel

retestBuyOk = not useRetest or bullRetest
retestSellOk = not useRetest or bearRetest

// ===================== LIQUIDITY SWEEP =====================
bullSweep = not na(lastSwingLow) and low < lastSwingLow and close > lastSwingLow
bearSweep = not na(lastSwingHigh) and high > lastSwingHigh and close < lastSwingHigh

sweepBuyOk = not useSweep or bullSweep or ta.barssince(bullSweep) <= 5
sweepSellOk = not useSweep or bearSweep or ta.barssince(bearSweep) <= 5

// ===================== SCORE SYSTEM =====================
buyScore = 0
buyScore += emaBull ? 15 : 0
buyScore += stBull ? 15 : 0
buyScore += rsiBull ? 15 : 0
buyScore += htfBuyOk ? 15 : 0
buyScore += adxOk ? 10 : 0
buyScore += atrOk ? 10 : 0
buyScore += sweepBuyOk ? 10 : 0
buyScore += retestBuyOk ? 10 : 0

sellScore = 0
sellScore += emaBear ? 15 : 0
sellScore += stBear ? 15 : 0
sellScore += rsiBear ? 15 : 0
sellScore += htfSellOk ? 15 : 0
sellScore += adxOk ? 10 : 0
sellScore += atrOk ? 10 : 0
sellScore += sweepSellOk ? 10 : 0
sellScore += retestSellOk ? 10 : 0

// ===================== FINAL SIGNALS =====================
buySignal =
     buyScore >= minScore and
     emaBull and
     stBull and
     rsiBull and
     htfBuyOk and
     adxOk and
     atrOk and
     sweepBuyOk and
     retestBuyOk and
     sessionOk

sellSignal =
     sellScore >= minScore and
     emaBear and
     stBear and
     rsiBear and
     htfSellOk and
     adxOk and
     atrOk and
     sweepSellOk and
     retestSellOk and
     sessionOk

// ===================== TRADE LEVELS =====================
buySL = close - atr * atrMultSL
buyTP1 = close + atr * atrMultSL
buyTP2 = close + atr * atrMultSL * 2
buyTP3 = close + atr * atrMultSL * 3

sellSL = close + atr * atrMultSL
sellTP1 = close - atr * atrMultSL
sellTP2 = close - atr * atrMultSL * 2
sellTP3 = close - atr * atrMultSL * 3

// ===================== PLOTS =====================
plot(fastEma, "Fast EMA", linewidth=2)
plot(slowEma, "Slow EMA", linewidth=2)
plot(supertrend, "Supertrend", linewidth=2)

plotshape(buySignal, title="BUY", text="BUY", style=shape.labelup, location=location.belowbar, size=size.normal, color=color.green, textcolor=color.white)
plotshape(sellSignal, title="SELL", text="SELL", style=shape.labeldown, location=location.abovebar, size=size.normal, color=color.red, textcolor=color.white)

plotshape(bullBOS, title="Bullish BOS", text="BOS", style=shape.triangleup, location=location.belowbar, size=size.tiny, color=color.lime)
plotshape(bearBOS, title="Bearish BOS", text="BOS", style=shape.triangledown, location=location.abovebar, size=size.tiny, color=color.orange)

plotshape(bullSweep, title="Bullish Sweep", text="SWEEP", style=shape.circle, location=location.belowbar, size=size.tiny, color=color.aqua)
plotshape(bearSweep, title="Bearish Sweep", text="SWEEP", style=shape.circle, location=location.abovebar, size=size.tiny, color=color.fuchsia)

// ===================== LABELS =====================
if buySignal and showTargets
    label.new(
        bar_index,
        low,
        "BUY\nScore: " + str.tostring(buyScore) +
        "\nEntry: " + str.tostring(close, format.mintick) +
        "\nSL: " + str.tostring(buySL, format.mintick) +
        "\nTP1: " + str.tostring(buyTP1, format.mintick) +
        "\nTP2: " + str.tostring(buyTP2, format.mintick) +
        "\nTP3: " + str.tostring(buyTP3, format.mintick),
        style=label.style_label_up,
        color=color.green,
        textcolor=color.white
    )

if sellSignal and showTargets
    label.new(
        bar_index,
        high,
        "SELL\nScore: " + str.tostring(sellScore) +
        "\nEntry: " + str.tostring(close, format.mintick) +
        "\nSL: " + str.tostring(sellSL, format.mintick) +
        "\nTP1: " + str.tostring(sellTP1, format.mintick) +
        "\nTP2: " + str.tostring(sellTP2, format.mintick) +
        "\nTP3: " + str.tostring(sellTP3, format.mintick),
        style=label.style_label_down,
        color=color.red,
        textcolor=color.white
    )

// ===================== DASHBOARD =====================
var table dash = table.new(position.top_right, 2, 8, border_width=1)

if showDashboard and barstate.islast
    table.cell(dash, 0, 0, "BOBO Signal Engine", text_color=color.white, bgcolor=color.black)
    table.cell(dash, 1, 0, "Status", text_color=color.white, bgcolor=color.black)

    table.cell(dash, 0, 1, "Buy Score")
    table.cell(dash, 1, 1, str.tostring(buyScore))

    table.cell(dash, 0, 2, "Sell Score")
    table.cell(dash, 1, 2, str.tostring(sellScore))

    table.cell(dash, 0, 3, "HTF Trend")
    table.cell(dash, 1, 3, htfBull ? "Bullish" : htfBear ? "Bearish" : "Neutral")

    table.cell(dash, 0, 4, "ADX")
    table.cell(dash, 1, 4, str.tostring(adx, "#.##"))

    table.cell(dash, 0, 5, "RSI")
    table.cell(dash, 1, 5, str.tostring(rsi, "#.##"))

    table.cell(dash, 0, 6, "ATR")
    table.cell(dash, 1, 6, str.tostring(atr, "#.##"))

    table.cell(dash, 0, 7, "Signal")
    table.cell(dash, 1, 7, buySignal ? "BUY" : sellSignal ? "SELL" : "WAIT")

// ===================== ALERTS =====================
alertcondition(buySignal, title="BOBO BUY Signal", message="BOBO Institutional BUY Signal")
alertcondition(sellSignal, title="BOBO SELL Signal", message="BOBO Institutional SELL Signal")
alertcondition(bullBOS, title="Bullish BOS", message="Bullish Break of Structure")
alertcondition(bearBOS, title="Bearish BOS", message="Bearish Break of Structure")
````
