//@version=5
indicator("Alishba 02 [CLEVER] - RSI Elite Toolkit", shorttitle="Alishba Elite", overlay=true)

// --- RSI Settings ---
group_rsi      = "RSI Configuration"
src            = input.source(close, "Source", group=group_rsi)
rsiLength      = input.int(14, "RSI Length", minval=1, group=group_rsi)
rsiOb          = input.int(80, "RSI Overbought", minval=50, maxval=100, group=group_rsi)
rsiOs          = input.int(20, "RSI Oversold",  minval=0,  maxval=50, group=group_rsi)
slopeThresh    = input.int(5, "Velocity Threshold", minval=1, group=group_rsi, tooltip="Minimum RSI change per bar to consider it a high-velocity signal")
depthThresh    = input.int(10, "Extreme Depth", minval=1, group=group_rsi, tooltip="How deep into OB/OS the RSI must go for a 'Deep' reversal signal (e.g. if 10, Buy needs RSI <= 10)")

// --- Trend Filter Settings (EMA 50/200) ---
group_trend    = "Trend Filter (EMA 50/200)"
useTrendFilter = input.bool(false, "Enable Trend Filter", group=group_trend, tooltip="Only Buy in Uptrend (Price > EMA 50/200) and Sell in Downtrend")
emaFastLen     = input.int(50, "Fast EMA Length", minval=1, group=group_trend)
emaSlowLen     = input.int(200, "Slow EMA Length", minval=1, group=group_trend)

// --- HTF RSI Settings ---
group_htf      = "Higher Timeframe Filter"
useHtfFilter   = input.bool(false, "Enable HTF Confirmation", group=group_htf, tooltip="Requires Higher Timeframe RSI to be in the same direction")
htfResolution  = input.timeframe("60", "HTF Resolution", group=group_htf)

// --- Visual Settings ---
group_vis      = "Visual Appearance"
showDash        = input.bool(true, "Show Performance Dashboard", group=group_vis)
showEma         = input.bool(false, "Show Trend EMAs on Chart", group=group_vis)
showBg          = input.bool(false, "Show Zone Highlights", group=group_vis)
signalCooldown  = input.int(5, "Signal Cooldown (Bars)", minval=1, group=group_vis, tooltip="Minimum bars between signals to avoid noise")
rsiBuyColor    = input.color(color.rgb(0, 230, 219), "Buy Signal Color", group=group_vis)
rsiSellColor   = input.color(color.rgb(243, 82, 255), "Sell Signal Color", group=group_vis)
strongColor    = input.color(color.rgb(255, 215, 0), "Strong Signal (Star)", group=group_vis)
rsiTextCol     = input.color(color.white, "Text Color", group=group_vis)

// --- Logic: Local Calculations ---
rsiValue = ta.rsi(src, rsiLength)
emaFast  = ta.ema(src, emaFastLen)
emaSlow  = ta.ema(src, emaSlowLen)

// --- Logic: HTF Calculations (Fixed to prevent repainting) ---
htfRsi = request.security(syminfo.tickerid, htfResolution, ta.rsi(src, rsiLength)[1], lookahead=barmerge.lookahead_off)

// --- Logic: Trend Conditions ---
isUptrend   = emaFast > emaSlow and close > emaFast
isDowntrend = emaFast < emaSlow and close < emaFast

htfBuyConfirm  = htfRsi > 50
htfSellConfirm = htfRsi < 50

// --- Logic: Velocity & Depth ---
rsiSlope = rsiValue - rsiValue[1]
isHighVelocity = math.abs(rsiSlope) >= slopeThresh

// Check depth of the last 3 bars
wasDeepOS = ta.lowest(rsiValue, 5)[1] <= (rsiOs - depthThresh)
wasDeepOB = ta.highest(rsiValue, 5)[1] >= (rsiOb + depthThresh)

// --- Logic: Cooldown System ---
var int lastSignalBar = 0
canSignal = (bar_index - lastSignalBar) >= signalCooldown

// --- Logic: Signals ---
rsiBuyTrigger  = ta.crossover(rsiValue, rsiOs)
rsiSellTrigger = ta.crossunder(rsiValue, rsiOb)

// Filtering
buySignal  = rsiBuyTrigger  and (useTrendFilter ? isUptrend : true) and (useHtfFilter ? htfBuyConfirm : true) and canSignal
sellSignal = rsiSellTrigger and (useTrendFilter ? isDowntrend : true) and (useHtfFilter ? htfSellConfirm : true) and canSignal

// Update lastSignalBar
if buySignal or sellSignal
    lastSignalBar := bar_index

// Signal Strength: Strong if all filters + Momentum AND Depth are met (Elite Strictness)
isStrongBuy  = buySignal  and isUptrend and htfBuyConfirm and isHighVelocity and wasDeepOS
isStrongSell = sellSignal and isDowntrend and htfSellConfirm and isHighVelocity and wasDeepOB

// --- Plotting ---
bgcolor(showBg ? (rsiValue >= rsiOb ? color.new(rsiSellColor, 90) : rsiValue <= rsiOs ? color.new(rsiBuyColor, 90) : na) : na)

plot(showEma ? emaFast : na, "EMA Fast", color=color.new(rsiBuyColor, 50))
plot(showEma ? emaSlow : na, "EMA Slow", color=color.new(rsiSellColor, 50))

// Signal Labels
if buySignal
    string txt = isStrongBuy ? "STRONG BUY ★" : "BUY"
    color  col = isStrongBuy ? strongColor : rsiBuyColor
    label.new(bar_index, low, text=txt + "\n" + str.tostring(rsiValue, "#.#"), color=col, textcolor=rsiTextCol, style=label.style_label_up, yloc=yloc.belowbar, size=size.small)

if sellSignal
    string txt = isStrongSell ? "STRONG SELL ★" : "SELL"
    color  col = isStrongSell ? strongColor : rsiSellColor
    label.new(bar_index, high, text=txt + "\n" + str.tostring(rsiValue, "#.#"), color=col, textcolor=rsiTextCol, style=label.style_label_down, yloc=yloc.abovebar, size=size.small)

// --- Performance Dashboard ---
var table dash = table.new(position.bottom_right, 2, 4, bgcolor=color.new(#1e222d, 10), border_color=#363c4e, border_width=1)
if barstate.islast and showDash
    color trendCol = isUptrend ? rsiBuyColor : isDowntrend ? rsiSellColor : color.gray
    table.cell(dash, 0, 0, "RSI (" + str.tostring(rsiLength) + ")", text_color=color.white, text_size=size.small)
    table.cell(dash, 1, 0, str.tostring(rsiValue, "#.#"), text_color=trendCol, text_size=size.normal)
    
    table.cell(dash, 0, 1, "HTF RSI (" + htfResolution + ")", text_color=color.white, text_size=size.small)
