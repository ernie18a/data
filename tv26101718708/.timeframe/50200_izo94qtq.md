//@version=6
indicator("50/200", overlay=true, max_labels_count=500)
//====================================================
// INPUTS
//====================================================
lotSize = input.int(100, "Lot Size")
showTable = input.bool(true, "Show Trade Table")

fastLength   = input.int(12, "MACD Fast Length")
slowLength   = input.int(26, "MACD Slow Length")
signalLength = input.int(9, "MACD Signal Length")

[macdLine, signalLine, _] = ta.macd(close, fastLength, slowLength, signalLength)
[wMacd, wSignal, _]       = request.security(syminfo.tickerid, "W", ta.macd(close, fastLength, slowLength, signalLength))


//====================================================
// SMA
//====================================================
sma25 = ta.sma(close, 25)
sma50 = ta.sma(close, 50)
sma200 = ta.sma(close, 200)

bgRed = sma50>sma200
bgcolor(bgRed ? color.new(#43b700, 80) : na)

A25 = plot(sma25, color=  color.new(#060606, 100) , linewidth=1, title="SMA 25")
A50 = plot(sma50, color=  color.blue , linewidth=2, title="SMA 50")
A200 = plot(sma200, color=color.red, linewidth=2, title="SMA 200")
fill(A50, A200, color=wMacd<wSignal ? color.new(#ff0000, 80) : color.new(color.red, 100))

longSignal = ta.crossover(sma50, sma200)
shortSignal = ta.crossunder(sma50, sma200)

bullZone = sma50 > sma200
bearZone = sma50 < sma200

bgRedStart = bgRed and not bgRed[1]
bgRedEnd   = not bgRed and bgRed[1]

//====================================================
// Lable
//====================================================
C = ta.crossover(macdLine, signalLine) and ((sma50>sma200 and wMacd<wSignal) or (sma50<sma200 and wMacd>wSignal))
if C
    label.new(
         bar_index, low * 0.95,
         "▲ " + str.tostring(close, "#,###.##"),
         style=label.style_label_up,
         color=color.new(#0000dd,60),
         textcolor=color.white,
         size=size.small)
//====================================================
// HEIKIN ASHI
//====================================================
haTicker = ticker.heikinashi(syminfo.tickerid)

haOpen = request.security(haTicker, timeframe.period, open)
haClose = request.security(haTicker, timeframe.period, close)

haRedToGreen = haClose > haOpen and haClose[1] < haOpen[1]
haGreenToRed = haClose < haOpen and haClose[1] > haOpen[1]

//====================================================
// VARIABLES
//====================================================
var string direction = "NONE"

var float firstEntryPrice = na
var float avgPrice = na
var float qty = 0

var int addCount = 0
var int entryTime = na

var float lastLongAddPrice = na
var float lastShortAddPrice = na

var float realizedPL = 0

var int wins = 0
var int losses = 0

var label dashboard = na

//====================================================
// TRADE HISTORY ARRAYS
//====================================================
var dirArr = array.new_string()
var plArr = array.new_float()
var plPctArr = array.new_float()
var dayArr = array.new_int()
var addArr = array.new_int()

var table tbl = table.new(
     position.middle_right,
     2,
     10,
     border_width = 1,
     border_color = color.gray)

addTrade(_dir, _pl, _plPct, _days, _adds)=>
    array.unshift(dirArr, _dir)
    array.unshift(plArr, _pl)
    array.unshift(plPctArr, _plPct)
    array.unshift(dayArr, _days)
    array.unshift(addArr, _adds)

    if array.size(dirArr) > 5
        array.pop(dirArr)
        array.pop(plArr)
        array.pop(plPctArr)
        array.pop(dayArr)
        array.pop(addArr)

//====================================================
// CLOSE SHORT
//====================================================
if longSignal and direction == "SHORT"

    tradePL = (avgPrice - close) * qty
    tradePct = ((avgPrice - close) / avgPrice) * 100
    tradeDays = math.round((time - entryTime) / 86400000)

    realizedPL += tradePL

    if tradePL > 0
        wins += 1
    else
        losses += 1

    addTrade("SHORT", tradePL, tradePct, tradeDays, addCount)

    label.new(
         bar_index,
         high*1.1,
         "SHORT CLOSED\nP/L: " +
         str.tostring(tradePL, "#,###.#") +
         "\n" +
         str.tostring(tradePct, "#.#") + "%",
         style=label.style_label_down,
         color=tradePL >= 0 ? color.rgb(0, 112, 4) : color.rgb(244, 3, 3),
         textcolor=color.white,size=size.small )

//====================================================
// CLOSE LONG
//====================================================
if shortSignal and direction == "LONG"

    tradePL = (close - avgPrice) * qty
    tradePct = ((close - avgPrice) / avgPrice) * 100
    tradeDays = math.round((time - entryTime) / 86400000)

    realizedPL += tradePL

    if tradePL > 0
        wins += 1
    else
        losses += 1

    addTrade("LONG", tradePL, tradePct, tradeDays, addCount)

    label.new(
         bar_index,
         high*1.1,
         "LONG CLOSED\nP/L: " +
         str.tostring(tradePL, "#,###.#") +
         "\n" +
         str.tostring(tradePct, "#.##") + "%",
         style=label.style_label_down,
         color=tradePL >= 0 ? color.rgb(0, 112, 4) : color.rgb(244, 3, 3),
         textcolor=color.white , size=size.small)

//====================================================
// NEW LONG
//====================================================
if longSignal

    direction := "LONG"
    firstEntryPrice := close
    avgPrice := close
    qty := lotSize
    addCount := 0
    entryTime := time

    lastLongAddPrice := na
    lastShortAddPrice := na

    label.new(
         bar_index,
         low*0.9,
         "LONG\n@" + str.tostring(close,"#,###.#"),
         style=label.style_label_up,
         color=color.rgb(1, 95, 4))

//====================================================
// NEW SHORT
//====================================================
if shortSignal

    direction := "SHORT"
    firstEntryPrice := close
    avgPrice := close
    qty := lotSize
    addCount := 0
    entryTime := time

    lastLongAddPrice := na
    lastShortAddPrice := na

    label.new(
         bar_index,
         low*0.9,
         "SHORT\n@" + str.tostring(close,"#,###.#"),
         style=label.style_label_up,
         color=color.rgb(144, 0, 0))

//====================================================
// LONG ADD RULE B
//====================================================
newLower =
     na(lastLongAddPrice)
     ? close < firstEntryPrice*0.95
     : close < lastLongAddPrice*0.95

if direction == "LONG" and close < firstEntryPrice and haRedToGreen and newLower

    avgPrice :=
         ((avgPrice * qty) + (close * lotSize)) /
         (qty + lotSize)

    qty += lotSize
    addCount += 1
    lastLongAddPrice := close

    label.new(
         bar_index,
         low,
         "▲" + str.tostring(addCount),
         style=label.style_label_up,
         color=color.green,
         textcolor=color.black,
         size=size.small)
//====================================================
// SHORT ADD RULE B
//====================================================
newHigher =
     na(lastShortAddPrice)
     ? close > firstEntryPrice*1.05
     : close > lastShortAddPrice*1.05

if direction == "SHORT" and close > firstEntryPrice and haGreenToRed and newHigher

    avgPrice :=
         ((avgPrice * qty) + (close * lotSize)) /
         (qty + lotSize)

    qty += lotSize
    addCount += 1
    lastShortAddPrice := close

    label.new(
         bar_index,
         high,
         "▼" + str.tostring(addCount),
         style=label.style_label_down,
         color=color.red,
         textcolor=color.white,
         size=size.small)
    
//====================================================
// ALERTS
//====================================================
alertcondition(longSignal, "Long Entry")
alertcondition(shortSignal, "Short Entry")
alertcondition(direction == "LONG" and haRedToGreen and newLower, "Add Long")
alertcondition(direction == "SHORT" and haGreenToRed and newHigher, "Add Short")

//====================================================
// LIVE LABEL
//====================================================
var label liveLabel = na

openPL =
     direction == "LONG"  ? (close - avgPrice) * qty :
     direction == "SHORT" ? (avgPrice - close) * qty :
     0

openPLPct =
     direction == "LONG" and avgPrice > 0 ?
         ((close - avgPrice) / avgPrice) * 100 :
     direction == "SHORT" and avgPrice > 0 ?
         ((avgPrice - close) / avgPrice) * 100 :
     0

daysInTrade =
     not na(entryTime)
     ? math.round((time - entryTime) / 86400000)
     : 0

annualReturn =
     daysInTrade > 0
     ? openPLPct / daysInTrade * 365
     : 0

labelColor =
     openPL > 0 ? color.rgb(2, 104, 6) :
     openPL < 0 ? color.rgb(247, 3, 3) :
     color.gray

if not na(liveLabel)
    label.delete(liveLabel)

liveLabel := label.new(
     bar_index + 5,
     high,
     "Mode: " + direction +
     "\nEntry: " + str.tostring(firstEntryPrice, "#,###.#") +
     "\nAvg: " + str.tostring(avgPrice, "#,###.#") +
     "\nQty: " + str.tostring(qty) +
     "\nAdds: " + str.tostring(addCount) +
     "\nDays: " + str.tostring(daysInTrade) +
     "\nOpen P/L: " + str.tostring(openPL, "#,###") +
     "\nReturn: " + str.tostring(openPLPct, "#.##") + "%" +
     "\nYr Return: " + str.tostring(annualReturn, "#.##") + "%",
     style=label.style_label_left,
     color=labelColor,
     textcolor=color.white,
     size=size.normal)

///
//====================================================
// WEEKLY MACD STRATEGY
//====================================================
var string wDirection = "NONE"

var float wEntryPrice = na
var float wQty = 0
var int wEntryTime = na

var float wRealizedPL = 0
var int wWins = 0
var int wLosses = 0

wLongSignal  = ta.crossover(wMacd, wSignal)
wShortSignal = ta.crossunder(wMacd, wSignal)

if wLongSignal and wDirection == "SHORT"

    wPL = (wEntryPrice - close) * wQty
    wRealizedPL += wPL

    if wPL > 0
        wWins += 1
    else
        wLosses += 1
if wShortSignal and wDirection == "LONG"

    wPL = (close - wEntryPrice) * wQty
    wRealizedPL += wPL

    if wPL > 0
        wWins += 1
    else
        wLosses += 1
if wLongSignal

    wDirection := "LONG"
    wEntryPrice := close
    wQty := lotSize
    wEntryTime := time

    label.new(
         bar_index, low * 0.95,
         "▲ " + str.tostring(close, "#,###.##"),
         style=label.style_label_up,
         color=color.new(#00dddd,60),
         textcolor=color.white,
         size=size.small)
if wShortSignal

    wDirection := "SHORT"
    wEntryPrice := close
    wQty := lotSize
    wEntryTime := time

    label.new(
         bar_index, high * 1.05,
         "▲ " + str.tostring(close, "#,###.##"),
         style=label.style_label_down,
         color=color.new(#FFf000,60),
         textcolor=color.white,
         size=size.small)

wOpenPL =
     wDirection == "LONG" ?
     (close - wEntryPrice) * wQty :

     wDirection == "SHORT" ?
     (wEntryPrice - close) * wQty :

     0

wDays =
     not na(wEntryTime)
     ? math.round((time - wEntryTime) / 86400000)
     : 0

wReturn =
     not na(wEntryPrice)
     ? wOpenPL / (wEntryPrice * wQty) * 100
     : 0
if barstate.islast and showTable

    table.clear(tbl, 0, 0, 1, 9)

    table.cell(tbl,0,0,"Status")
    table.cell(tbl,1,0,wDirection , bgcolor= wDirection== "LONG" ? color.green : color.red)

    table.cell(tbl,0,1,"Entry")
    table.cell(tbl,1,1,str.tostring(wEntryPrice,"#,###"))

    table.cell(tbl,0,2,"Qty")
    table.cell(tbl,1,2,str.tostring(wQty))

    table.cell(tbl,0,3,"Days")
    table.cell(tbl,1,3,str.tostring(wDays))

    table.cell(tbl,0,4,"P/L")
    table.cell(tbl,1,4,
         str.tostring(wOpenPL,"#,###"),
         text_color=color.white, bgcolor=wOpenPL >= 0 ? color.green : color.red)

    table.cell(tbl,0,5,"P/L%")
    table.cell(tbl,1,5,
         str.tostring(wReturn,"#.##")+"%",
         text_color=color.white, bgcolor=wReturn >= 0 ? color.green : color.red)
