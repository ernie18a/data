## Description

This indicator automates the classic [b]Stan Weinstein 4-Stage Market Cycle Analysis[/b] (from his legendary book [i]“Secrets for Profiting in Bull and Bear Markets”[/i]) using a custom context-aware layout engine. 

Most automated stage analysis scripts create immense visual clutter, resetting indicators and stacking labels on every minor bar cross. This script uses a human-like design philosophy, relying on multi-week absolute price structures to print exactly one clean, dynamic tracking label per macro phase.

[b]Human-Centric Visual Design[/b]
The indicator adapts its graphics engine based on the style of phase the market is currently experiencing:
[list]
[*] [b]Stage 1 (Accumulation)[/b]: Draws a clean, horizontal Dashed Blue Box around the absolute support/resistance boundary floor, with a floating blue label centered over the base.
[*] [b]Stage 2 (Markup)[/b]: Strips away consolidation frames and softly tints the entire chart vertical background Green to signify a clean, institutional buying regime.
[*] [b]Stage 3 (Distribution)[/b]: Places a precise, horizontal Dashed Orange Box around the macro topping range to track turning points.
[*] [b]Stage 4 (Markdown)[/b]: Softly tints the chart vertical background Red to warn traders away from catching a falling knife.
[/list]

[b]How the Boundary Engine Works[/b]
Instead of calculating mathematical slope percentages (which shift falsely on volatile weeks), the engine operates via a shifted Donchian Price Channel Model:
[list=1]
[*] [b]Breakout Tracking[/b]: It evaluates the asset's closing price against the Highest High / Lowest Low of the prior 20 weeks (using a [1] lookback offset to remove current bar skew).
[*] [b]Expansion Filtering[/b]: A true Stage 2 or Stage 4 regime shift will only trigger when price completely bursts out of the historical multi-month ceiling or floor line while confirming its position relative to the 30-Week Simple Moving Average.
[*] [b]Automatic Clean-Up[/b]: The script natively tracks historical nodes. The moment a new cycle phase is mathematically confirmed, it deletes overlapping data to keep your screen spotless.
[/list]

[b]Best Practices for Setup[/b]
[list]
[*] [b]Timeframe[/b]: To stay true to Weinstein's original methodology, always run this indicator on a Weekly (1W) chart.
[*] [b]Customization[/b]: Open the settings gear panel to adjust the Consolidation Range Lookback (Weeks) up or down depending on whether you want to focus on intermediate market movements or structural multi-year cycles.
[/list]

## Source Code

````pine
//@version=6
indicator("Weinstein Human-Like Stage Layout [v6]", overlay=true, max_boxes_count=100, max_labels_count=100)

// ==========================================
// 1. INPUTS & CONFIGURATION
// ==========================================
grp_stage  = "Macro Range Settings"
range_look = input.int(20, title="Consolidation Range Lookback (Weeks)", minval=5, group=grp_stage)
ma_len     = input.int(30, title="Baseline Stage MA Length", minval=1, group=grp_stage)

// ==========================================
// 2. MATHEMATICAL PRICE BOUNDARY ENGINE
// ==========================================
stage_ma = ta.sma(close, ma_len)

// Find the absolute highest ceiling and lowest floor of the last N weeks
range_high = ta.highest(high[1], range_look)
range_low  = ta.lowest(low[1], range_look)

// Persistent state variables
var int macro_stage = 4
var int stage_start_idx = bar_index
var int prev_macro_stage = 4

// Absolute Rule Execution
if ta.crossover(close, range_high) and close > stage_ma
    macro_stage := 2 // Confirmed Stage 2 Markup Breakout
else if ta.crossunder(close, range_low) and close < stage_ma
    macro_stage := 4 // Confirmed Stage 4 Markdown Breakdown
else if (macro_stage == 2 or macro_stage == 3) and close < stage_ma
    macro_stage := 3 // Churning at the top -> Stage 3 Distribution
else if (macro_stage == 4 or macro_stage == 1) and close > stage_ma
    macro_stage := 1 // Churning at the bottom -> Stage 1 Accumulation

stage_changed = macro_stage != macro_stage[1]

// ==========================================
// 3. SPOTS & GRAPHICS ENGINE (ONE LABEL PER ZONE)
// ==========================================
plot(stage_ma, color=color.blue, linewidth=2, title="30-Week Stage MA")

var box stage_box = na
var label stage_tag_label = na

// Track local extremes within this individual macro zone block
var float local_high = na
var float local_low = na

// Define explicit aesthetic colors for every stage capsule
color stage_txt_color = color.white
color stage_bg_color   = color.gray

if macro_stage == 1
    stage_txt_color := color.white
    stage_bg_color   := color.blue       // Stage 1: Blue Box / Capsule
else if macro_stage == 2
    stage_txt_color := color.white
    stage_bg_color   := color.green      // Stage 2: Green Capsule
else if macro_stage == 3
    stage_txt_color := color.black
    stage_bg_color   := color.orange     // Stage 3: Orange Box / Capsule
else if macro_stage == 4
    stage_txt_color := color.white
    stage_bg_color   := color.red        // Stage 4: Red Capsule

if stage_changed
    // FORCE CLEANUP: Instantly vaporize the old text label so they never crowd together
    if not na(stage_tag_label)
        label.delete(stage_tag_label)
        
    stage_start_idx := bar_index
    local_high      := high
    local_low       := low
    
    // Build an invisible framework block to manage chart space safely
    stage_box := box.new(left=stage_start_idx, top=local_high, right=bar_index, bottom=local_low,
                         bgcolor=color.new(color.white, 100), border_color=color.new(color.white, 100))
                         
    // Create ONE clean capsule text box above the center of this new zone
    string lbl_text = "Stage " + str.tostring(macro_stage)
    stage_tag_label := label.new(x=stage_start_idx, y=high * 1.05, text=lbl_text,
                                 color=stage_bg_color, textcolor=stage_txt_color, 
                                 style=label.style_label_down, size=size.normal)
else
    // Update local highs and lows dynamically
    local_high := math.max(high, na(local_high) ? high : local_high)
    local_low  := math.min(low, na(local_low) ? low : local_low)
    
    if not na(stage_box)
        box.set_right(stage_box, bar_index)
        
        // ONLY render a crisp dashed box framing when inside Stage 1 or Stage 3 consolidations
        if macro_stage == 1 or macro_stage == 3
            box.set_top(stage_box, local_high)
            box.set_bottom(stage_box, local_low)
            box.set_border_color(stage_box, macro_stage == 1 ? color.new(color.blue, 30) : color.new(color.orange, 30))
            box.set_border_style(stage_box, line.style_dashed)
            box.set_bgcolor(stage_box, macro_stage == 1 ? color.new(color.blue, 97) : color.new(color.orange, 97))
            
    if not na(stage_tag_label)
        // Keep the text label centered perfectly in the middle of the timeline block
        label.set_x(stage_tag_label, math.round(math.avg(stage_start_idx, bar_index)))
        label.set_y(stage_tag_label, local_high * 1.06)

// ==========================================
// 4. SOFT REGIME BACKGROUND TINTS
// ==========================================
color bg_tint = color.new(color.white, 100)
if macro_stage == 2
    bg_tint := color.new(color.green, 96) // Clear green background for uptrends
else if macro_stage == 4
    bg_tint := color.new(color.red, 96)   // Clear red background for downtrends

bgcolor(bg_tint, title="Stage Background Fill")
````
