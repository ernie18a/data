# WORBUX USDT.D Regime Feed

## Description

This Indicator was created to pull USDT.D feed via webhook for HTF filtering during live tests.
It can be used along with Python to store the current USDT.D trend on any time frame within a .csv file.

Example: Store within .csv file, the messages define different time frames. 
               {"source":"tv_test","feed":"USDT.D","tf":"1h","time":"{{time}}","close":{{close}}}
               {"source":"tv_test","feed":"USDT.D","tf":"4h","time":"{{time}}","close":{{close}}}

You may or may not find this useful..

//@version=5
indicator("WORBUX USDT.D Regime Feed", overlay=false, max_labels_count=500)

// =====================================================
// WORBUX USDT.D Regime Feed
// Sends USDT.D regime data to Python webhook.
// Symbol intended: CRYPTOCAP:USDT.D
// =====================================================

emaLen       = input.int(20, "EMA Length", minval=1)
slopeLookback = input.int(3, "EMA Slope Lookback", minval=1)
alertEveryBar = input.bool(false, "Alert Every Bar? Otherwise Daily Close Only")

usdtdClose = close
usdtdEma   = ta.ema(usdtdClose, emaLen)
emaSlope   = usdtdEma - usdtdEma[slopeLookback]

// Risk-on for crypto longs:
// USDT.D falling = stablecoin dominance falling = capital moving into crypto risk.
riskOn = emaSlope < 0

plot(usdtdClose, "USDT.D Close", color=color.white, linewidth=2)
plot(usdtdEma, "EMA", color=color.orange, linewidth=2)
bgcolor(riskOn ? color.new(color.green, 84) : color.new(color.red, 84))

plotchar(riskOn, title="Risk On", char="▲", location=location.bottom, color=color.lime, size=size.tiny)
plotchar(not riskOn, title="Risk Off", char="▼", location=location.top, color=color.red, size=size.tiny)

// Use confirmed daily close by default.
// If you run this on 1D chart, this updates once per candle close.
shouldAlert = barstate.isconfirmed
payload = '{"source":"tradingview","feed":"USDT.D","time":"' + str.tostring(time) + '","unix":' + str.tostring(time) + ',"close":' + str.tostring(usdtdClose) + ',"ema":' + str.tostring(usdtdEma) + ',"ema_slope":' + str.tostring(emaSlope) + ',"risk_on":' + str.tostring(riskOn) + ',"ema_len":' + str.tostring(emaLen) + ',"slope_lookback":' + str.tostring(slopeLookback) + '}'
if shouldAlert
    alert(payload, alert.freq_once_per_bar_close)
